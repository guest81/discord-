local function randomizeAutoSettings()
local ranges = {
{flag = "StrafeDistanceInput", min = 3, max = 12, decimals = 0},
{flag = "StrafeAmplitudeInput", min = 0.5, max = 2.5, decimals = 1},
{flag = "StrafeSpeedInput", min = 8, max = 25, decimals = 0},
{flag = "FollowDurationInput", min = 4, max = 15, decimals = 0},
{flag = "MinDistanceInput", min = 1, max = 4, decimals = 1},
{flag = "RotationSpeedInput", min = 0.2, max = 0.8, decimals = 2},
{flag = "AdhesionForceInput", min = 1.5, max = 5, decimals = 1},
} local function genValues()
local vals = {}
for _, r in ipairs(ranges) do
local raw = r.min + math.random() * (r.max - r.min)
local mult = 10 ^ r.decimals
vals[r.flag] = math.floor(raw * mult + 0.5) / mult
end return vals
end local newValues = genValues()
local attempts = 0
while lastRandomSettings and attempts < 10 do
local same = true
for flag, val in pairs(newValues) do
if lastRandomSettings[flag] ~= val then same = false; break end
end if not same then break end
newValues = genValues()
attempts += 1
end lastRandomSettings = newValues

end; end

local function checkWallTouch(hrp, char)
local params = RaycastParams.new()
params.FilterType = Enum.RaycastFilterType.Exclude
params.FilterDescendantsInstances = {char}
local rays = {
workspace:Raycast(hrp.Position, hrp.CFrame.LookVector * 2.8, params),
workspace:Raycast(hrp.Position, hrp.CFrame.RightVector * 2.4, params),
workspace:Raycast(hrp.Position, -hrp.CFrame.RightVector * 2.4, params),
workspace:Raycast(hrp.Position, (hrp.CFrame.LookVector + hrp.CFrame.RightVector).Unit * 2.5, params),
workspace:Raycast(hrp.Position, (hrp.CFrame.LookVector - hrp.CFrame.RightVector).Unit * 2.5, params)
} for _, ray in ipairs(rays) do
if ray and ray.Instance and ray.Instance.CanCollide then
return true, ray.Normal
end; end return false, nil
end local function startWallHop()
