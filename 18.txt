local function isGuid(n)

end

local function parseCounter(v) if type(v) ~= "string" then return end; local n = v:match("^X%-(%d+)$"); return n and tonumber(n) end
local function looksLikeState(t, r)
    if type(t) ~= "table" then return false end
    local hR, hM = false, false
    pcall(function() for _, v in pairs(t) do if v == r then hR = true elseif type(v) == "string" and v:match("^X%-%d+$") then hM = true end end end)
    return hR and hM
end
local function findBacState(r)
    for l = 2, 24 do
        local _, fn = pcall(debug.info, l, "f")
        if type(fn) == "function" then
            local _, ups = pcall(debug.getupvalues, fn)
            if type(ups) == "table" then
                for _, v in pairs(ups) do
                    if looksLikeState(v, r) then return v end
                    if type(v) == "table" then local n; pcall(function() for _, x in pairs(v) do if looksLikeState(x, r) then n = x; return end end end); if n then return n end end
                end
            end
        end
    end
end
local function mapBacState(st, a1, a2)
    local m = {}
    for k, v in pairs(st) do
        if type(v) == "string" then
            if v:match("^X%-%d+$") then m.marker = m.marker or k
            elseif a1 and v == a1 then m.arg1 = m.arg1 or k
            elseif a2 and v == a2 then m.arg2 = m.arg2 or k end
        end
    end
    return m
end
local bacModel = nil
local function digits(n) n = n % 1000; return math.floor(n/100), math.floor(n/10)%10, n%10 end
local function encodeBAC(m, c) local d1,d2,d3 = digits(c); return m.prefix..string.char(bxor(d1,m.k1),bxor(d2,m.k2),bxor(d3,m.k3)) end
local function learnBAC(r, a1, a2)
    local st = findBacState(r); if not st then return end
    local mp = mapBacState(st, a1, a2); if not mp.marker then return end
    local c = parseCounter(rawget(st, mp.marker)); if not c then return end
    local d1,d2,d3 = digits(c)
    local m = { state=st, map=mp, remote=r, prefix=a1:sub(1,9), k1=bxor(a1:byte(10),d1), k2=bxor(a1:byte(11),d2), k3=bxor(a1:byte(12),d3), offset=c-os.time(), arg2=a2 }
    if encodeBAC(m, c) == a1 then return m end
end
local function liveCounter(m)
    if m.state and m.map.marker then local _, raw = pcall(rawget, m.state, m.map.marker); local c = parseCounter(raw); if c and math.abs((c-os.time())-m.offset) <= 5 then return c end end
    return os.time() + m.offset
end

local function tryRequire(...)
    local cur = RS
    for _, name in ipairs({...}) do
        if not cur then return nil end
        cur = cur:FindFirstChild(name) or cur:WaitForChild(name, 4)
    end
    if not cur then return nil end
    local ok, m = pcall(require, cur); return ok and m or nil
end

local function findMod(name)
    for _, d in ipairs(RS:GetDescendants()) do
        if d:IsA("ModuleScript") and d.Name == name then
            local ok, m = pcall(require, d); if ok then return m end
        end
    end
end

local function findRemote(...)
    local parts = {...}
    for _, d in ipairs(RS:GetDescendants()) do
        if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then
            local ok = true
            for _, p in ipairs(parts) do if not d.Name:find(p, 1, true) then ok = false; break end end
            if ok then return d end
        end
    end
end

local function netCall(remote, ...)
    if not remote or typeof(remote) ~= "Instance" then return end
    if remote:IsA("RemoteFunction") then
        local ok, r = pcall(function() return remote:InvokeServer(...) end); return ok and r or nil
    elseif remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer(...) end)
    end
end

local function MoveStep(target, speed, ease)
    local hrp = findHRP(); if not hrp or not target then return end
    local start = hrp.Position; local dist = (target - start).Magnitude
    if dist < 1 then
        hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
        hrp.AssemblyLinearVelocity = Vector3.zero; return
    end
    speed = math.clamp(speed or 750, 50, 750)
    local t0 = os.clock(); local total = dist
    repeat
        local dt = RunService.Heartbeat:Wait(); if HUB.dead then break end
        local h2 = findHRP(); if not h2 then break end
        local to = target - h2.Position; local rem = to.Magnitude
        if rem < 1 then break end
        local ss = ease and math.max(speed*(1-math.clamp(1-rem/total,0,1)*0.8),35) or speed
        local dir = to.Unit
        h2.CFrame = CFrame.lookAt(h2.Position + dir*math.min(ss*math.min(dt,0.1),rem), h2.Position + dir*(math.min(ss*0.066,rem)+0.01))
        h2.AssemblyLinearVelocity = Vector3.zero; h2.AssemblyAngularVelocity = Vector3.zero
    until os.clock()-t0 > total/50+5
    local h3 = findHRP()
    if h3 then h3.CFrame = CFrame.new(target.X, math.max(target.Y,70), target.Z); h3.AssemblyLinearVelocity = Vector3.zero end
end

local function FlyStep(target, speed, ease)
    local hrp = findHRP(); if not hrp or not target then return end
    local start = hrp.Position; local dist = (target - start).Magnitude
    if dist < 1 then hrp.CFrame = CFrame.new(target.X,math.max(target.Y,70),target.Z); hrp.AssemblyLinearVelocity = Vector3.zero; return end
    speed = math.clamp(speed or 750, 50, 750)
    local mt = math.max(dist/speed,0.02)*(ease and 1.25 or 1)
    local t0 = os.clock(); local delta = target-start
    local dir = delta.Magnitude > 0.001 and delta.Unit or Vector3.new(1,0,0)
    repeat
        local dt = RunService.Heartbeat:Wait(); if HUB.dead then break end
        local la = math.clamp((os.clock()-t0)/mt,0,1)
        local a  = ease and math.sin(la*(math.pi/2)) or la
        local h2 = findHRP(); if not h2 then break end
        h2.CFrame = CFrame.lookAt(start:Lerp(target,a), start:Lerp(target,a)+dir)
        local cs = ease and math.max(speed*(1-la*0.8),35) or speed
        h2.AssemblyLinearVelocity = Vector3.new(dir.X*cs, math.clamp(dir.Y*cs,-15,150), dir.Z*cs)
        h2.AssemblyAngularVelocity = Vector3.zero
    until os.clock()-t0 >= mt
    local h3 = findHRP(); if h3 then h3.CFrame = CFrame.new(target.X,math.max(target.Y,70),target.Z); h3.AssemblyLinearVelocity = Vector3.zero end
end

local function NeutralizeTraps()
    local db = Workspace:FindFirstChild("__DEBRIS"); if not db then return end
    for _, d in ipairs(db:GetChildren()) do
        if d.Name == "PlayerTrap" and d:GetAttribute("Owner") ~= LP.Name then
            if d:IsA("BasePart") then d.CanTouch = false; d.CanQuery = false end
            for _, c in ipairs(d:GetChildren()) do
                if c:IsA("BasePart") then c.CanTouch = false; c.CanQuery = false
                    if c.Name == "Hitbox" then c.CFrame = CFrame.new(0,-999,0) end
                end
            end
            local tt = d:FindFirstChildWhichIsA("TouchTransmitter", true)
            if tt then pcall(function() tt:Destroy() end) end
        end
    end
end

local function TravelRoad(target, speed, ease)
    local hrp = findHRP(); if not hrp or not target then return end
    if avoidTraps then pcall(NeutralizeTraps) end
    local sp = hrp.Position; local sy = math.max(sp.Y, target.Y, 70.4)
    local isReturn = target.X < 560
    if isReturn and sp.X > SAFE_X then
        MoveStep(Vector3.new(sp.X,sy,ROAD_Z), speed)
        MoveStep(Vector3.new(SAFE_X,sy,ROAD_Z), speed)
        MoveStep(Vector3.new(target.X,sy,ROAD_Z), SAFE_SPEED)
        MoveStep(target+Vector3.new(0,1.2,0), SAFE_SPEED, ease)
    else
        MoveStep(Vector3.new(sp.X,sy,ROAD_Z), speed)
        MoveStep(Vector3.new(target.X,sy,ROAD_Z), speed)
        MoveStep(target+Vector3.new(0,1.2,0), speed, ease)
    end
end

local function TravelFly(target, speed, ease)
    local hrp = findHRP(); if not hrp or not target then return end
    if avoidTraps then pcall(NeutralizeTraps) end
    local sp = hrp.Position; local alt = math.max(sp.Y,target.Y,70.4)+28
    local isReturn = target.X < 560
    if isReturn and sp.X > SAFE_X then
        FlyStep(Vector3.new(sp.X,alt,sp.Z), speed)
        FlyStep(Vector3.new(SAFE_X,alt,ROAD_Z), speed)
        FlyStep(Vector3.new(SAFE_X,70.4,ROAD_Z), SAFE_SPEED)
        MoveStep(Vector3.new(target.X,70.4,ROAD_Z), SAFE_SPEED)
        MoveStep(target+Vector3.new(0,1.2,0), SAFE_SPEED, ease)
    else
        if (target-sp).Magnitude < 25 then FlyStep(target+Vector3.new(0,1.2,0), speed, ease); return end
        FlyStep(Vector3.new(sp.X,alt,sp.Z), speed)
        FlyStep(Vector3.new(target.X,alt,target.Z), speed)
        FlyStep(target+Vector3.new(0,1.2,0), speed, ease)
    end
end

local function TravelWalk(target)
    local hum = findHum(); local hrp = findHRP(); if not hum or not hrp or not target then return end
    local sp = hrp.Position
    for _, pt in ipairs({Vector3.new(sp.X,sp.Y,ROAD_Z), Vector3.new(target.X,target.Y,ROAD_Z), target+Vector3.new(0,1.2,0)}) do
        if HUB.dead then break end
        hum:MoveTo(pt); local t0 = os.clock()
        while findHRP() and (findHRP().Position-pt).Magnitude > 4.5 and os.clock()-t0 < 5 and not HUB.dead do task.wait(0.05) end
    end
end

local function travelTo(pos, ease)
    local sp = glideSpeed or 750
    if stealMethod == "Fly Glide"  then TravelFly(pos, sp, ease)
    elseif stealMethod == "Safe Walk" then TravelWalk(pos)
    else TravelRoad(pos, sp, ease) end
end

local function scoreEgg(egg)
    local rarity = getEggRarity(egg); local sc = RARITY_SCORE[rarity] or 0
    if stealPriority == "Rarest"      then return -sc
    elseif stealPriority == "Nearest" then local h=findHRP(); return h and egg.BoundsCFrame and (egg.BoundsCFrame.Position-h.Position).Magnitude or 9999
    elseif stealPriority == "Furthest" then local h=findHRP(); return h and egg.BoundsCFrame and -(egg.BoundsCFrame.Position-h.Position).Magnitude or 9999
    elseif stealPriority == "Biggest Size" then return -(tonumber(egg.AssetScale) or 1) end
    return -sc
end

local function passesFilter(egg)
    local rarity = getEggRarity(egg)
    if next(selectedStealRarities) ~= nil then
        local f = false; for _, r in ipairs(selectedStealRarities) do if rarity == r then f = true; break end end; if not f then return false end
    end
    if next(selectedStealAreas) ~= nil then
        local f = false; for _, a in ipairs(selectedStealAreas) do if egg.AreaId == a or egg.Area == a then f = true; break end end; if not f then return false end
    end
    if stealParasiteOnly and not (egg.IsParasite or egg.Parasite) then return false end
    if next(selectedMutations) ~= nil then
        local isMut = egg.IsMutated or egg.Mutated; local isPar = egg.IsParasite or egg.Parasite
        for _, m in ipairs(selectedMutations) do
            if m == "Normal Only"          and (isMut or isPar)            then return false end
            if m == "Mutated Only"         and not isMut                   then return false end
            if m == "Parasite / Infested"  and not isPar                   then return false end
            if m == "Rainbow Only"         and rarity ~= "Rainbow"         then return false end
            if m == "Gold Only"            and not (egg.IsGold or egg.Gold) then return false end
            if m == "Silver Only"          and not (egg.IsSilver or egg.Silver) then return false end
            if m == "Monstrous"            and not egg.IsMonstrous          then return false end
        end
    end
    if stealBigEggs and (tonumber(egg.AssetScale) or 1) < stealBigEggScale then return false end
    return true
end

local function pickTarget()
    local eggs = getFieldEggs(); local candidates = {}
    for _, egg in (type(eggs)=="table" and pairs or ipairs)(eggs) do
        if type(egg)=="table" and egg.Uid and egg.BoundsCFrame and (egg.State=="Slot" or egg.State==nil) then
            local ok = autoStealAll or (autoStealSelected and passesFilter(egg)) or (stealBigEggs and (tonumber(egg.AssetScale) or 1) >= stealBigEggScale)
            if ok and not ignoredEggs[egg.Uid] then table.insert(candidates, egg) end
        end
    end
    table.sort(candidates, function(a,b) return scoreEgg(a) < scoreEgg(b) end)
    return candidates[1]
end

local function StealEgg(egg)
    if not egg or not egg.Uid or not egg.BoundsCFrame then return false end
    local hrp = findHRP(); if not hrp then return false end

    local slotKey = nil
    if AreaEggSlotIdentity and AreaEggSlotIdentity.LooksLikeFirstAreaUid and AreaEggSlotIdentity.LooksLikeFirstAreaUid(egg.Uid) then
        slotKey = AreaEggSlotIdentity.SlotKey(egg.AreaId, egg.NestId)
    end

    travelTo(egg.BoundsCFrame.Position + Vector3.new(0,1.2,0), true); task.wait(0.4)

    if R.CarryEgg then
        if R.CarryEgg:IsA("RemoteFunction") then pcall(function() R.CarryEgg:InvokeServer({Uid=egg.Uid, FirstAreaSlotKey=slotKey}) end)
        else pcall(function() R.CarryEgg:FireServer(egg.Uid, slotKey) end) end
    end
    if EggState and EggState.CarryFieldEgg then pcall(EggState.CarryFieldEgg, egg.Uid, slotKey) end

    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Name == "CarryAreaEgg" and d.Enabled then
            local p = d.Parent; if p:IsA("Attachment") then p = p.Parent end
            local h = findHRP()
            if p and h and (p.Position-h.Position).Magnitude < 14 then
                d.HoldDuration = 0; pcall(function() fireproximityprompt(d) end); pcall(function() fireproximityprompt(d,0) end); break
            end
        end
    end

    local t0 = os.clock()
    while os.clock()-t0 < 2.5 and not HUB.dead do
        if isCarrying() then
            IsCarryingEgg = true; SessionStolenEggs = SessionStolenEggs + 1
            table.insert(ObtainedEggLog, getEggRarity(egg).." ("..tostring(egg.AssetCategory or "?")..")")
            if autoReturn then travelTo(GetLocalPlotCenter(), true); task.wait(0.4); PlantAllCarriedEggs() end
            IsCarryingEgg = isCarrying(); return true
        end
        if EggState and EggState.CarryFieldEgg then pcall(EggState.CarryFieldEgg, egg.Uid, slotKey) end
        task.wait(0.08)
    end

    ignoredEggs[egg.Uid] = os.clock()
    task.delay(30, function() ignoredEggs[egg.Uid] = nil end)
    return false
end

local function StealBestEggOnce()
    pcall(HatchAllReadyEggs)
    local target = pickTarget(); if not target then return false end
    return StealEgg(target)
end

local function AutoFusePets()
    if not R.FuseInsert or not R.FuseStart then return end
    local save = getSave(); if not save then return end
    local inv = save.Inventory or save.PetInventory; if type(inv)~="table" then return end
    local groups = {}
    for uid, pd in pairs(inv) do
        if type(pd)=="table" then
            local isMut = (pd.Mutations and next(pd.Mutations)~=nil) or pd.BaseMutation~=nil
            local scale = tonumber(pd.Scale) or 1
            local passR = next(fuseRarities)==nil; if not passR then for _,r in ipairs(fuseRarities) do if (pd.Rarity or "Common")==r then passR=true;break end end end
            if passR and scale<=fuseMaxScale and not(fuseKeepMutated and isMut) and not(fuseKeepEquipped and pd.Equipped) then
                local cat = pd.AssetCategory or "?"; groups[cat] = groups[cat] or {}; table.insert(groups[cat], uid)
            end
        end
    end
    for _, uids in pairs(groups) do
        if #uids >= 3 then
            local keep = math.max(fuseKeepPerCat, 0)
            local batches = math.floor((#uids-keep)/3)
            for b = 1, batches do
                local base = keep+(b-1)*3+1
                local a, bb, c = uids[base], uids[base+1], uids[base+2]
                if a and bb and c then
                    netCall(R.FuseInsert, a); task.wait(0.1)
                    netCall(R.FuseInsert, bb); task.wait(0.1)
                    netCall(R.FuseInsert, c); task.wait(0.2)
                    netCall(R.FuseStart); task.wait(0.5)
                    if fuseAutoReveal then if R.FuseAck then netCall(R.FuseAck) end; task.wait(0.1); if R.FuseReveal then netCall(R.FuseReveal) end end
                    task.wait(0.3)
                end
            end
        end
    end
end

-- SERVER HOP
-- ══════════════════════════════════════════════════════
local function serverHop()
    if ServerHopBusy or os.clock() < ServerHopRetryAt then return false end
    ServerHopBusy = true
    local ok, body = pcall(function() return game:HttpGet(("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):format(game.PlaceId)) end)
    if not ok then ServerHopBusy=false; return false end
    local dok, page = pcall(function() return HttpService:JSONDecode(body) end)
    if not dok or type(page)~="table" or type(page.data)~="table" then ServerHopBusy=false; return false end
    local best = nil
    for _, s in ipairs(page.data) do
        if type(s)=="table" and type(s.id)=="string" and s.id~=game.JobId and not VisitedServers[s.id] then
            local pl=tonumber(s.playing) or 0; local mx=tonumber(s.maxPlayers) or 0
            if mx>0 and pl<mx-1 then if not best or pl<(tonumber(best.playing) or 0) then best=s end end
        end
    end
    if best and best.id then
        VisitedServers[best.id]=true
        Notify("Server Hop","Hopping → "..tostring(best.playing).." players",3)
        pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId,best.id,LP) end)
        ServerHopBusy=false; return true
    end
    ServerHopRetryAt=os.clock()+20; ServerHopBusy=false
    Notify("Server Hop","No suitable server found",3); return false
end

        local fr=Vector3.new(rt.X,0,rt.Z); fr=fr.Magnitude>0.001 and fr.Unit or Vector3.new(1,0,0)
        local dir=Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir=dir+fl end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir=dir-fl end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir=dir-fr end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir=dir+fr end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir=dir+Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir=dir-Vector3.new(0,1,0) end
        if dir.Magnitude>0 then hrp.CFrame=hrp.CFrame+dir.Unit*flySpeed*math.min(dt,0.1) end
        gyro.CFrame=CFrame.lookAt(hrp.Position,hrp.Position+lk)
    end)) }
end

local function httpPost(payload)
    local requestFn=(syn and syn.request) or (http and http.request) or http_request or request
    if typeof(requestFn)~="function" or webhookUrl=="" then return false end
    local ok,body=pcall(function() return HttpService:JSONEncode(payload) end); if not ok then return false end
    return pcall(requestFn,{Url=webhookUrl,Method="POST",Headers={["Content-Type"]="application/json"},Body=body})
end

local function sendSummary()
    if not webhookEnabled or webhookUrl=="" then return false end
    local embed = {
        title="Session Summary",
        fields={
            {name="Player",value=LP.Name,inline=true},
            {name="Eggs Stolen",value=tostring(SessionStolenEggs),inline=true},
            {name="Runtime",value=string.format("%dm",math.floor((os.clock()-SessionStartedAt)/60)),inline=true},
        },
        color=3066993, timestamp=os.date("!%Y-%m-%dT%H:%M:%SZ"),
        footer={text="Steal An Egg Hub | WindUI"},
    }
    if webhookEggSpawns and #ObtainedEggLog>0 then
        local recent={}; for i=math.max(1,#ObtainedEggLog-9),#ObtainedEggLog do table.insert(recent,ObtainedEggLog[i]) end
        table.insert(embed.fields,{name="Recent Eggs",value=table.concat(recent,"\n"),inline=false})
    end
    local payload={username="Steal An Egg Hub",embeds={embed}}
    if webhookPingId~="" then payload.content=string.format("<@%s>",webhookPingId:gsub("%D","")) end
    return httpPost(payload)
end
