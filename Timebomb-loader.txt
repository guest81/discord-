local function MakeDraggablePill(frame)
	local dragging, dragInput, dragStart, startPos
	frame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	frame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and input == dragInput then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

local function IsAlly(char)
	if not char then return false end
	for _, v in ipairs(char:GetChildren()) do
		if v:IsA("Highlight") then
			local oc = v.OutlineColor
			if oc.G > 0.4 and oc.G > oc.R and oc.G > oc.B then return true end
			local fc = v.FillColor
			if fc.G > 0.4 and fc.G > fc.R and fc.G > fc.B then return true end
		end
		if v:IsA("SelectionBox") then
			local c = v.Color3
			if c.G > 0.4 and c.G > c.R and c.G > c.B then return true end
		end
	end
	return false
end

local HitboxPage = CreatePage("Hitbox")

local hitboxEnabled = false
local hitboxSize = 7
local transparentPlayers = true
local espEnabled = true
local hitboxData = {}
local espData = {}

local function clearChar(char)
	local ownerPlr = Players:GetPlayerFromCharacter(char)
	if ownerPlr == player then return end
	if hitboxData[char] then
		pcall(function() if hitboxData[char].box then hitboxData[char].box:Destroy() end end)
		if hitboxData[char].orig then
			for part, size in pairs(hitboxData[char].orig) do
				pcall(function() if part and part.Parent then part.Size = size end end)
			end
		end
		hitboxData[char] = nil
	end
	if espData[char] then
		pcall(function() espData[char].bb:Destroy() end)
		espData[char] = nil
	end
end

local function applyHitbox(char, plr)
	if plr == player then return end
	if not char or not char.Parent then return end
	if IsAlly(char) then clearChar(char) return end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	clearChar(char)
	local origSizes = {}

	if not hrp:GetAttribute("KHUB_OrigSize") then
		hrp:SetAttribute("KHUB_OrigSize", hrp.Size)
	end
	origSizes[hrp] = hrp:GetAttribute("KHUB_OrigSize")
	hrp.Size = Vector3.new(hitboxSize, hitboxSize * 1.15, hitboxSize)

	local bodyParts = {
		"UpperTorso","LowerTorso","Torso",
		"Left Arm","Right Arm","Left Leg","Right Leg",
		"LeftUpperArm","RightUpperArm","LeftLowerArm","RightLowerArm",
		"LeftUpperLeg","RightUpperLeg","LeftLowerLeg","RightLowerLeg"
	}
	for _, name in ipairs(bodyParts) do
		local p = char:FindFirstChild(name)
		if p and p:IsA("BasePart") then
			if not p:GetAttribute("KHUB_OrigSize") then
				p:SetAttribute("KHUB_OrigSize", p.Size)
			end
			origSizes[p] = p:GetAttribute("KHUB_OrigSize")
			local o = p:GetAttribute("KHUB_OrigSize")
			p.Size = Vector3.new(
				math.max(o.X, hitboxSize * 0.55),
				math.max(o.Y, hitboxSize * 0.55),
				math.max(o.Z, hitboxSize * 0.55)
			)
		end
	end

	if transparentPlayers then
		for _, p in ipairs(char:GetDescendants()) do
			if p:IsA("BasePart") then
				p.Transparency = p.Name == "Head" and 0.08 or 0.96
			end
			if (p:IsA("Decal") or p:IsA("Texture")) and p.Parent and p.Parent.Name ~= "Head" then
				p.Transparency = 0.95
			end
		end
	end

	local box = Instance.new("SelectionBox")
	box.Adornee = hrp
	box.Color3 = Color3.fromRGB(255, 120, 180)
	box.LineThickness = 0.03
	box.Transparency = 0.72
	box.Parent = hrp
	hitboxData[char] = {box = box, orig = origSizes}

	if espEnabled then
		local bb = Instance.new("BillboardGui")
		bb.Name = "KHUB_ESP"
		bb.Adornee = hrp
		bb.Size = UDim2.new(0, 110, 0, 48)
		bb.StudsOffset = Vector3.new(0, 3.5, 0)
		bb.AlwaysOnTop = true
		bb.Parent = hrp

		local headshot = Instance.new("ImageLabel", bb)
		headshot.Size = UDim2.new(0, 26, 0, 26)
		headshot.Position = UDim2.new(0.5, -13, 0, 0)
		headshot.BackgroundColor3 = Color3.fromRGB(30, 15, 45)
		headshot.Image = "https://www.roblox.com/headshot-thumbnail/image?userId=" .. plr.UserId .. "&width=100&height=100&format=png"
		Instance.new("UICorner", headshot).CornerRadius = UDim.new(1, 0)

		local nameLbl = Instance.new("TextLabel", bb)
		nameLbl.Size = UDim2.new(1, 0, 0, 16)
		nameLbl.Position = UDim2.new(0, 0, 0, 28)
		nameLbl.BackgroundTransparency = 1
		nameLbl.Text = plr.DisplayName
		nameLbl.Font = Enum.Font.GothamBold
		nameLbl.TextSize = 11
		nameLbl.TextColor3 = Color3.new(1, 1, 1)
		nameLbl.TextStrokeTransparency = 0.4
		espData[char] = {bb = bb}
	end
end

local function updateAll()
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr == player then continue end
		if not plr.Character then continue end
		local char = plr.Character

		if hitboxEnabled then
			if IsAlly(char) then
				clearChar(char)
			else
				applyHitbox(char, plr)
			end
		else
			for _, p in ipairs(char:GetDescendants()) do
				if p:IsA("BasePart") then
					if p:GetAttribute("KHUB_OrigSize") then
						p.Size = p:GetAttribute("KHUB_OrigSize")
					end
					p.Transparency = 0
				end
				if p:IsA("Decal") or p:IsA("Texture") then
					p.Transparency = 0
				end
			end
			clearChar(char)
		end
	end
end

	for _, dir in ipairs({hrp.CFrame.LookVector, -hrp.CFrame.LookVector, hrp.CFrame.RightVector, -hrp.CFrame.RightVector}) do

end

RunService.RenderStepped:Connect(function()
	isCurrentlyTouchingWall = checkWallTouch()

	if isAimbotEnabled and not isCurrentlyTouchingWall then
		local myChar = player.Character
		local hrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
		if hrp then
			local closest, dist = nil, math.huge
			for _, p in ipairs(Players:GetPlayers()) do
				if p ~= player and p.Character then
					local tHRP = p.Character:FindFirstChild("HumanoidRootPart")
					if tHRP then
						local d = (hrp.Position - tHRP.Position).Magnitude
						if d < dist then dist = d closest = tHRP end
					end
				end
			end
			if closest then
				hrp.CFrame = CFrame.new(hrp.Position, Vector3.new(closest.Position.X, hrp.Position.Y, closest.Position.Z))
			end
		end
	end

	if AutoFollowEnabled and FollowStatusGui and FollowStatusGui:FindFirstChild("StatusFrame") then
		local StatusLabel = FollowStatusGui.StatusFrame:FindFirstChild("StatusLabel")
		local char = player.Character
		if char then
			local humanoid = char:FindFirstChildOfClass("Humanoid")
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if humanoid and hrp then
				if humanoid.MoveDirection.Magnitude > 0.05 then
					if StatusLabel then StatusLabel.Text = "Follow: Paused (Input)" end
				else
					local targetPlr, shortestDist = nil, math.huge
					for _, p in ipairs(Players:GetPlayers()) do
						if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
							local dist = (hrp.Position - p.Character.HumanoidRootPart.Position).Magnitude
							if dist < shortestDist then shortestDist = dist; targetPlr = p end
						end
					end
					if targetPlr and targetPlr.Character then
						local targetHrp = targetPlr.Character:FindFirstChild("HumanoidRootPart")
						if targetHrp then
							if StatusLabel then StatusLabel.Text = "Touching: " .. targetPlr.Name end
							humanoid:MoveTo(targetHrp.Position)
						end
					end
				end
			end
		end
	end
end)

RunService.Stepped:Connect(function()
	if not AutoFlickEnabled or isFlicking then return end
	local char = player.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= player and p.Character then
			local pRoot = p.Character:FindFirstChild("HumanoidRootPart")
			if pRoot and (hrp.Position - pRoot.Position).Magnitude <= 5 then
				local currentTime = os.clock()
				if not lastTouchTimes[p] or (currentTime - lastTouchTimes[p] > 1) then
					lastTouchTimes[p] = currentTime
					isFlicking = true
					showFlickNotification(p.Name)
					task.spawn(function()
						local spinSpeed = 120
						local accumulatedLeft = 0
						while accumulatedLeft < 90 and AutoFlickEnabled do
							local dt = RunService.RenderStepped:Wait()
							local step = math.min(spinSpeed * dt * 60, 90 - accumulatedLeft)
							hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(step), 0)
							accumulatedLeft = accumulatedLeft + step
						end
						local accumulatedRight = 0
						while accumulatedRight < 90 and AutoFlickEnabled do
							local dt = RunService.RenderStepped:Wait()
							local step = math.min(spinSpeed * dt * 60, 90 - accumulatedRight)
							hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(-step), 0)
							accumulatedRight = accumulatedRight + step
						end
						isFlicking = false
					end)
				end
			end
		end
	end
end)

		local oldIndex = rawget(mt, "__index")
		mt.__index = newcclosure(function(self, key)
			if mode == "pc" then
				if key == "TouchEnabled" then return false end
				if key == "GamepadEnabled" then return false end
				if key == "KeyboardEnabled" then return true end
