local function readExecutorHwid()
    local candidates = {
        function() return type(gethwid) == "function" and gethwid() end,
        function() return type(get_hwid) == "function" and get_hwid() end,
        function() return type(getHWID) == "function" and getHWID() end,
        function() return syn and type(syn.gethwid) == "function" and syn.gethwid() end,
    }
    for _, getter in ipairs(candidates) do
        local ok, value = pcall(getter)
        if ok and type(value) == "string" then
            value = value:gsub("%s+", "")
            if #value >= 8 and #value <= 256 then return value end
        end
    end
    return nil
end

local function sendRequest(method, path, body)
    local ok, response = pcall(function()
        return httpRequest({
            Url = API .. path,
            Method = method,
            Headers = {
                ["Content-Type"] = "application/json",
                ["Accept"] = "application/json",
            },
            Body = body and HttpService:JSONEncode(body) or nil,
        })
    end)
    if not ok or type(response) ~= "table" then
        return nil, nil, "HTTP_REQUEST_FAILED"
    end
    if type(response.Body) ~= "string" or response.Body == "" then
        return response, nil, "EMPTY_RESPONSE"
    end
    local decoded, data = pcall(function()
        return HttpService:JSONDecode(response.Body)
    end)
    if not decoded or type(data) ~= "table" then
        return response, nil, "INVALID_JSON"
    end
    return response, data, nil
end

local function isoToUnix(iso)
    if type(iso) ~= "string" then return nil end
    local ok, date = pcall(function() return DateTime.fromIsoDate(iso) end)
    if not ok or not date then return nil end
    return date.UnixTimestamp
end

local function formatRemaining(seconds)
    seconds = math.max(0, math.floor(seconds))
    local days = math.floor(seconds / 86400)
    local hours = math.floor((seconds % 86400) / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = seconds % 60
    if days > 0 then
        return string.format("%dd %02dh %02dm", days, hours, minutes)
    end
    if hours > 0 then
        return string.format("%dh %02dm %02ds", hours, minutes, secs)
    end
    return string.format("%02dm %02ds", minutes, secs)
end

local function saveSession(token, expiry)
    if not writefile then return end
    pcall(writefile, SESSION_FILE,
        token .. "|" .. tostring(expiry) .. "|" .. tostring(os.time()))
end

local function loadSession()
    if not (readfile and isfile and isfile(SESSION_FILE)) then return nil, nil end
    local ok, data = pcall(readfile, SESSION_FILE)
    if not ok or type(data) ~= "string" or data == "" then return nil, nil end
    local token, expiry, savedAt = data:match("^([^|]+)|(%d+)|(%d+)$")
    if not (token and expiry and savedAt) then return nil, nil end
    expiry, savedAt = tonumber(expiry), tonumber(savedAt)
    if not (expiry and savedAt) then return nil, nil end
    local now = os.time()
    if now >= expiry or (now - savedAt) >= SESSION_MAX_AGE then
        return nil, nil
    end
    return token, expiry
end

local function clearSession()
    if delfile then
        pcall(delfile, SESSION_FILE)
    elseif writefile then
        pcall(writefile, SESSION_FILE, "")
    end
end

local function saveAllowedPlatforms(platforms)
    if not writefile or type(platforms) ~= "table" then return end
    local valid = {}
    for _, platform in ipairs(platforms) do
        local normalized = platform == "pc" and "PC"
            or platform == "mobile" and "Mobile"
            or platform == "console" and "Console"
            or platform
        if normalized == "PC" or normalized == "Mobile" or normalized == "Console" then
            table.insert(valid, normalized)
        end
    end
    if #valid > 0 then
        pcall(writefile, PLATFORMS_FILE, table.concat(valid, "|"))
    end
end

local function loadAllowedPlatforms()
    if not (readfile and isfile and isfile(PLATFORMS_FILE)) then return nil end
    local ok, raw = pcall(readfile, PLATFORMS_FILE)
    if not ok or type(raw) ~= "string" or raw == "" then return nil end
    local allowed = {}
    for value in raw:gmatch("[^|]+") do allowed[value] = true end
    return allowed
end

local function restrictToAllowedPlatforms(allowed)
    if not allowed then return end
    local compatible = {}
    for _, platform in ipairs(ENABLED_PLATFORMS) do
        if allowed[platform] then table.insert(compatible, platform) end
    end
    ENABLED_PLATFORMS = compatible
end

        { Position = UDim2.new(1, -24, 0, 80) }
    ):Play()
    task.delay(seconds or 8, function()
        if token ~= hideToken then return end
        local exitTween = TweenService:Create(
            card,

            { Position = UDim2.new(1, 360, 0, 80) }
        )
        exitTween:Play()
        local completed = pcall(function() exitTween.Completed:Wait() end)
        if completed and token == hideToken then card.Visible = false end
    end)
end

        if not targetDevice or not getgc then return 0, 0 end

        local modified, scanned = 0, 0
        for _, obj in ipairs(gc) do
            scanned = scanned + 1
            if scanned > 150000 then break end
            if type(obj) == "table" then
                local devOk, device = pcall(rawget, obj, KEY_DEVICE)
                if devOk and (device == "Mobile" or device == "PC" or device == "Console") then
                    pcall(function()
                        rawset(obj, KEY_DEVICE, targetDevice)
                        -- SmallScreen ما نلمسه
                        local fix = rawget(obj, KEY_FIX_ASPECT)
                        if type(fix) == "function" then pcall(fix, obj) end
                        modified = modified + 1
                    end)
                end
                local prefOk, pref = pcall(rawget, obj, KEY_PREF_INPUT)
                if prefOk and type(pref) == "table" then
                    local target = (targetDevice == "Console")
                        and rawget(obj, KEY_GAMEPAD)
                        or rawget(obj, KEY_KEYBOARD)
                    if type(target) == "table" and pref ~= target then
                        pcall(function()
                            rawset(obj, KEY_PREF_INPUT, target)
                            modified = modified + 1
                        end)
                    end
                end
            end
        end
        return modified, scanned
    end

showToast("Connecting to secure license service...", Color3.fromRGB(151, 112, 255), 5)

local _, product, productError = sendRequest("GET", "/api/license/product")
if not product then fail("Service unavailable", productError) return end
if product.enabled ~= true then
    fail("DeviceMode is currently disabled", "PRODUCT_DISABLED")
    return
end
if product.minimumSupportedVersion and VERSION < product.minimumSupportedVersion then
    fail("Please update your DeviceMode script", "CLIENT_UPDATE_REQUIRED")
    return
end

local activationBody = {
    licenseKey = LICENSE,
    clientVersion = VERSION,
    scriptVariant = SCRIPT_VARIANT,
    deviceFingerprint = DEVICE_FINGERPRINT,
    robloxUserId = ROBLOX_USER_ID,
}

local _, activation, activationError = sendRequest("POST", "/api/license/activate", activationBody)
if not activation or activation.success ~= true then
    local code = activation and activation.error or activationError or "ACTIVATION_FAILED"
    if code == "LICENSE_EXPIRED" then
        fail("License expired", code)
    elseif code == "USER_NOT_ALLOWED" then
        fail("This Roblox account is not authorized", code)
    elseif code == "DEVICE_MISMATCH" then
        fail("This device is not bound to the license", code)
    elseif code == "INVALID_LICENSE" then
        fail("License key was not found", code)
    else
        fail("License verification failed", code)
    end
    return
end

local sessionToken = activation.sessionToken
local expiryUnix = activation.license and isoToUnix(activation.license.expiresAt)
if type(sessionToken) ~= "string" or sessionToken == "" or not expiryUnix then
    fail("Server returned an incomplete license response", "INVALID_ACTIVATION")
    return
end

saveAllowedPlatforms(activation.allowedPlatforms)
restrictToAllowedPlatforms(loadAllowedPlatforms())
if #ENABLED_PLATFORMS == 0 then
    fail("No compatible device option is enabled for this account", "NO_COMPATIBLE_PLATFORM")
    return
end
if os.time() >= expiryUnix then
    fail("License expired", "LICENSE_EXPIRED")
    return
end

showToast(
    "License verified · " .. formatRemaining(expiryUnix - os.time()) .. " remaining",
    Color3.fromRGB(85, 215, 164),
    8
)

saveSession(sessionToken, expiryUnix)
task.spawn(runSpoofer, sessionToken, expiryUnix)

local running = true
while running do
    task.wait(30)
    if os.time() >= expiryUnix then
        running = false
        clearSession()
        fail("License expired", "LICENSE_EXPIRED")
        break
    end
    local _, hb = sendRequest("POST", "/api/license/heartbeat", {
        sessionToken = sessionToken,
        scriptVariant = SCRIPT_VARIANT,
        deviceFingerprint = DEVICE_FINGERPRINT,
        robloxUserId = ROBLOX_USER_ID,
    })
    if not hb or hb.success ~= true then
        local code = hb and hb.error or "HEARTBEAT_FAILED"
        running = false
        clearSession()
        if code == "LICENSE_EXPIRED" then
            fail("License expired", code)
        elseif code == "LICENSE_REVOKED" then
            fail("License revoked by administrator", code)
        elseif code == "SESSION_INVALID" then
            fail("License session is no longer valid", code)
        else
            fail("License session ended", code)
        end
        break
    end
end
