do
    P.fn = {}

    P.fn.makeDraggable = function(handle, frame)
        local dragging = false
        local dragStart = nil
        local startPos = nil
        local lastInput = nil
        handle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = frame.Position
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then dragging = false end
                end)
            end
        end)
        handle.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                lastInput = input
            end
        end)
        P.UIS.InputChanged:Connect(function(input)
            if input == lastInput and dragging then
                local delta = input.Position - dragStart
                frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end)
    end

-- ═══════════════════════════════════════════════════════════
--  WALL SLIDE
-- ═══════════════════════════════════════════════════════════
do
    local wsCharacter = nil
    local wsHumanoid = nil
    local wsHRP = nil
    local wsAttachment = nil
    local wsVelocity = nil
    local wsCharConn = nil
    local wsLoopConn = nil
    local rayParams = RaycastParams.new()
    pcall(function() rayParams.FilterType = Enum.RaycastFilterType.Exclude end)
    rayParams.IgnoreWater = true

    local function flatUnit(v)
        local f = Vector3.new(v.X, 0, v.Z)
        return f.Magnitude > 0.001 and f.Unit or Vector3.new(0, 0, 0)
    end

    local function safeUnit(v, fallback)
        return v.Magnitude > 0.001 and v.Unit or fallback
    end

    local function destroySlideVelocity()
        if wsVelocity then pcall(function() wsVelocity:Destroy() end) wsVelocity = nil end
    end

    local function restoreWallSlide()
        destroySlideVelocity()
        if wsHumanoid and wsHumanoid.Parent then wsHumanoid.AutoRotate = true end
    end

    local function createSlideVelocity()
        if not wsHRP or not wsAttachment then return nil end
        local ok, vel = pcall(function()
            local lv = Instance.new("LinearVelocity")
            lv.Name = "SuiteSlideVel"
            lv.Attachment0 = wsAttachment
            lv.RelativeTo = Enum.ActuatorRelativeTo.World
            lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
            lv.MaxForce = 100000
            pcall(function()
                lv.ForceLimitsEnabled = true
                lv.ForceLimitMode = Enum.ForceLimitMode.PerAxis
                lv.MaxAxesForce = Vector3.new(100000, 0, 100000)
            end)
            lv.Parent = wsHRP
            return lv
        end)
        if ok and vel then return vel end
        local ok2, bv = pcall(function()
            local b = Instance.new("BodyVelocity")
            b.Name = "SuiteSlideVel"
            b.MaxForce = Vector3.new(100000, 0, 100000)
            b.P = 1250
            b.Parent = wsHRP
            return b
        end)
        return ok2 and bv or nil
    end

    local function getWallNormal(moveDir)
        if not wsHRP then return nil end
        local normals = {}
        local side = Vector3.new(-moveDir.Z, 0, moveDir.X)
        local pos = wsHRP.Position
        local origins = {pos, pos + side * 0.25, pos - side * 0.25}
        local dirs = {moveDir, safeUnit(moveDir + side * 0.25, moveDir), safeUnit(moveDir - side * 0.25, moveDir)}
        for i = 1, 3 do
            local result = workspace:Raycast(origins[i], dirs[i] * 1.35, rayParams)
            if result and result.Instance and result.Instance:IsA("BasePart") and result.Instance.CanCollide and math.abs(result.Normal.Y) <= 0.3 then
                local n = flatUnit(result.Normal)
                if n.Magnitude > 0.001 and moveDir:Dot(n) < -0.01 then
                    table.insert(normals, n)
                end
            end
        end
        if #normals == 0 then return nil end
        local sum = Vector3.new(0, 0, 0)
        for _, n in ipairs(normals) do sum = sum + n end
        return safeUnit(sum, normals[1])
    end

    local function onChar(char)
        restoreWallSlide()
        if wsAttachment then pcall(function() wsAttachment:Destroy() end) wsAttachment = nil end
        wsCharacter = char
        wsHumanoid = char:WaitForChild("Humanoid", 5)
        wsHRP = char:WaitForChild("HumanoidRootPart", 5)
        if not wsHumanoid or not wsHRP then return end
        for _, n in ipairs({"SuiteSlideVel", "SuiteSlideAtt"}) do
            local old = wsHRP:FindFirstChild(n)
            if old then pcall(function() old:Destroy() end) end
        end
        wsAttachment = Instance.new("Attachment")
        wsAttachment.Name = "SuiteSlideAtt"
        wsAttachment.Parent = wsHRP
    end

    local function step()
        if not P.s.wallSlideEnabled then return end
        if not wsCharacter or not wsCharacter.Parent or not wsHumanoid or wsHumanoid.Health <= 0 or not wsHRP or not wsHRP.Parent then
            restoreWallSlide() return
        end
        local chars = {}
        for _, p in ipairs(P.Players:GetPlayers()) do
            if p.Character then table.insert(chars, p.Character) end
        end
        rayParams.FilterDescendantsInstances = chars
        local moveDir = flatUnit(wsHumanoid.MoveDirection)
        if moveDir.Magnitude < 0.08 then restoreWallSlide() return end
        local normal = getWallNormal(moveDir)
        if not normal then restoreWallSlide() return end
        local vel = wsHRP.Velocity
        local yT = wsHumanoid.FloorMaterial ~= Enum.Material.Air and 0 or math.min(vel.Y, 1.5)
        if math.abs(yT - vel.Y) > 0.01 then
            wsHRP.Velocity = Vector3.new(vel.X, yT, vel.Z)
        end
        if not wsVelocity or not wsVelocity.Parent then wsVelocity = createSlideVelocity() end
        if not wsVelocity then return end
        local slideDir = flatUnit(moveDir - normal * moveDir:Dot(normal))
        local finalDir = slideDir.Magnitude > 0.05 and slideDir or safeUnit(Vector3.new(-normal.Z, 0, normal.X), Vector3.new(1, 0, 0))
        local target = finalDir * wsHumanoid.WalkSpeed + normal * 0.01
        local vv = Vector3.new(target.X, 0, target.Z)
        if wsVelocity:IsA("LinearVelocity") then
            wsVelocity.VectorVelocity = vv
        elseif wsVelocity:IsA("BodyVelocity") then
            wsVelocity.Velocity = vv

-- ═══════════════════════════════════════════════════════════
--  ENEMY DETECTION + ASSIST
-- ═══════════════════════════════════════════════════════════
do
    local enemyCache = {enemies = {}, arena = nil, mySide = nil, time = 0}

    local function detectEnemies()
        local now = tick()
        if now - enemyCache.time < 0.1 then
            return enemyCache.enemies, enemyCache.arena, enemyCache.mySide
        end
        local enemies = {}
        local myTeam, myArena = nil, nil
        if workspace:FindFirstChild("Arenas") then
            for aName, arena in pairs(workspace.Arenas:GetChildren()) do
                if aName == "Arena5" or aName == "Arena5ICED" then continue end
                if arena:FindFirstChild("Slots") then
                    if arena.Slots:FindFirstChild("Left") then
                        for i = 1, 4 do
                            local slot = arena.Slots.Left:FindFirstChild(tostring(i))
                            if slot and slot:FindFirstChild("Data") and slot.Data:FindFirstChild("Player") then
                                if slot.Data.Player.Value == P.lp then myTeam = "Left" myArena = arena break end
                            end
                        end
                    end
                    if not myArena and arena.Slots:FindFirstChild("Right") then
                        for i = 1, 4 do
                            local slot = arena.Slots.Right:FindFirstChild(tostring(i))
                            if slot and slot:FindFirstChild("Data") and slot.Data:FindFirstChild("Player") then
                                if slot.Data.Player.Value == P.lp then myTeam = "Right" myArena = arena break end
                            end
                        end
                    end
                    if myTeam and myArena then
                        local et = (myTeam == "Left") and "Right" or "Left"
                        if myArena.Slots:FindFirstChild(et) then
                            for i = 1, 4 do
                                local slot = myArena.Slots[et]:FindFirstChild(tostring(i))
                                if slot and slot:FindFirstChild("Data") and slot.Data:FindFirstChild("Player") then
                                    local e = slot.Data.Player.Value
                                    if e and e ~= P.lp then table.insert(enemies, e) end
                                end
                            end
                        end
                        break
                    end
                end
            end
        end
        enemyCache.enemies, enemyCache.arena, enemyCache.mySide, enemyCache.time = enemies, myArena, myTeam, now
        return enemies, myArena, myTeam
    end
    P.fn.detectEnemies = detectEnemies

    local function isPlayerAlive(plr)
        if plr and plr.Character and plr.Character:FindFirstChild("Humanoid") then
            return plr.Character.Humanoid.Health > 0
        end
        return false
    end
    P.fn.isPlayerAlive = isPlayerAlive

    local function isPlayerOnFloor(plr, arena)
        if not plr or not arena then return false end
        if arena:FindFirstChild("Slots") then
            for _, side in pairs({"Left", "Right"}) do
                if arena.Slots:FindFirstChild(side) then
                    for i = 1, 4 do
                        local s = arena.Slots[side]:FindFirstChild(tostring(i))
                        if s and s:FindFirstChild("Data") and s.Data:FindFirstChild("Player") then
                            if s.Data.Player.Value == plr then
                                if s:FindFirstChild("Pad") then
                                    local pad = s.Pad
                                    local c = pad.Color
                                    if math.abs(c.R - 1) < 0.01 and math.abs(c.G - 89/255) < 0.01 and math.abs(c.B - 89/255) < 0.01 then
                                        return false
                                    else
                                        return true
                                    end
                                end
                                return true
                            end
                        end
                    end
                end
            end
        end
        return false
    end
    P.fn.isPlayerOnFloor = isPlayerOnFloor

    local function hasTargetLOS(hrp, character, targetHRP)
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {P.lp.Character}
        params.IgnoreWater = true
        local origin = hrp.Position + Vector3.new(0, 1.5, 0)
        local points = {targetHRP.Position, targetHRP.Position + Vector3.new(0, 1.5, 0), targetHRP.Position + Vector3.new(0, -1.5, 0)}
        local head = character:FindFirstChild("Head")
        if head then table.insert(points, head.Position) end
        for _, point in ipairs(points) do
            local result = workspace:Raycast(origin, point - origin, params)
            if result and result.Instance and result.Instance:IsDescendantOf(character) then return true end
        end
        return false
    end

    local function getAssistTarget(hrp, maxDist)
        local enemies, arena = detectEnemies()
        local bestDist = maxDist
        local best = nil
        for _, plr in ipairs(enemies) do
            if plr.Character then
                local char = plr.Character
                local targetHRP = char:FindFirstChild("HumanoidRootPart")
                if targetHRP and isPlayerAlive(plr) and isPlayerOnFloor(plr, arena) then
                    local dist = (targetHRP.Position - hrp.Position).Magnitude
                    if dist < bestDist and hasTargetLOS(hrp, char, targetHRP) then
                        bestDist = dist
                        best = plr
                    end
                end
            end
        end
        return best, bestDist
    end

    local function getPredictedPosition(targetHrp, myHrp)
        local targetPos = targetHrp.Position
        local velocity = targetHrp.AssemblyLinearVelocity
        velocity = Vector3.new(velocity.X, 0, velocity.Z)
        local speed = velocity.Magnitude
        if speed > 50 then
            velocity = velocity.Unit * 50
        elseif speed < 1.5 then
            return targetPos
        end
        local dist = (targetPos - myHrp.Position).Magnitude
        local predTime = math.clamp(dist / 42, 0.12, 0.42)
        local predictedPos = targetPos + (velocity * predTime)
        local params = RaycastParams.new()
        params.FilterDescendantsInstances = {targetHrp.Parent, P.lp.Character}
        params.FilterType = Enum.RaycastFilterType.Exclude
        local res = workspace:Raycast(targetPos, predictedPos - targetPos, params)
        if res then return res.Position - (velocity.Unit * 1.2) end
        return predictedPos
    end

    local fastTouchBusy = false
    local function fastTouch(targetChar, tool)
        if fastTouchBusy then return end
        if not targetChar or not tool then return end
        local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildOfClass("Part")
        if not handle then return end
        local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
        if not targetHRP then return end
        local myHRP = P.lp.Character and P.lp.Character:FindFirstChild("HumanoidRootPart")
        if not myHRP then return end
        local dist = (myHRP.Position - targetHRP.Position).Magnitude
        if dist < 4.5 then
            fastTouchBusy = true
            task.spawn(function()
                for _ = 1, 15 do
                    pcall(function()
                        firetouchinterest(targetHRP, handle, 0)
                        firetouchinterest(targetHRP, handle, 1)
                    end)
                end
                fastTouchBusy = false
            end)
        end
    end

    P.fn.refreshAssistUI = function()
        local pp = math.floor(P.vals.powerValues[P.s.powerIndex] * 1000 + 0.5) / 10
        P.ui.powerBtn.Text = tostring(pp) .. "%"
        P.ui.rangeBtn.Text = tostring(P.vals.rangeValues[P.s.rangeIndex]) .. "m"
        P.ui.assistBtn.Text = P.s.assistEnabled and "ON" or "OFF"
        if not P.s.assistEnabled then P.ui.assistTargetLabel.Text = "target · none" end
        P.fn.styleButton(P.ui.assistBtn, P.s.assistEnabled)
    end

    local function smoothLookAt(hrp, targetPos, delta)
        if P.UIS.MouseBehavior == Enum.MouseBehavior.LockCenter then return end
        local flatTarget = Vector3.new(targetPos.X, hrp.Position.Y, targetPos.Z)
        local dir = flatTarget - hrp.Position
        if dir.Magnitude < 0.5 then return end
        local targetCF = CFrame.lookAt(hrp.Position, flatTarget)
        hrp.CFrame = hrp.CFrame:Lerp(targetCF, math.clamp(delta * 10, 0, 1))
    end

    P.RunService.RenderStepped:Connect(function(delta)
        if not P.s.assistEnabled then P.s.assistHasTarget = false return end
        if P.s.grabBombActive or P.s.autoHoldActive then return end
        local character = P.lp.Character
        if not character then P.s.assistHasTarget = false return end
        local tool = character:FindFirstChildOfClass("Tool")
        if not tool then
            P.ui.assistTargetLabel.Text = "target · waiting tool"
            P.s.assistHasTarget = false
            return
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not hrp or humanoid.Health <= 0 then P.s.assistHasTarget = false return end
        local target, dist = getAssistTarget(hrp, P.vals.rangeValues[P.s.rangeIndex])
        if not target or not target.Character then
            P.ui.assistTargetLabel.Text = "target · none (0-" .. P.vals.rangeValues[P.s.rangeIndex] .. "m)"
            P.s.assistHasTarget = false
            return
        end
        local targetHRP = target.Character:FindFirstChild("HumanoidRootPart")
        if not targetHRP then P.s.assistHasTarget = false return end
        P.ui.assistTargetLabel.Text = string.format("target · %s · %dm", target.Name, math.floor(dist + 0.5))
        P.s.assistHasTarget = true
        local predictedPos = getPredictedPosition(targetHRP, hrp)
        if humanoid.MoveDirection.Magnitude > 0 then
            local diff = predictedPos - hrp.Position
            local flat = Vector3.new(diff.X, 0, diff.Z)
            if flat.Magnitude > 0.1 then
                local power = P.vals.powerValues[P.s.powerIndex]
                local blend = humanoid.MoveDirection.Unit * (1 - power) + flat.Unit * power
                if blend.Magnitude > 0 then humanoid:Move(blend.Unit, false) end
            end
        end
        smoothLookAt(hrp, predictedPos, delta)
        if dist < 4.5 then fastTouch(target.Character, tool) end
    end)
end

        end

        if not tool then
            P.ui.autoHoldStatus.Text = "status · no bomb"
            return

                local th = t.Character:FindFirstChild("HumanoidRootPart")
                if th then
                    local targetCF = CFrame.new(myHRP.Position, Vector3.new(th.Position.X, myHRP.Position.Y, th.Position.Z)) * CFrame.Angles(0, math.rad(35), 0)
                    myHRP.CFrame = myHRP.CFrame:Lerp(targetCF, ahSmoothTurn)
                    myHum:MoveTo(th.Position)
                    pcall(function() tool:Activate() end)
                end
            end
            return
        end
        local t = ahGetClosestEnemy()
        if not t or not t.Character then
            P.ui.autoHoldStatus.Text = "status · no target"
            return
        end
        P.ui.autoHoldStatus.Text = string.format("status · orbit %s · %.1fs", t.Name, rt)
        local th = t.Character:FindFirstChild("HumanoidRootPart")
        if not th then return end
        local ev = th.Position - ahLastEnemyPos
        local mr = myHRP.CFrame.RightVector
        local dot = ev:Dot(mr)
        if dot > 0.05 then ahOrbitDirection = -1
        elseif dot < -0.05 then ahOrbitDirection = 1 end
        ahLastEnemyPos = th.Position
        ahOrbitAngle = ahOrbitAngle + (dt * ahOrbitSpeed * ahOrbitDirection)
        local of = Vector3.new(math.sin(ahOrbitAngle), 0, math.cos(ahOrbitAngle)) * ahTargetDistance
        local gp = th.Position + of
        local ld = (myHRP.Position - th.Position).Unit
        myHRP.CFrame = CFrame.new(myHRP.Position, myHRP.Position + Vector3.new(ld.X, 0, ld.Z))
        myHum:MoveTo(gp)
    end)

    P.fn.ahSetDist = function(idx) ahDistIndex = idx; ahTargetDistance = ahDistValues[idx] end
    P.fn.ahSetRush = function(idx) ahRushIndex = idx; ahRushAtTime = ahRushValues[idx] end
    P.fn.ahSetOrbit = function(idx) ahOrbitIndex = idx; ahOrbitSpeed = ahOrbitValues[idx] end
    P.fn.ahSetSmooth = function(idx) ahSmoothIndex = idx; ahSmoothTurn = ahSmoothValues[idx] end
    P.fn.ahNextDist = function() return ahDistIndex % #ahDistValues + 1 end
    P.fn.ahNextRush = function() return ahRushIndex % #ahRushValues + 1 end
    P.fn.ahNextOrbit = function() return ahOrbitIndex % #ahOrbitValues + 1 end
    P.fn.ahNextSmooth = function() return ahSmoothIndex % #ahSmoothValues + 1 end
end

-- ═══════════════════════════════════════════════════════════
--  ANTI-LAG
-- ═══════════════════════════════════════════════════════════
do
    local antiLagConns = {}
    local antiLagSaved = {}
    local antiLagSavedCount = 0
    local ANTI_LAG_MAX = 80000
    local antiLagDetached = {}
    local antiLagLightingSaved = nil
    local antiLagTerrainSaved = nil

    local function saveAndSet(inst, prop, value)
        if antiLagSavedCount >= ANTI_LAG_MAX then return end
        local ok, orig = pcall(function() return inst[prop] end)
        if not ok then return end
        if not antiLagSaved[inst] then
            antiLagSaved[inst] = {}
            antiLagSavedCount = antiLagSavedCount + 1
        end
        if antiLagSaved[inst][prop] == nil then
            antiLagSaved[inst][prop] = orig
        end
        pcall(function() inst[prop] = value end)
    end

    local function detachTemp(inst)
        if antiLagDetached[inst] then return end
        antiLagDetached[inst] = inst.Parent
        pcall(function() inst.Parent = nil end)
    end

    local function process(v)
        if v.Name and v.Name:sub(1, 3) == "PX_" then return end
        pcall(function()
            if v:IsA("Model") then
                saveAndSet(v, "LevelOfDetail", "Disabled")
                saveAndSet(v, "ModelStreamingMode", "Nonatomic")
            elseif v:IsA("MeshPart") then
                saveAndSet(v, "CastShadow", false)
                saveAndSet(v, "DoubleSided", false)
                saveAndSet(v, "RenderFidelity", Enum.RenderFidelity.Performance)
            elseif v:IsA("BasePart") then
                saveAndSet(v, "CastShadow", false)
                saveAndSet(v, "Material", Enum.Material.Plastic)
                saveAndSet(v, "Reflectance", 0)
                saveAndSet(v, "MaterialVariant", "")
            elseif v:IsA("Decal") or v:IsA("Texture") then
                saveAndSet(v, "Transparency", 1)
            elseif v:IsA("SpecialMesh") then
                saveAndSet(v, "TextureId", 0)
            elseif v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") or v:IsA("SpotLight") then
                saveAndSet(v, "Enabled", false)
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then
                saveAndSet(v, "Enabled", false)
            elseif v:IsA("Attachment") then
                saveAndSet(v, "Visible", false)
            elseif v:IsA("Explosion") then
                saveAndSet(v, "BlastPressure", 1)
                saveAndSet(v, "BlastRadius", 1)
            elseif v:IsA("Sky") then
                detachTemp(v)
            elseif v:IsA("Atmosphere") then
                saveAndSet(v, "Density", 0)
                saveAndSet(v, "Haze", 0)
            elseif v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("SunRaysEffect")
                or v:IsA("DepthOfFieldEffect") or v:IsA("ColorCorrectionEffect") then
                saveAndSet(v, "Enabled", false)
            end
        end)
    end

    P.fn.applyAntiLag = function(st)
        P.s.antiLagActive = st
        if st then
            pcall(function() setfpscap(999999999) end)
            for _, v in next, workspace:GetDescendants() do process(v) end
            for _, v in next, P.Lighting:GetDescendants() do process(v) end
            antiLagLightingSaved = {
                GlobalShadows = P.Lighting.GlobalShadows,
                FogEnd = P.Lighting.FogEnd,
                Brightness = P.Lighting.Brightness,
                Ambient = P.Lighting.Ambient,
                OutdoorAmbient = P.Lighting.OutdoorAmbient,
            }
            P.Lighting.GlobalShadows = false
            P.Lighting.FogEnd = 9e9
            P.Lighting.Brightness = 0
            local terrain = workspace:FindFirstChildOfClass("Terrain")
            if terrain then
                antiLagTerrainSaved = {
                    WaterReflectance = terrain.WaterReflectance,
                    WaterTransparency = terrain.WaterTransparency,
                    WaterWaveSize = terrain.WaterWaveSize,
                    WaterWaveSpeed = terrain.WaterWaveSpeed,
                }
                pcall(function()
                    terrain.WaterReflectance = 0
                    terrain.WaterTransparency = 0.7
                    terrain.WaterWaveSize = 0
                    terrain.WaterWaveSpeed = 0
                end)
            end
            local wConn = workspace.DescendantAdded:Connect(function(v)
                if P.s.antiLagActive then process(v) end
            end)
            table.insert(antiLagConns, wConn)
            local lConn = P.Lighting.ChildAdded:Connect(function(v)
                if P.s.antiLagActive then process(v) end
            end)
            table.insert(antiLagConns, lConn)
        else
            for _, conn in pairs(antiLagConns) do if conn then conn:Disconnect() end end
            antiLagConns = {}
            for inst, props in pairs(antiLagSaved) do
                if typeof(inst) == "Instance" and inst.Parent then
                    for prop, value in pairs(props) do pcall(function() inst[prop] = value end) end
                end
            end
            antiLagSaved = {}
            antiLagSavedCount = 0
            for inst, parent in pairs(antiLagDetached) do
                if typeof(inst) == "Instance" and parent then pcall(function() inst.Parent = parent end) end
            end
            antiLagDetached = {}
            if antiLagLightingSaved then
                P.Lighting.GlobalShadows = antiLagLightingSaved.GlobalShadows
                P.Lighting.FogEnd = antiLagLightingSaved.FogEnd
                P.Lighting.Brightness = antiLagLightingSaved.Brightness
                P.Lighting.Ambient = antiLagLightingSaved.Ambient
                P.Lighting.OutdoorAmbient = antiLagLightingSaved.OutdoorAmbient
                antiLagLightingSaved = nil
            end
            local terrain = workspace:FindFirstChildOfClass("Terrain")
            if terrain and antiLagTerrainSaved then
                pcall(function()
                    terrain.WaterReflectance = antiLagTerrainSaved.WaterReflectance
                    terrain.WaterTransparency = antiLagTerrainSaved.WaterTransparency
                    terrain.WaterWaveSize = antiLagTerrainSaved.WaterWaveSize
                    terrain.WaterWaveSpeed = antiLagTerrainSaved.WaterWaveSpeed
                end)
                antiLagTerrainSaved = nil
            end
            pcall(function() setfpscap(60) end)
        end
    end
end

-- ═══════════════════════════════════════════════════════════
--  RECORDING MODE
-- ═══════════════════════════════════════════════════════════
do
    local origRecTransparency = {}

    local function setObjHidden(obj, hidden)
        pcall(function()
            if obj:IsA("GuiObject") then
                local key = obj
                if hidden then
                    if origRecTransparency[key] == nil then
                        origRecTransparency[key] = {
                            bg = obj.BackgroundTransparency,
                            txt = (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) and obj.TextTransparency or nil,
                            img = (obj:IsA("ImageLabel") or obj:IsA("ImageButton")) and obj.ImageTransparency or nil,
                        }
                    end
                    obj.BackgroundTransparency = 1
                    if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                        obj.TextTransparency = 1
                    end
                    if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
                        obj.ImageTransparency = 1
                    end
                else
                    local d = origRecTransparency[key]
                    if d then
                        obj.BackgroundTransparency = d.bg
                        if d.txt then obj.TextTransparency = d.txt end
                        if d.img then obj.ImageTransparency = d.img end
                    end
                end
            elseif obj:IsA("UIStroke") then
                local key = obj
                if hidden then
                    if origRecTransparency[key] == nil then origRecTransparency[key] = obj.Transparency end
                    obj.Transparency = 1
                else
                    if origRecTransparency[key] then obj.Transparency = origRecTransparency[key] end
                end
            end
        end)
    end

    P.fn.setRecording = function(hidden)
        local cg = game.CoreGui
        for _, g in pairs(cg:GetChildren()) do
            if g:IsA("ScreenGui") and (g.Name:find("Phantom") or g.Name:find("Wind") or g.Name:find("Dance") or g.Name:find("InvisibleGhost") or g.Name:find("Kaito")) then
                for _, d in ipairs(g:GetDescendants()) do setObjHidden(d, hidden) end
            end
        end
        if not hidden then origRecTransparency = {} end
    end
end

    local Folder = Instance.new("Folder")
    Folder.Name = "PX_KFX"
    Folder.Parent = workspace
    local rng = Random.new()
    local H = {}

    H.parts = function(m)
        local l = {}
        for _, d in ipairs(m:GetDescendants()) do
            if d:IsA("BasePart") then table.insert(l, d) end
        end
        return l
    end

    H.spawn = function(props)
        local p = Instance.new("Part")
        for k, v in pairs(props) do p[k] = v end
        p.Anchored = true p.CanCollide = false p.CanQuery = false p.CanTouch = false
        p.Parent = Folder
        return p
    end

    H.anim = function(dur, step)
        local t0 = os.clock()
        local conn
        conn = P.RunService.Heartbeat:Connect(function(dt)
            local a = (os.clock() - t0) / dur
            if a >= 1 then conn:Disconnect() pcall(step, 1, dt) return end
            pcall(step, a, dt)
        end)
        return conn
    end

    H.fade = function(m, dur)
        for _, d in ipairs(m:GetDescendants()) do
            if d:IsA("BasePart") then
                P.TS:Create(d, TweenInfo.new(dur), {Transparency = 1}):Play()
            elseif d:IsA("Decal") or d:IsA("Texture") then
                P.TS:Create(d, TweenInfo.new(dur), {Transparency = 1}):Play()
            elseif d:IsA("Highlight") then
                P.TS:Create(d, TweenInfo.new(dur), {FillTransparency = 1, OutlineTransparency = 1}):Play()
            elseif d:IsA("PointLight") then
                P.TS:Create(d, TweenInfo.new(dur), {Brightness = 0}):Play()
            elseif d:IsA("ParticleEmitter") or d:IsA("Fire") or d:IsA("Smoke") then
                d.Enabled = false
            end
        end
    end

    H.strip = function(m)
        for _, d in ipairs(m:GetDescendants()) do
            pcall(function()
                if d:IsA("Shirt") or d:IsA("Pants") or d:IsA("ShirtGraphic") or d:IsA("BodyColors")
                    or d:IsA("SurfaceAppearance") or d:IsA("Decal") or d:IsA("Texture") or d:IsA("CharacterMesh") then
                    d:Destroy()
                elseif d:IsA("MeshPart") then d.TextureID = ""
                elseif d:IsA("SpecialMesh") then d.TextureId = "" end
            end)
        end
    end

    H.paint = function(parts, color, mat, reflect, dur)
        for _, p in ipairs(parts) do
            if p.Name ~= "HumanoidRootPart" then
                p.Material = mat
                p.Reflectance = reflect or 0
                if dur then P.TS:Create(p, TweenInfo.new(dur), {Color = color}):Play()
                else p.Color = color end
            end
        end
    end

    H.particles = function(parts, props)
        for _, p in ipairs(parts) do
            if p.Name == "Torso" or p.Name == "UpperTorso" or p.Name == "Head" then
                local e = Instance.new("ParticleEmitter")
                for k, v in pairs(props) do e[k] = v end
                e.Parent = p
            end
        end
    end

    H.shards = function(pos, color, mat, count, speed, life)
        local list = {}
        for i = 1, count do
            local s = rng:NextNumber(0.3, 0.9)
            local p = H.spawn({
                Size = Vector3.new(s, s * rng:NextNumber(0.6, 1.4), s * rng:NextNumber(0.6, 1.4)),
                Color = color, Material = mat,
                CFrame = CFrame.new(pos + Vector3.new(rng:NextNumber(-1.2, 1.2), rng:NextNumber(-2, 2), rng:NextNumber(-1.2, 1.2)))
                    * CFrame.Angles(rng:NextNumber(0, 6), rng:NextNumber(0, 6), rng:NextNumber(0, 6)),
            })
            local dir = Vector3.new(rng:NextNumber(-1, 1), rng:NextNumber(0.2, 1.2), rng:NextNumber(-1, 1)).Unit
            list[i] = {p = p, v = dir * rng:NextNumber(0.5, 1) * speed,
                w = Vector3.new(rng:NextNumber(-6, 6), rng:NextNumber(-6, 6), rng:NextNumber(-6, 6))}
            P.Debris:AddItem(p, life + 0.5)
        end
        H.anim(life, function(a, dt)
            for _, s in ipairs(list) do
                if s.p.Parent then
                    s.v = s.v + Vector3.new(0, -60 * dt, 0)
                    s.p.CFrame = CFrame.new(s.p.Position + s.v * dt) * (s.p.CFrame - s.p.Position)
                        * CFrame.Angles(s.w.X * dt, s.w.Y * dt, s.w.Z * dt)
                    if a > 0.6 then s.p.Transparency = (a - 0.6) / 0.4 end
                end
            end
        end)
    end

    H.ring = function(pos, color, size)
        local r = H.spawn({Shape = Enum.PartType.Cylinder, Size = Vector3.new(0.3, 2, 2), Color = color,
            Material = Enum.Material.Neon, CFrame = CFrame.new(pos - Vector3.new(0, 2.5, 0)) * CFrame.Angles(0, 0, math.pi/2)})
        P.TS:Create(r, TweenInfo.new(0.75, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
            {Size = Vector3.new(0.3, size or 22, size or 22), Transparency = 1}):Play()
        P.Debris:AddItem(r, 1.1)
    end

    H.bolt = function(top, bottom, color, thick)
        local pts = {top}
        for i = 1, 8 do
            table.insert(pts, top:Lerp(bottom, i / 9) + Vector3.new(rng:NextNumber(-3, 3), 0, rng:NextNumber(-3, 3)))
        end
        table.insert(pts, bottom)
        local made = {}
        for i = 1, #pts - 1 do
            local a, b = pts[i], pts[i + 1]
            local len = (b - a).Magnitude
            local p = H.spawn({Size = Vector3.new(thick, thick, len), Color = color, Material = Enum.Material.Neon,
                CFrame = CFrame.lookAt(a, b) * CFrame.new(0, 0, -len/2)})
            P.Debris:AddItem(p, 0.6)
            table.insert(made, p)
        end
        return made
    end

    FX.specter = function(m, hrp)
        H.strip(m)
        local parts = H.parts(m)
        H.paint(parts, Color3.fromRGB(180, 130, 240), Enum.Material.ForceField, 0)
        local hl = Instance.new("Highlight", m)
        hl.FillColor = Color3.fromRGB(150, 100, 220)
        hl.OutlineColor = Color3.fromRGB(230, 200, 255)
        hl.FillTransparency = 0.65
        hl.OutlineTransparency = 0.35
        local base = m:GetPivot()
        H.anim(3.2, function(a)
            local wave = math.sin(a * 9) * 0.85
            local y = 15 * a
            m:PivotTo(base * CFrame.new(wave, y, math.cos(a * 6) * 0.5) * CFrame.Angles(0, a * math.pi * 3, 0))
            local tr = 0.3 + 0.7 * math.max(0, (a - 0.5) / 0.5)
            for _, p in ipairs(parts) do
                if p.Name ~= "HumanoidRootPart" then p.Transparency = tr end
            end
        end)
    end

            end
        end
        local sm = Instance.new("Smoke", hrp)

        local lt = Instance.new("PointLight", hrp)

        local pos = hrp.Position
        task.delay(1.4, function()
            H.ring(pos, Color3.fromRGB(255, 90, 15), 26)
            H.shards(pos, Color3.fromRGB(50, 40, 35), Enum.Material.Slate, 14, 22, 1.2)
        end)
        task.delay(2.5, function() H.fade(m, 0.8) end)
    end

    FX.zerog = function(m, hrp)
        local parts = H.parts(m)
        local head = m:FindFirstChild("Head")
        local owner = {}
        local vel = {}
        local spin = {}
        for _, p in ipairs(parts) do
            local o = p
            if head and p:FindFirstAncestorOfClass("Accessory") then o = head end
            owner[p] = o
            if not vel[o] then
                vel[o] = Vector3.new(rng:NextNumber(-4, 4), rng:NextNumber(5, 9), rng:NextNumber(-4, 4))
                spin[o] = Vector3.new(rng:NextNumber(-4, 4), rng:NextNumber(-4, 4), rng:NextNumber(-4, 4))
            end
        end
        H.anim(3.4, function(a, dt)
            local centers = {}
            for o in pairs(vel) do centers[o] = o.Position end
            for _, p in ipairs(parts) do
                local o = owner[p]
                local c = centers[o]
                local s = spin[o]
                p.CFrame = CFrame.new(vel[o] * dt) * (CFrame.new(c) * CFrame.Angles(s.X * dt, s.Y * dt, s.Z * dt) * CFrame.new(-c) * p.CFrame)
            end
        end)
        task.delay(2.4, function() H.fade(m, 0.8) end)
    end

    FX.helium = function(m, hrp)
        local base = m:GetPivot()
        local bpos = hrp.Position
        local bl = H.spawn({Shape = Enum.PartType.Ball, Size = Vector3.new(0.5, 0.5, 0.5),

            CFrame = CFrame.new(bpos + Vector3.new(0, 9, 0))})
        local st = H.spawn({Size = Vector3.new(0.1, 3.2, 0.1), Color = Color3.fromRGB(240, 240, 240),

        P.Debris:AddItem(bl, 7) P.Debris:AddItem(st, 7)
        task.delay(4.4, function()
            for _, o in ipairs({bl, st}) do
                P.TS:Create(o, TweenInfo.new(0.7), {Transparency = 1}):Play()
            end
            H.fade(m, 0.7)
        end)
        H.anim(5.2, function(a)
            local grow = math.min(1, a / 0.06)
            local h = (a < 0.06) and 0 or (((a - 0.06) / 0.94) ^ 1.85) * 95
            local of = Vector3.new(math.sin(a * 16) * 1.3, h, 0)
            local sway = math.sin(a * 8) * 0.14
            m:PivotTo((base + of) * CFrame.Angles(0, a * 2.5, sway))
            local r = 0.5 + 5.5 * (1 - (1 - grow) ^ 3)
            bl.Size = Vector3.new(r, r * 1.15, r)
            bl.CFrame = CFrame.new(bpos + of + Vector3.new(0, 6 + r * 0.55, 0)) * CFrame.Angles(0, 0, sway)
            st.CFrame = CFrame.new(bpos + of + Vector3.new(0, 4, 0))
        end)
    end

    local hooked = setmetatable({}, {__mode = "k"})
    local played = setmetatable({}, {__mode = "k"})

    local function snapshot(char, cf)
        local clone
        pcall(function()
            local old = char.Archivable
            char.Archivable = true
            clone = char:Clone()
            char.Archivable = old
        end)
        if not clone then return nil end
        for _, d in ipairs(clone:GetDescendants()) do
            pcall(function()
                if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("Humanoid")
                    or d:IsA("Sound") or d:IsA("BillboardGui") or d:IsA("ForceField")
                    or d:IsA("Tool") or d:IsA("Highlight") or d:IsA("ParticleEmitter")
                    or d:IsA("Fire") or d:IsA("Smoke") or d:IsA("Sparkles") then
                    d:Destroy()
                end
            end)
        end
        for _, p in ipairs(H.parts(clone)) do
            p.Anchored = true p.CanCollide = false p.CanTouch = false p.CanQuery = false p.Massless = true
        end
        if cf then clone:PivotTo(cf) end
        clone.Parent = Folder
        P.Debris:AddItem(clone, 12)
        return clone
    end

    local function hideOriginal(char)
        local function hide()
            for _, d in ipairs(char:GetDescendants()) do
                if d:IsA("BasePart") then d.LocalTransparencyModifier = 1 d.Transparency = 1
                elseif d:IsA("Decal") or d:IsA("Texture") then d.Transparency = 1
                elseif d:IsA("BillboardGui") then d.Enabled = false end
            end
        end
        hide()
        task.spawn(function()
            local t = os.clock()
            while os.clock() - t < 8 and char.Parent do
                hide() P.RunService.Heartbeat:Wait()
            end
        end)
    end

    local function play(id, char, cf)
        local fn = FX[id]
        if not fn or not char then return end
        local m = snapshot(char, cf)
        if not m then return end
        if not cf then hideOriginal(char) end
        local hrp = m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
        if not hrp then m:Destroy() return end
        local ok = pcall(fn, m, hrp)
        if not ok then m:Destroy() end
    end

    local function onDeath(char)
        if played[char] then return end
        played[char] = true
        if P.kfx.current <= 1 then return end
        local mR = P.lp.Character and P.lp.Character:FindFirstChild("HumanoidRootPart")
        local rt = char:FindFirstChild("HumanoidRootPart")
        if not (mR and rt) then return end
        if (rt.Position - mR.Position).Magnitude > P.kfx.range then return end
        play(P.kfx.list[P.kfx.current].id, char)
    end

    local function watchChar(char)
        if hooked[char] then return end
        hooked[char] = true
        local hum = char:WaitForChild("Humanoid", 5)
        if not hum then return end
        local fired = false
        local function died()
            if fired then return end
            fired = true
            pcall(onDeath, char)
        end
        hum.Died:Connect(died)
        hum.HealthChanged:Connect(function(h)
            if h <= 0 then died() end
        end)
    end

    local function watchPlayer(p)
        if p == P.lp then return end
        if p.Character then task.spawn(watchChar, p.Character) end
        p.CharacterAdded:Connect(watchChar)
    end

    for _, p in ipairs(P.Players:GetPlayers()) do watchPlayer(p) end
    P.Players.PlayerAdded:Connect(watchPlayer)

    P.fn.refreshKFX = function()
        local e = P.kfx.list[P.kfx.current]
        P.ui.kfxLabel.Text = e and (e.en .. " · " .. e.ar) or "None"
        P.fn.styleButton(P.ui.kfxLabel, P.kfx.current > 1)
    end
    P.fn.refreshKFX()
end
