local function findMurderer()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            local k = p.Character:FindFirstChild("Knife") or (p.Backpack and p.Backpack:FindFirstChild("Knife"))
            if k then return p end
        end
    end
    return nil
end

local function findSheriff()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            local g = p.Character:FindFirstChild("Gun") or (p.Backpack and p.Backpack:FindFirstChild("Gun"))
            if g then return p end
        end
    end
    return nil
end

local function findGun()
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and (obj.Name == "GunDrop" or obj.Name == "DropGun") then
            return obj
        end
    end
    return nil
end

end

local function farmGetHRP()
    if not player.Character then return nil end
    return player.Character:FindFirstChild("HumanoidRootPart")
end

local function farmGetHum()
    if not player.Character then return nil end
    return player.Character:FindFirstChildOfClass("Humanoid")
end

local function farmCanCollect()
    local c = player.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    local r = c:FindFirstChild("HumanoidRootPart")
    if not h or not r or h.Health <= 0 then return false end
    local ok, s = pcall(function() return h:GetState() end)
    if ok and s == Enum.HumanoidStateType.Dead then return false end
    return true
end

local function farmIsAlive()
    local R = ReplicatedStorage:FindFirstChild("Remotes")
    local G = R and R:FindFirstChild("Gameplay")
    local D = G and G:FindFirstChild("GetCurrentPlayerData")
    if not D then return true end
    local ok, res = pcall(function() return D:InvokeServer() end)
    if not ok or type(res) ~= "table" then return true end
    local me = res[player.Name]
    if not me then return false end
    if me.Dead == true or me.Killed == true then return false end
    return true
end

local function farmDetectMurderer()
    local hrp = farmGetHRP()
    if not hrp then return false end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            local k = p.Character:FindFirstChild("Knife") or (p.Backpack and p.Backpack:FindFirstChild("Knife"))
            if k then
                local th = p.Character:FindFirstChild("HumanoidRootPart")
                if th and (hrp.Position - th.Position).Magnitude <= Farm.MurdererDetect then
                    return true
                end
            end
        end
    end
    return false
end

        local c = workspace:FindFirstChild(name)
        if c then
            local cc = (name == "Lobby") and c or c:FindFirstChild("CoinContainer")
            if cc then
                for _, coin in ipairs(cc:GetChildren()) do
                    if coin:IsA("BasePart") and coin.Name == "Coin_Server"
                        and not coin:FindFirstChild("CollectedCoin")
                        and coin:FindFirstChild("TouchInterest") then
                        if Farm.PickMode == "Randomize" then
                            table.insert(list, coin)
                        else
                            local d = (hrp.Position - coin.Position).Magnitude
                            if d < dist then dist = d; best = coin end
                        end
                    end
                end
            end
        end
    end
    if Farm.PickMode == "Randomize" then
        return #list > 0 and list[math.random(1, #list)] or nil
    end
    return best
end

local function farmEvacuate()

    if not hrp then return end

            hrp.CFrame = ch[math.random(1, #ch)].CFrame + Vector3.new(0, 3, 0)
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

local function farmSmoothToCoin(coin)
    local hrp = farmGetHRP()
    if not hrp or not coin then return end
    local s = hrp.Position
    local e = coin.Position + Vector3.new(0, 3, 0)
    local d = (e - s).Magnitude
    local dur = d / Farm.Speed
    local t0 = tick()
    while tick() - t0 < dur and Farm.Enabled do
        if not coin.Parent then break end
        local p = math.min((tick() - t0) / dur, 1)
        hrp.CFrame = CFrame.new(s:Lerp(e, p))
        task.wait()
    end
end

local function farmFloatToCoin(coin)
    local hrp = farmGetHRP()
    if not hrp or not coin then return end
    local fp = Instance.new("Part")
    fp.Name = "PhantomFloat"
    fp.Size = Vector3.one
    fp.Transparency = 1
    fp.Anchored = true
    fp.CanCollide = false
    fp.Massless = true
    fp.CFrame = hrp.CFrame
    fp.Parent = workspace
    Farm.FloatPart = fp
    local oldA = hrp.Anchored
    hrp.Anchored = true
    local goal = coin.Position + Vector3.new(0, Farm.Lift, 0)
    local d = (fp.Position - goal).Magnitude
    local dur = d / Farm.Speed
    local tw = TweenService:Create(fp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {Position = goal})
    tw:Play()
    local t0 = tick()
    while tick() - t0 < dur + 1 and Farm.Enabled do
        if not coin.Parent then break end
        if hrp and fp then
            hrp.CFrame = fp.CFrame
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
        task.wait()
    end
    pcall(function() tw:Cancel() end)
    if fp then fp:Destroy() end
    Farm.FloatPart = nil
    if hrp then
        hrp.Anchored = oldA
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end
end

local function farmLoop()
    Farm.Gen = Farm.Gen + 1
    local g = Farm.Gen
    while Farm.Enabled and g == Farm.Gen do
        if not farmCanCollect() then task.wait(0.5)
        elseif not farmIsAlive() then task.wait(1)
        elseif farmDetectMurderer() then farmEvacuate() task.wait(0.3)
        else
            local c = farmFindCoin()
            if c then
                if Farm.Mode == "Floating" then farmFloatToCoin(c)
                elseif Farm.Mode == "Teleport" then farmTpToCoin(c)
                elseif Farm.Mode == "Smooth" then farmSmoothToCoin(c)
                else farmFlyToCoin(c) end
                farmCollect(c)
            else task.wait(0.5) end
        end
        task.wait(0.1)
    end
end

end

end

local ShootOP = {
    NormalBaseLead  = 0.085,
    NormalPingWeight = 1.05,
    NormalSmoothFactor = 0.2,
    WBBaseLead = 0.145,
    WBPingWeight = 1.1,
    WBSpreadRadius = 0.8,
    Gui = nil,
    VelocityCache = {},
}

local function shootOpGetPing()
    local ping = 0.033
    pcall(function()
        if StatsService and StatsService.Network and StatsService.Network.ServerStatsItem then
            ping = StatsService.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
        end
    end)
    return math.clamp(ping, 0.005, 0.5)
end

local function shootOpPredict(targetPlayer, mode)
    if not targetPlayer or not targetPlayer.Character then return nil end
    local char = targetPlayer.Character
    local hrp  = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso")
    local hum  = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return nil end

    local ping   = shootOpGetPing()
    local curPos = hrp.Position
    local curVel = hrp.AssemblyLinearVelocity

    local last  = ShootOP.VelocityCache[targetPlayer.UserId]
    local accel = Vector3.zero
    if last and (os.clock() - last.t) < 0.2 then
        local dt = os.clock() - last.t
        if dt > 0 then accel = (curVel - last.v) / dt end
    end
    ShootOP.VelocityCache[targetPlayer.UserId] = { v = curVel, t = os.clock() }

    local t
    if mode == "Normal" then
        t = ShootOP.NormalBaseLead + ping * ShootOP.NormalPingWeight
    else
        t = ShootOP.WBBaseLead + ping * ShootOP.WBPingWeight
    end

    local predictedPos = curPos + curVel * t + 0.5 * accel * t * t

    local state = hum:GetState()
    if state == Enum.HumanoidStateType.Freefall or state == Enum.HumanoidStateType.Jumping then
        predictedPos = predictedPos + Vector3.new(0, -0.5 * workspace.Gravity * t * t, 0)
    end

    if hum.MoveDirection.Magnitude > 0 then
        local md = hum.MoveDirection
        local vd = curVel.Magnitude > 0 and curVel.Unit or Vector3.zero
        if md:Dot(vd) < 0.3 then
            predictedPos = predictedPos:Lerp(curPos + md * hum.WalkSpeed * t, ShootOP.NormalSmoothFactor)
        end
    end

    return predictedPos
end

local SilentAim = {
    Enabled = false, Mode = "Dynamic",
    AutoEquip = true, WallCheck = true,
    BaseLead = 0.113, MinLead = 0.02, MaxLead = 0.6,
    Hook = nil,
}

local function saFlat(v) return Vector3.new(v.X, 0, v.Z) end

local function saPredict(target, mode)
    if not target or not target.Character then return nil end
    local c = target.Character
    local hrp = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    if not hrp or not h then return nil end
    local ping = player:GetNetworkPing()
    local lead = math.clamp(SilentAim.BaseLead + ping, SilentAim.MinLead, SilentAim.MaxLead)

    if mode == "Default" then
        local v = hrp.AssemblyLinearVelocity
        if v.Magnitude == 0 then return hrp.Position, 0 end
        local n = v.Unit * v.Magnitude / 16.5
        local y = math.clamp(n.Y, -2, 2.65)
        return hrp.Position + Vector3.new(n.X, y, n.Z / 1.25), 0
    end

    if mode == "Piercing" then mode = "Dynamic" end

    if mode == "Dynamic" then
        local v = hrp.AssemblyLinearVelocity
        local md = h.MoveDirection
        local ws = h.WalkSpeed
        local isAir = h:GetState() == Enum.HumanoidStateType.Freefall or h:GetState() == Enum.HumanoidStateType.Jumping
        if h.Sit or h.PlatformStand then return hrp.Position, 0 end
        local iv = md * ws
        local bv = saFlat(v):Lerp(saFlat(iv), isAir and 0.5 or 0.85)
        local p = hrp.Position + bv * lead
        local floor = hrp.Position.Y - 1.2
        if isAir then
            local jp = h.UseJumpPower and h.JumpPower or 50
            local g = workspace.Gravity
            local mj = jp * jp / (2 * g)
            if mj < hrp.Position.Y - floor then
                local ny = hrp.Position.Y + v.Y * lead - 0.5 * g * lead * lead
                p = Vector3.new(p.X, math.max(ny, floor), p.Z)
            else
                p = Vector3.new(p.X, floor + 5, p.Z)
            end
        end
        return p, lead
    end
    return hrp.Position, 0
end

local function saWallBlocked(from, to, tchar)
    local d = to - from
    if d.Magnitude < 0.001 then return false end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = {player.Character}
    if tchar then table.insert(rp.FilterDescendantsInstances, tchar) end
    return workspace:Raycast(from, d, rp) ~= nil
end

local function saHook()
    if SilentAim.Hook then return end
    if not hookmetamethod or not newcclosure then return end
    local old
    old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        local m = getnamecallmethod()
        if m == "FireServer" and SilentAim.Enabled then
            local c = player.Character
            if c then
                local g = c:FindFirstChild("Gun")
                if g then
                    local s = g:FindFirstChild("Shoot")
                    if s and self == s then
                        local murderer = findMurderer()
                        if murderer and murderer.Character then
                            local p = saPredict(murderer, SilentAim.Mode)
                            if p then
                                if SilentAim.WallCheck and SilentAim.Mode ~= "Piercing" then
                                    local o = (c:FindFirstChild("Head") and c.Head.Position) or c.HumanoidRootPart.Position
                                    if saWallBlocked(o, p, murderer.Character) then
                                        return old(self, ...)
                                    end
                                end
                                local a = {...}
                                a[1] = CFrame.new(p)
                                a[2] = CFrame.new(p)
                                return old(self, unpack(a))
                            end
                        end
                    end
                end
            end
        end
        return old(self, ...)
    end))
    SilentAim.Hook = old
end

local function draw3DBox(esp, cf, pos, cam, color, boxSize)
    local corners = {
        Vector3.new( boxSize.X/2,  boxSize.Y/2,  boxSize.Z/2),
        Vector3.new(-boxSize.X/2,  boxSize.Y/2,  boxSize.Z/2),
        Vector3.new( boxSize.X/2, -boxSize.Y/2,  boxSize.Z/2),
        Vector3.new(-boxSize.X/2, -boxSize.Y/2,  boxSize.Z/2),
        Vector3.new( boxSize.X/2,  boxSize.Y/2, -boxSize.Z/2),
        Vector3.new(-boxSize.X/2,  boxSize.Y/2, -boxSize.Z/2),
        Vector3.new( boxSize.X/2, -boxSize.Y/2, -boxSize.Z/2),
        Vector3.new(-boxSize.X/2, -boxSize.Y/2, -boxSize.Z/2),
    }
    local edges = {{1,2},{3,4},{5,6},{7,8},{1,3},{2,4},{5,7},{6,8},{1,5},{2,6},{3,7},{4,8}}

    while #esp.boxLines < #edges do
        local l = Drawing.new("Line")
        l.Thickness = 1
        l.Visible   = false
        table.insert(esp.boxLines, l)
    end

    local screenPoints = {}
    for i, corner in ipairs(corners) do
        local worldPos = cf * CFrame.new(corner)
        local vec, onScreen = cam:WorldToViewportPoint(worldPos.p)
        screenPoints[i] = {vec=vec, on=onScreen}
    end

    for i, edge in ipairs(edges) do
        local l = esp.boxLines[i]
        local a = screenPoints[edge[1]]
        local b = screenPoints[edge[2]]
        if a.on and b.on then
            l.From    = Vector2.new(a.vec.X, a.vec.Y)
            l.To      = Vector2.new(b.vec.X, b.vec.Y)
            l.Color   = color
            l.Visible = true
        else
            l.Visible = false
        end
    end
end

local function isAnyRoleESPActive()
    local i = featureStates.InnocentESP
    local m = featureStates.MurderESP
    local s = featureStates.HeroSheriffESP
    return i.names or i.boxes or i.tracers or i.distance
        or m.names or m.boxes or m.tracers or m.distance
        or s.names or s.boxes or s.tracers or s.distance
end

local function manageRoleUpdating()
    local needRole = isAnyRoleESPActive() or checkAnyHighlights()
    if needRole and not roleUpdating then
        roleUpdating = true
        if roleUpdateConnection then roleUpdateConnection:Disconnect() end
        roleUpdateConnection = RunService.Heartbeat:Connect(function()
            if tick() - lastRoleUpdate < 2 then return end
            lastRoleUpdate = tick()
            local ok, res = pcall(function()
                return ReplicatedStorage:FindFirstChild("GetPlayerData", true):InvokeServer()
            end)
            if ok and res then roleData = res end
        end)
    elseif not needRole and roleUpdating then
        roleUpdating = false
        if roleUpdateConnection then
            roleUpdateConnection:Disconnect()
            roleUpdateConnection = nil
        end
    end
end

local function updateRoleHighlights()
    if not roleData then return end
    local sheriffName = nil
    for name, data in pairs(roleData) do
        if data.Role == "Sheriff" then sheriffName = name break end
    end
    local sheriffPlayer = sheriffName and Players:FindFirstChild(sheriffName)
    local isSheriffAlive = sheriffPlayer and IsAliveESP(sheriffPlayer)

    for _, plr in ipairs(getCachedPlayers()) do
        if plr ~= player and plr.Character then
            local model     = plr.Character
            local highlight = model:FindFirstChild("PlayerHighlight")
            local data      = roleData[plr.Name]
            local role      = data and data.Role
            local alive     = IsAliveESP(plr)

            local hlEnabled = false
            if not role then                  hlEnabled = featureStates.InnocentHighlights
            elseif role=="Murderer" then      hlEnabled = featureStates.MurderHighlights
            elseif role=="Sheriff" or role=="Hero" then hlEnabled = featureStates.SheriffHeroHighlights
            else                              hlEnabled = featureStates.InnocentHighlights end

            if hlEnabled then
                local color = Color3.new(1,1,1)
                if alive then
                    if role=="Murderer"   then color = Color3.fromRGB(225,0,0)
                    elseif role=="Sheriff" then color = Color3.fromRGB(0,0,225)
                    elseif role=="Hero" and not isSheriffAlive then color = Color3.fromRGB(255,250,0)
                    else color = Color3.fromRGB(0,225,0) end
                end
                if not highlight then
                    highlight = Instance.new("Highlight")
                    highlight.Name            = "PlayerHighlight"
                    highlight.Adornee         = model
                    highlight.FillTransparency   = 0.5
                    highlight.OutlineTransparency = 0
                    highlight.DepthMode       = Enum.HighlightDepthMode.AlwaysOnTop
                    highlight.Parent          = model
                end
                highlight.FillColor    = color
                highlight.OutlineColor = getOutlineColor(color)
                highlight.Enabled = true
            else
                if highlight then highlight:Destroy() end
            end
        end
    end

    if featureStates.CoinHighlights then
        for _, coin in ipairs(findCoinServerParts()) do
            if coin and coin.Parent then
                local h = coin:FindFirstChild("ItemHighlight")
                if not h then
                    h = Instance.new("Highlight")
                    h.Name   = "ItemHighlight"
                    h.Adornee = coin
                    h.FillTransparency = 0.5
                    h.Parent = coin
                end
                h.FillColor    = Color3.fromRGB(255,215,0)
                h.OutlineColor = Color3.new(0,0,0)
            end
        end
    end

    if featureStates.GunHighlights then
        for _, gun in ipairs(findDropGunParts()) do
            if gun and gun.Parent then
                local h = gun:FindFirstChild("ItemHighlight")
                if not h then
                    h = Instance.new("Highlight")
                    h.Name   = "ItemHighlight"
                    h.Adornee = gun
                    h.FillTransparency = 0.5
                    h.Parent = gun
                end
                h.FillColor    = Color3.fromRGB(255,0,255)
                h.OutlineColor = Color3.new(1,1,1)
            end
        end
    end
end

local function clearAllHighlights()
    for _, plr in pairs(Players:GetPlayers()) do
        if plr.Character then
            local h = plr.Character:FindFirstChild("PlayerHighlight")
            if h then h:Destroy() end
        end
    end
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj.Name=="Coin_Server" or obj.Name=="DropGun" or obj.Name=="GunDrop" then
            local h = obj:FindFirstChild("ItemHighlight")
            if h then h:Destroy() end
        end
    end
end

local function manageHighlightsConnection()
    if checkAnyHighlights() then
        if not HighlightsConnection then
            HighlightsConnection = RunService.Heartbeat:Connect(updateRoleHighlights)
        end
    else
        if HighlightsConnection then
            HighlightsConnection:Disconnect()
            HighlightsConnection = nil
            clearAllHighlights()
        end
    end
    manageRoleUpdating()
end

local function refreshHighlights()
    if HighlightsConnection then clearAllHighlights() end
end

local function updateRoleESP()
    local cam = workspace.CurrentCamera
    if not cam then return end
    local screenBot = Vector2.new(cam.ViewportSize.X/2, cam.ViewportSize.Y)
    local current   = {Innocent={}, Murder={}, Sheriff={}}

    for _, otherPlayer in ipairs(getCachedPlayers()) do
        if otherPlayer ~= player and otherPlayer.Character then
            local hrp = otherPlayer.Character:FindFirstChild("HumanoidRootPart")
            local hum = otherPlayer.Character:FindFirstChild("Humanoid")
            if hrp and hum then
                local role   = getPlayerRole(otherPlayer)
                local isDead = hum.Health <= 0
                local config, elements, targetTable, color

                if isDead then
                    config=featureStates.InnocentESP; elements=innocentEspElements; targetTable=current.Innocent; color=Color3.fromRGB(200,200,200)
                elseif role=="Murderer" then
                    config=featureStates.MurderESP; elements=murderEspElements; targetTable=current.Murder; color=Color3.fromRGB(255,0,0)
                elseif role=="Sheriff" or role=="Hero" then
                    config=featureStates.HeroSheriffESP; elements=sheriffEspElements; targetTable=current.Sheriff; color=Color3.fromRGB(0,0,255)
                else
                    config=featureStates.InnocentESP; elements=innocentEspElements; targetTable=current.Innocent; color=Color3.fromRGB(0,255,0)
                end

                if config.names or config.boxes or config.tracers or config.distance then
                    targetTable[otherPlayer.Character] = true
                    if not elements[otherPlayer.Character] then
                        elements[otherPlayer.Character] = createESPObject()
                        setupESPObject(elements[otherPlayer.Character])
                    end
                    local esp = elements[otherPlayer.Character]
                    local vec, onScreen = cam:WorldToViewportPoint(hrp.Position)

                    if onScreen then
                        local topY    = cam:WorldToViewportPoint(hrp.Position + Vector3.new(0,3,0)).Y
                        local bottomY = cam:WorldToViewportPoint(hrp.Position - Vector3.new(0,3,0)).Y
                        local sizeY   = math.abs(bottomY - topY)
                        local sizeX   = sizeY / 2

                        if config.boxes then
                            if config.boxType == "2D" then
                                esp.box.Visible  = true
                                esp.box.Size     = Vector2.new(sizeX, sizeY)
                                esp.box.Position = Vector2.new(vec.X-sizeX/2, vec.Y-sizeY/2)
                                esp.box.Color    = color
                                for _, l in ipairs(esp.boxLines) do l.Visible=false end
                            else
                                esp.box.Visible = false
                                draw3DBox(esp, hrp.CFrame, hrp.Position, cam, color, Vector3.new(2,5,2))
                            end
                        else
                            esp.box.Visible = false
                            for _, l in ipairs(esp.boxLines) do l.Visible=false end
                        end

                        if config.tracers then
                            esp.tracer.Visible = true
                            esp.tracer.From    = screenBot
                            esp.tracer.To      = Vector2.new(vec.X, vec.Y)
                            esp.tracer.Color   = color
                        else esp.tracer.Visible = false end

                        if config.names then
                            esp.name.Visible   = true
                            esp.name.Text      = otherPlayer.Name
                            esp.name.Position  = Vector2.new(vec.X, vec.Y-sizeY/2-15)
                            esp.name.Color     = color
                        else esp.name.Visible = false end

                        if config.distance then
                            esp.distance.Visible  = true
                            esp.distance.Text     = math.floor((hrp.Position-cam.CFrame.Position).Magnitude).."m"
                            esp.distance.Position = Vector2.new(vec.X, vec.Y+sizeY/2+5)
                            esp.distance.Color    = color
                        else esp.distance.Visible = false end
                    else
                        esp.box.Visible=false; esp.tracer.Visible=false
                        esp.name.Visible=false; esp.distance.Visible=false
                        for _, l in ipairs(esp.boxLines) do l.Visible=false end
                    end
                end
            end
        end
    end

    for target, esp in pairs(innocentEspElements) do
        if not current.Innocent[target] then cleanupDrawingTable(esp); innocentEspElements[target]=nil end
    end
    for target, esp in pairs(murderEspElements) do
        if not current.Murder[target] then cleanupDrawingTable(esp); murderEspElements[target]=nil end
    end
    for target, esp in pairs(sheriffEspElements) do
        if not current.Sheriff[target] then cleanupDrawingTable(esp); sheriffEspElements[target]=nil end
    end
end

local function managePlayerESPConnection()
    if isAnyRoleESPActive() then
        if not playerEspConnection then
            playerEspConnection = RunService.RenderStepped:Connect(updateRoleESP)
        end
    else
        if playerEspConnection then
            playerEspConnection:Disconnect(); playerEspConnection=nil
            for _, tbl in pairs({innocentEspElements,murderEspElements,sheriffEspElements}) do
                for _, esp in pairs(tbl) do cleanupDrawingTable(esp) end
            end
            innocentEspElements={}; murderEspElements={}; sheriffEspElements={}
        end
    end
    manageRoleUpdating()
end

local function updateItemESP(config, elementTable, findFunc, itemName, color)
    local cam = workspace.CurrentCamera
    if not cam then return end
    local screenBot  = Vector2.new(cam.ViewportSize.X/2, cam.ViewportSize.Y)
    local currentItems = {}

    for _, item in ipairs(findFunc()) do
        if item and item.Parent then
            currentItems[item] = true
            if not elementTable[item] then
                elementTable[item] = createESPObject()
                setupESPObject(elementTable[item])
            end
            local esp = elementTable[item]
            local vec, onScreen = cam:WorldToViewportPoint(item.Position)

            if onScreen and (config.names or config.boxes or config.tracers or config.distance) then
                local size = 20
                if config.boxes then
                    if config.boxType=="2D" then
                        esp.box.Visible=true; esp.box.Size=Vector2.new(size,size)
                        esp.box.Position=Vector2.new(vec.X-size/2,vec.Y-size/2); esp.box.Color=color
                        for _, l in ipairs(esp.boxLines) do l.Visible=false end
                    else
                        esp.box.Visible=false
                        draw3DBox(esp, item.CFrame, item.Position, cam, color, Vector3.new(2,2,2))
                    end
                else
                    esp.box.Visible=false
                    for _, l in ipairs(esp.boxLines) do l.Visible=false end
                end

                if config.tracers then
                    esp.tracer.Visible=true; esp.tracer.From=screenBot
                    esp.tracer.To=Vector2.new(vec.X,vec.Y); esp.tracer.Color=color
                else esp.tracer.Visible=false end

                if config.names then
                    esp.name.Visible=true; esp.name.Text=itemName
                    esp.name.Position=Vector2.new(vec.X,vec.Y-size/2-15); esp.name.Color=color
                else esp.name.Visible=false end

                if config.distance then
                    esp.distance.Visible=true
                    esp.distance.Text=math.floor((item.Position-cam.CFrame.Position).Magnitude).."m"
                    esp.distance.Position=Vector2.new(vec.X,vec.Y+size/2+5); esp.distance.Color=color
                else esp.distance.Visible=false end
            else
                esp.box.Visible=false; esp.tracer.Visible=false
                esp.name.Visible=false; esp.distance.Visible=false
                for _, l in ipairs(esp.boxLines) do l.Visible=false end
            end
        end
    end

    for item, esp in pairs(elementTable) do
        if not currentItems[item] then cleanupDrawingTable(esp); elementTable[item]=nil end
    end
end

            coinEspConnection = RunService.RenderStepped:Connect(function()
                updateItemESP(featureStates.CoinESP, coinEspElements, findCoinServerParts, "Coin", Color3.fromRGB(255,215,0))
            end)
        end

            coinEspConnection:Disconnect(); coinEspConnection=nil
            for _, esp in pairs(coinEspElements) do cleanupDrawingTable(esp) end
            coinEspElements={}

end

local function cleanupAllESP()
    if playerEspConnection  then playerEspConnection:Disconnect();  playerEspConnection=nil  end
    if coinEspConnection    then coinEspConnection:Disconnect();    coinEspConnection=nil    end
    if gunEspConnection     then gunEspConnection:Disconnect();     gunEspConnection=nil     end
    if HighlightsConnection then HighlightsConnection:Disconnect(); HighlightsConnection=nil end
    if roleUpdateConnection then roleUpdateConnection:Disconnect(); roleUpdateConnection=nil end
    for _, tbl in pairs({innocentEspElements,murderEspElements,sheriffEspElements,coinEspElements,gunEspElements}) do
        for _, esp in pairs(tbl) do cleanupDrawingTable(esp) end
    end
    innocentEspElements={}; murderEspElements={}; sheriffEspElements={}
    coinEspElements={};     gunEspElements={}
    clearAllHighlights()
end

local function setupFly()
    local Camera = workspace.CurrentCamera
    if _G.FlyEnabled then
        if BodyVelocity then BodyVelocity:Destroy() end
        if BodyGyro     then BodyGyro:Destroy()     end
        local Character = player.Character
        if not Character then return end
        local RootPart = Character:FindFirstChild("HumanoidRootPart")
        local Humanoid = Character:FindFirstChild("Humanoid")
        if not RootPart or not Humanoid then return end
        BodyVelocity = Instance.new("BodyVelocity")
        BodyVelocity.MaxForce = Vector3.new(math.huge,math.huge,math.huge)
        BodyVelocity.Velocity = Vector3.new(0,0.1,0)
        BodyVelocity.Parent   = RootPart
        BodyGyro = Instance.new("BodyGyro")
        BodyGyro.MaxTorque = Vector3.new(math.huge,math.huge,math.huge)
        BodyGyro.P = 10000; BodyGyro.D = 500
        BodyGyro.CFrame = RootPart.CFrame
        BodyGyro.Parent = RootPart
        Humanoid.PlatformStand = true
        if espConnections.fly then espConnections.fly:Disconnect() end
        espConnections.fly = RunService.RenderStepped:Connect(function()
            if not _G.FlyEnabled or not RootPart or not RootPart.Parent then return end
            local moveDirection  = Humanoid.MoveDirection
            local targetVelocity = Vector3.new(0,0.1,0)
            if moveDirection.Magnitude > 0 then
                local cameraCFrame = Camera.CFrame
                targetVelocity = moveDirection * _G.FlySpeedValue
                if math.abs(cameraCFrame.LookVector.Y) > 0.1 then
                    targetVelocity = Vector3.new(targetVelocity.X, cameraCFrame.LookVector.Y*_G.FlySpeedValue, targetVelocity.Z)
                end
            end
            BodyVelocity.Velocity = BodyVelocity.Velocity:Lerp(targetVelocity, 0.15)
            BodyGyro.CFrame       = BodyGyro.CFrame:Lerp(Camera.CFrame, 0.15)
        end)
    else
        if espConnections.fly then espConnections.fly:Disconnect(); espConnections.fly=nil end
        if BodyVelocity then BodyVelocity:Destroy() end
        if BodyGyro     then BodyGyro:Destroy()     end
        BodyVelocity=nil; BodyGyro=nil
        local Character = player.Character
        if Character then
            local Humanoid = Character:FindFirstChild("Humanoid")
            if Humanoid then Humanoid.PlatformStand=false end
        end
    end
end

local function setupLockCamera()
    if _G.LockCameraEnabled then
        if espConnections.lockCamera then espConnections.lockCamera:Disconnect() end
        espConnections.lockCamera = RunService.RenderStepped:Connect(function()
            if not _G.LockCameraEnabled then
                if espConnections.lockCamera then espConnections.lockCamera:Disconnect(); espConnections.lockCamera=nil end
                return
            end
            local target = findMurderer() or findSheriff()
            if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                workspace.CurrentCamera.CFrame = CFrame.new(
                    workspace.CurrentCamera.CFrame.Position,
                    target.Character.HumanoidRootPart.Position
                )
            end
        end)
    end
end

local function setupAimbot(targetType)
    if targetType=="sheriff" and not _G.AimbotEnabled then
        if espConnections.aimbot then espConnections.aimbot:Disconnect(); espConnections.aimbot=nil end
        return
    end
    local connection = RunService.Heartbeat:Connect(function()
        local target = targetType=="sheriff" and findSheriff() or findMurderer()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                player.Character.HumanoidRootPart.CFrame = CFrame.new(
                    player.Character.HumanoidRootPart.Position,
                    target.Character.HumanoidRootPart.Position
                )
            end
        end
    end)
    if targetType=="sheriff" then
        if espConnections.aimbot then espConnections.aimbot:Disconnect() end
        espConnections.aimbot = connection
    end
end

-- SHOOT MURDERER (SHERIFF) - Using SilentAim
-- =============================================
local function GetMurdererViaRemote()
    local ok, roles = pcall(function()
        return ReplicatedStorage:FindFirstChild("GetPlayerData", true):InvokeServer()
    end)
    if ok and roles then
        for name, data in pairs(roles) do
            if data.Role == "Murderer" then return Players:FindFirstChild(name) end
        end
    end
    return findMurderer()
end

local function ShootMurdererFunc()
    if not player.Character then return false end
    local murderer = GetMurdererViaRemote()
    if not murderer or not murderer.Character then
        SendNotification("Gun System", _G.SelectedLanguage=="Saudi" and "ما لقينا القاتل" or "Murderer not found")
        return false
    end
    local gun = player.Character:FindFirstChild("Gun") or player.Backpack:FindFirstChild("Gun")
    if not gun then
        SendNotification("Gun System", _G.SelectedLanguage=="Saudi" and "ما عندك سلاح" or "Gun not found")
        return false
    end
    if gun.Parent ~= player.Character then
        gun.Parent = player.Character
        task.wait(0.1)
    end
    gun = player.Character:FindFirstChild("Gun")
    if not gun then return false end

    local targetPart = murderer.Character:FindFirstChild("HumanoidRootPart")
    if not targetPart then return false end

    local targetPos = targetPart.Position
    local mHum = murderer.Character:FindFirstChild("Humanoid")
    if shootOffset ~= 0 and mHum then
        local moveDir = mHum.MoveDirection
        if moveDir.Magnitude > 0 then
            local finalOffset = shootOffset + (pingMultiplierVal * 0.5)
            targetPos = targetPos + (moveDir.Unit * finalOffset)
        end
    end

    local shootRemote = gun:FindFirstChild("Shoot")
    if shootRemote then
        local originCF = player.Character.HumanoidRootPart.CFrame
        pcall(function()
            shootRemote:FireServer(originCF, CFrame.new(targetPos))
        end)
        SendNotification("Gun System", _G.SelectedLanguage=="Saudi" and "تم التصويب" or "Shot fired")
        return true
    end
    return false
end

local function createSheriffGui()
    if sheriffGui then sheriffGui:Destroy() end
    sheriffGui = Instance.new("ScreenGui")
    sheriffGui.Name           = "SheriffGui"
    sheriffGui.Parent         = CoreGui
    sheriffGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

    local Frame = Instance.new("Frame", sheriffGui)
    Frame.BackgroundColor3       = Color3.fromRGB(26,26,26)
    Frame.BackgroundTransparency = 0.45
    Frame.Position = UDim2.new(0.5,-80,0.2,0)
    Frame.Size     = UDim2.new(0,161,0,78)
    Frame.Active   = true
    Instance.new("UICorner", Frame)

    local TextLabel = Instance.new("TextLabel", Frame)
    TextLabel.BackgroundTransparency = 1
    TextLabel.Position = UDim2.new(0,0,0.1,0)
    TextLabel.Size     = UDim2.new(0,161,0,29)
    TextLabel.Font     = Enum.Font.GothamBold
    TextLabel.Text     = T("shootOpNormal")
    TextLabel.TextColor3 = Color3.new(1,1,1)
    TextLabel.TextScaled = true

    local TriggerButton = Instance.new("TextButton", Frame)
    TriggerButton.BackgroundColor3 = Color3.fromRGB(0,200,100)
    TriggerButton.Position  = UDim2.new(0.05,0,0.55,0)
    TriggerButton.Size      = UDim2.new(0,144,0,27)
    TriggerButton.Font      = Enum.Font.GothamBold
    TriggerButton.Text      = _G.SelectedLanguage=="Saudi" and "إطلاق" or "FIRE"
    TriggerButton.TextColor3 = Color3.new(1,1,1)
    TriggerButton.TextScaled = true
    Instance.new("UICorner", TriggerButton)

    local dragging, dragInput, dragStart, startPos
    Frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = Frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    Frame.InputChanged:Connect(function(input)
        if input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input==dragInput and dragging then
            local delta = input.Position - dragStart
            Frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+delta.X, startPos.Y.Scale, startPos.Y.Offset+delta.Y)
        end
    end)

    TriggerButton.MouseButton1Click:Connect(function() ShootMurdererFunc() end)
    sheriffButtonActive = true
end

local function removeSheriffGui()
    if sheriffGui then sheriffGui:Destroy(); sheriffGui=nil end
    sheriffButtonActive = false
end

local function ScanForGunDrops()
    GunSystem.ActiveGunDrops = {}
    for _, obj in pairs(workspace:GetDescendants()) do
        if (obj.Name=="GunDrop" or obj.Name=="DropGun") and obj:IsA("BasePart") then
            table.insert(GunSystem.ActiveGunDrops, obj)
        end
    end
end

local function EquipGun()
    local char = player.Character
    if not char then return false end
    if char:FindFirstChild("Gun") then return true end
    local bg = player.Backpack:FindFirstChild("Gun")
    if bg then
        bg.Parent = char
        task.wait(0.1)
        return true
    end
    return false
end

local function ManualGrab()
    if player.Backpack:FindFirstChild("Knife") or (player.Character and player.Character:FindFirstChild("Knife")) then
        SendNotification("Gun System", _G.SelectedLanguage=="Saudi" and "أنت قاتل" or "You are Murderer")
        return
    end

local function getPlayersForFling(input)
    local result = {}
    input = string.lower(input or "")
    if input=="all" then
        for _, p in ipairs(Players:GetPlayers()) do
            if p~=player then table.insert(result, p) end
        end
        return sortPlayersAlpha(result)
    elseif input=="nonfriends" then
        for _, p in ipairs(Players:GetPlayers()) do
            if p~=player then
                local ok, isFriend = pcall(function() return p:IsFriendsWith(player.UserId) end)
                if not (ok and isFriend) then table.insert(result, p) end
            end
        end
        return sortPlayersAlpha(result)
    elseif input=="murder" then
        if flingMurder then
            local mp = Players:FindFirstChild(flingMurder)
            if mp and mp~=player and mp.Character and getFlingAlive(mp) then
                table.insert(result, mp)
            end
        end
    elseif input=="sheriff" or input=="hero" then
        if flingSheriff then
            local sp = Players:FindFirstChild(flingSheriff)
            if sp and sp~=player and sp.Character and getFlingAlive(sp) then table.insert(result, sp) end
        end
        if flingHero then
            local hp = Players:FindFirstChild(flingHero)
            if hp and hp~=player and hp.Character and getFlingAlive(hp) then table.insert(result, hp) end
        end
    else
        local terms = {}
        for term in string.gmatch(input, "([^,]+)") do
            term = string.match(term, "^%s*(.-)%s*$")
            if term~="" then table.insert(terms, term) end
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if p~=player then
                local pn = string.lower(p.Name)
                local dn = p.DisplayName and string.lower(p.DisplayName) or ""
                for _, term in ipairs(terms) do
                    if string.find(pn,term) or string.find(dn,term) then
                        table.insert(result, p); break
                    end
                end
            end
        end
    end
    return result
end

local function Flinger(TargetPlayer, duration)
    local Character = player.Character
    local Humanoid  = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart  = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    if not TCharacter then return end
    local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    local THead     = TCharacter:FindFirstChild("Head")
    local Accessory = TCharacter:FindFirstChildOfClass("Accessory")
    local Handle    = Accessory and Accessory:FindFirstChild("Handle")

    if not (Character and Humanoid and RootPart) then return end
    if RootPart.Velocity.Magnitude < 50 then getgenv().OldPos = RootPart.CFrame end
    if THead then workspace.CurrentCamera.CameraSubject = THead
    elseif Handle then workspace.CurrentCamera.CameraSubject = Handle
    elseif THumanoid and TRootPart then workspace.CurrentCamera.CameraSubject = THumanoid end
    if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

    local FPos = function(BasePart, Pos, Ang)
        RootPart.CFrame = CFrame.new(BasePart.Position)*Pos*Ang
        Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position)*Pos*Ang)
        RootPart.Velocity    = Vector3.new(9e7,9e7*10,9e7)
        RootPart.RotVelocity = Vector3.new(9e8,9e8,9e8)
    end

    local SFBasePart = function(BasePart)
        local TimeToWait = duration or 2
        local Time = tick(); local Angle = 0
        repeat
            if RootPart and THumanoid then
                if BasePart.Velocity.Magnitude < 50 then
                    Angle=Angle+100
                    FPos(BasePart, CFrame.new(0,1.5,0)+THumanoid.MoveDirection*BasePart.Velocity.Magnitude/1.25, CFrame.Angles(math.rad(Angle),0,0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0,-1.5,0)+THumanoid.MoveDirection*BasePart.Velocity.Magnitude/1.25, CFrame.Angles(math.rad(Angle),0,0))
                    task.wait()
                else
                    FPos(BasePart, CFrame.new(0,1.5,THumanoid.WalkSpeed), CFrame.Angles(math.rad(90),0,0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0,-1.5,-THumanoid.WalkSpeed), CFrame.Angles(0,0,0))
                    task.wait()
                end
            else break end
        until not flingActive or BasePart.Velocity.Magnitude>500 or BasePart.Parent~=TargetPlayer.Character or TargetPlayer.Parent~=game.Players or THumanoid.Sit or tick()>Time+TimeToWait
    end

    local prevDestroy = workspace.FallenPartsDestroyHeight
    workspace.FallenPartsDestroyHeight = 0/0
    local BV = Instance.new("BodyVelocity", RootPart)
    BV.Name="EpixVel"; BV.Velocity=Vector3.new(9e8,9e8,9e8); BV.MaxForce=Vector3.new(1/0,1/0,1/0)
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    if TRootPart and THead then
        if (TRootPart.CFrame.p-THead.CFrame.p).Magnitude > 5 then SFBasePart(THead) else SFBasePart(TRootPart) end
    elseif TRootPart then SFBasePart(TRootPart)
    elseif THead then SFBasePart(THead)
    elseif Handle then SFBasePart(Handle) end

    BV:Destroy()
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    workspace.CurrentCamera.CameraSubject = Humanoid

    repeat
        if Character and Humanoid and RootPart and getgenv().OldPos then
            RootPart.CFrame = getgenv().OldPos*CFrame.new(0,.5,0)
            Character:SetPrimaryPartCFrame(getgenv().OldPos*CFrame.new(0,.5,0))
            Humanoid:ChangeState("GettingUp")
            for _, x in pairs(Character:GetChildren()) do
                if x:IsA("BasePart") then x.Velocity=Vector3.new(); x.RotVelocity=Vector3.new() end
            end
        end
        task.wait()
    until not flingActive or (RootPart and getgenv().OldPos and (RootPart.Position-getgenv().OldPos.p).Magnitude<25)
    workspace.FallenPartsDestroyHeight = prevDestroy
end

local function yeet(targetPlayer)
    local character       = player.Character
    local targetCharacter = targetPlayer.Character
    if not character or not targetCharacter or not targetCharacter:FindFirstChild("HumanoidRootPart") then return false end
    if character.HumanoidRootPart.Velocity.Magnitude < 50 then getgenv().OldPos = character.HumanoidRootPart.CFrame end

    local Thrust = Instance.new("BodyThrust", character.HumanoidRootPart)
    Thrust.Force = Vector3.new(9999,9999,9999); Thrust.Name = "YeetForce"
    local prevDestroy = workspace.FallenPartsDestroyHeight
    workspace.FallenPartsDestroyHeight = 0/0
    local startTime = tick()
    local dur = (currentInput=="all" or currentInput=="nonfriends") and 5 or math.huge

    local yeetConn
    yeetConn = RunService.Heartbeat:Connect(function()
        if not targetCharacter or not targetCharacter:FindFirstChild("HumanoidRootPart") or not flingActive or tick()>startTime+dur then
            yeetConn:Disconnect(); Thrust:Destroy()
            workspace.FallenPartsDestroyHeight = prevDestroy
            if character and character.HumanoidRootPart and getgenv().OldPos then
                character.HumanoidRootPart.CFrame = getgenv().OldPos*CFrame.new(0,.5,0)
                character.Humanoid:ChangeState("GettingUp")
                for _, x in pairs(character:GetDescendants()) do
                    if x:IsA("BasePart") then x.Velocity=Vector3.new(); x.RotVelocity=Vector3.new() end
                end
            end
            return
        end
        local targetHRP = targetCharacter.HumanoidRootPart
        local offsetPos = targetHRP.Position + (targetHRP.Velocity.Magnitude>0.1 and (targetHRP.Velocity.Unit*targetHRP.Velocity.Magnitude) or Vector3.zero)
        character.HumanoidRootPart.CFrame = CFrame.new(offsetPos)
        Thrust.Location = targetHRP.Position
    end)
    return true
end

local function addPlayerToProcessed(p)
    if not p or p==player then return end
    local inp = string.lower(currentInput)
    local match = false
    if inp=="all" then match=true
    elseif inp=="nonfriends" then
        local ok, isFriend = pcall(function() return p:IsFriendsWith(player.UserId) end)
        match = not (ok and isFriend)
    elseif inp=="murder" then
        if flingMurder and p.Name==flingMurder then match=getFlingAlive(p) end
    elseif inp=="sheriff" or inp=="hero" then
        if (flingSheriff and p.Name==flingSheriff) or (flingHero and p.Name==flingHero) then match=getFlingAlive(p) end
    else
        local terms = {}
        for term in string.gmatch(inp,"([^,]+)") do
            term=string.match(term,"^%s*(.-)%s*$")
            if term~="" then table.insert(terms,term) end
        end
        local pn=string.lower(p.Name)
        local dn=p.DisplayName and string.lower(p.DisplayName) or ""
        for _, term in ipairs(terms) do
            if string.find(pn,term) or string.find(dn,term) then match=true; break end
        end
    end
    if match then processedPlayers[p]=true end
end

local function flingPlayers()
    local plist = {}
    for p, _ in pairs(processedPlayers) do
        if p and p.Character and p.Character.Parent~=nil then table.insert(plist, p) end
    end
    if currentInput=="all" or currentInput=="nonfriends" then plist=sortPlayersAlpha(plist) end
    for _, p in ipairs(plist) do
        if not flingActive then break end
        if p and p.Character and p.Character.Parent~=nil then
            local dur = (currentInput=="all" or currentInput=="nonfriends") and 1.5 or nil
            if flingMode_val==1 then Flinger(p, dur)
            elseif flingMode_val==2 then
                yeet(p)
                if currentInput=="all" or currentInput=="nonfriends" then task.wait(1.5) end
            end
        end
    end
    if flingActive then task.wait(); flingPlayers() end
end

local Dodge = {

    JumpPower = 19, Gen = 0, Platform = nil, LastDodge = 0,

local function dodgeFlat(v) return Vector3.new(v.X, 0, v.Z) end

local function dodgeUntrack(model)
    local d = Dodge.Tracked[model]
    if not d then return end
    if d.render then d.render:Disconnect() end
    if d.pos then d.pos:Disconnect() end
    if d.vel then d.vel:Disconnect() end

    if not model then return nil end

    return nil
end

local function dodgePlatform(hrp, hum)
    if Dodge.Platform then Dodge.Platform:Destroy() end
    local p = Instance.new("Part")
    p.Name = "PhantomDodgePlatform"
    p.Size = Vector3.new(100, 1, 100)
    p.Position = hrp.Position - Vector3.new(1000, 3, 1000)
    p.Anchored = true
    p.CanCollide = true
    p.Transparency = 1
    p.Parent = workspace
    Dodge.Platform = p
    local saved = hrp.CFrame
    hrp.CFrame = CFrame.new(p.Position + Vector3.new(0, 4, 0))
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    if hum then hum:ChangeState(Enum.HumanoidStateType.Freefall) end
    task.delay(1.7, function()
        if hrp and hrp.Parent then
            hrp.CFrame = saved
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
        if p then p:Destroy() end
        if Dodge.Platform == p then Dodge.Platform = nil end
    end)
    return true
end

local function dodgeDo(dir, power, threat)
    local c = player.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    local hrp = c:FindFirstChild("HumanoidRootPart")
    if not h or not hrp then return false end
    local bk = player:FindFirstChild("Backpack")
    if (bk and bk:FindFirstChild("Knife")) or c:FindFirstChild("Knife") then return false end
    if Dodge.Mode == "Platform" then return dodgePlatform(hrp, h) end
    local flat = dodgeFlat(dir)
    if flat.Magnitude < 0.01 then return false end
    local u = flat.Unit
    local oldWS = h.WalkSpeed
    h.WalkSpeed = oldWS * 2.05
    h:ChangeState(Enum.HumanoidStateType.Jumping)
    local vel = Vector3.new(u.X * power, math.max(hrp.AssemblyLinearVelocity.Y, Dodge.JumpPower or 19), u.Z * power)
    hrp.AssemblyLinearVelocity = vel
    h:Move(u * 1.25, false)
    Dodge.Gen = Dodge.Gen + 1
    local g = Dodge.Gen
    task.spawn(function()
        local t0 = os.clock()
        while os.clock() - t0 < 0.16 do
            if g ~= Dodge.Gen or not hrp.Parent then break end
            h:Move(u * 1.25, false)
            RunService.RenderStepped:Wait()
        end
        if h.Parent then h.WalkSpeed = oldWS end
        if g == Dodge.Gen then h:Move(Vector3.zero, false) end
    end)
    Dodge.LastDodge = tick()
    return true
end

local function dodgePerpendicular(pos, threatPos)
    local dir = (threatPos - pos).Unit
    local flat = dodgeFlat(dir)
    if flat.Magnitude < 0.01 then flat = Vector3.new(1, 0, 0) end
    return Vector3.new(-flat.Z, 0, flat.X)
end

local function dodgeTrack(model)
    if Dodge.Tracked[model] then return end
    local visual = dodgeGetVisual(model)
    if not visual then return end
    local lastPos = visual.Position
    local hist = {}
    local function step()
        if not model.Parent then dodgeUntrack(model) return end
        local c = player.Character
        local hrp = c and c:FindFirstChild("HumanoidRootPart")
        if not hrp then lastPos = visual.Position return end
        local p = visual.Position
        local delta = p - lastPos
        if delta.Magnitude > 0.002 and delta.Magnitude < 40 then
            table.insert(hist, delta.Unit)
            if #hist > 3 then table.remove(hist, 1) end
        end
        lastPos = p
        local dir = nil
        if #hist > 0 then
            local sum = Vector3.zero
            for _, v in ipairs(hist) do sum = sum + v end
            if sum.Magnitude > 0.01 then dir = sum.Unit end
        end
        if dir then
            local dist = (p - hrp.Position).Magnitude
            if dist < 45 then
                local toPlayer = hrp.Position - p
                if dir:Dot(toPlayer.Unit) > 0.55 or (p + dir * 30 - hrp.Position).Magnitude < 6.5 then
                    if tick() - Dodge.LastDodge > 0.5 then
                        local perp = dodgePerpendicular(hrp.Position, p + dir * 10)
                        local rp = RaycastParams.new()
                        rp.FilterType = Enum.RaycastFilterType.Exclude
                        rp.FilterDescendantsInstances = {c}
                        local hit = workspace:Raycast(hrp.Position, perp * 12, rp)
                        if hit and hit.Distance < 10 then
                            perp = -perp
                        end
                        dodgeDo(perp, 42, p)
                    end
                end
            end
        end
    end
    Dodge.Tracked[model] = {
        render = RunService.PreSimulation:Connect(step),
        pos = visual:GetPropertyChangedSignal("Position"):Connect(step),
        vel = visual:GetPropertyChangedSignal("AssemblyLinearVelocity"):Connect(step),
    }
end

local function dodgeStart()
    for _, m in ipairs(workspace:GetDescendants()) do
        if m:IsA("Model") and m.Name == "ThrowingKnife" then dodgeTrack(m) end
    end
    if Dodge.Added then Dodge.Added:Disconnect() end
    Dodge.Added = workspace.DescendantAdded:Connect(function(d)
        if d:IsA("Model") and d.Name == "ThrowingKnife" then
            task.defer(dodgeTrack, d)
        end
    end)
end

local function dodgeStop()
    if Dodge.Added then Dodge.Added:Disconnect() Dodge.Added = nil end
    for k in pairs(Dodge.Tracked) do dodgeUntrack(k) end
    if Dodge.Platform then Dodge.Platform:Destroy() Dodge.Platform = nil end
end

    if AntiFling.Registered then return true end
    pcall(function() PhysicsService:RegisterCollisionGroup("PhantomLocal") end)
    pcall(function() PhysicsService:RegisterCollisionGroup("PhantomOthers") end)
    local ok = pcall(function()

    if ok then AntiFling.Registered = true end

local function afApplyChar(char, group)
    if not char then return end
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("BasePart") then
            if AntiFling.Saved[d] == nil then AntiFling.Saved[d] = d.CollisionGroup end
            if d.CollisionGroup ~= group then
                pcall(function() d.CollisionGroup = group end)
            end
        end
    end
end

local function afRestoreChar(char)
    if not char then return end
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("BasePart") and AntiFling.Saved[d] ~= nil then
            pcall(function() d.CollisionGroup = AntiFling.Saved[d] end)
            AntiFling.Saved[d] = nil
        end
    end
end

local function afApplyAll()
    if not AntiFling.Enabled then return end
    afApplyChar(player.Character, "PhantomLocal")
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then afApplyChar(p.Character, "PhantomOthers") end
    end
end

    if not afRegister() then return end
    afApplyAll()
    if not AntiFling.Heartbeat then
        local t = 0
        AntiFling.Heartbeat = RunService.Heartbeat:Connect(function()
            if not AntiFling.Enabled then return end
            if os.clock() - t < 0.2 then return end
            t = os.clock()
            afApplyAll()
        end)
    end
    AntiFling.PlayerAdded = Players.PlayerAdded:Connect(function(p)
        p.CharacterAdded:Connect(function()
            if AntiFling.Enabled then afApplyChar(p.Character, "PhantomOthers") end
        end)
    end)
    AntiFling.CharAdded = player.CharacterAdded:Connect(function(c)
        if AntiFling.Enabled then afApplyChar(c, "PhantomLocal") end
    end)
end

local function afStop()
    if AntiFling.Heartbeat then AntiFling.Heartbeat:Disconnect() AntiFling.Heartbeat = nil end
    if AntiFling.PlayerAdded then AntiFling.PlayerAdded:Disconnect() AntiFling.PlayerAdded = nil end
    if AntiFling.CharAdded then AntiFling.CharAdded:Disconnect() AntiFling.CharAdded = nil end
    afRestoreChar(player.Character)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then afRestoreChar(p.Character) end
    end
end

local function clutchUseBomb()
    local bp = player:FindFirstChild("Backpack")
    local c = player.Character or player.CharacterAdded:Wait()
    local fb = (bp and bp:FindFirstChild("FakeBomb")) or c:FindFirstChild("FakeBomb")
        or (bp and bp:FindFirstChild("Toys") and bp.Toys:FindFirstChild("FakeBomb"))
    if not fb then
        SendNotification("Clutch", _G.SelectedLanguage=="Saudi" and "ما عندك قنبلة" or "No bomb")
        return
    end
    if not Clutch.CanUse then return end
    Clutch.CanUse = false
    local hrp = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    if not hrp or not h then Clutch.CanUse = true return end
    if fb.Parent ~= c then
        if ReplicatedStorage.Remotes.Extras.ReplicateToy then
            pcall(function() ReplicatedStorage.Remotes.Extras.ReplicateToy:InvokeServer("FakeBomb") end)
            fb = bp:WaitForChild("FakeBomb", 2) or c:WaitForChild("FakeBomb", 2)
        end
    end
    if not fb then Clutch.CanUse = true return end
    fb.Parent = c
    local remote = fb:FindFirstChild("Remote")
    if not remote or not remote:IsA("RemoteEvent") then Clutch.CanUse = true return end
    Clutch.FromManual = true
    pcall(function()
        remote:FireServer(CFrame.new(hrp.Position - Vector3.new(0, 3.15, 0)), 50)
        task.wait(0.06)
        h:ChangeState(Enum.HumanoidStateType.Jumping)
        h.Jump = true
        task.wait(0.24)
        if fb.Parent == c then fb.Parent = bp end
    end)
    Clutch.FromManual = false
    clutchShow("Bomb: " .. tostring(Clutch.Cooldown) .. "s")
    local t0 = time()
    local charAtStart = player.Character
    while time() - t0 < Clutch.Cooldown do
        if player.Character ~= charAtStart then clutchHide() Clutch.CanUse = true return end
        clutchShow("Bomb: " .. tostring(Clutch.Cooldown - math.floor(time() - t0)) .. "s")
        task.wait(1)
    end
    clutchHide()
    Clutch.CanUse = true
end

local function clutchFireflyNear(part)
    local best, dist = nil, 40
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character and p.Character:FindFirstChild("Fireflies") then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (r.Position - part.Position).Magnitude
                if d < dist then dist = d best = p end
            end
        end
    end
    return best
end

local KORBLOX_MESH_ID    = "rbxassetid://101851696"
local KORBLOX_TEXTURE_ID = "rbxassetid://101851254"

local function applyKorbloxR6(ch)
    local rightLeg = ch:FindFirstChild("Right Leg")
    if not rightLeg then return end
    for _, child in ipairs(rightLeg:GetChildren()) do
        if child:IsA("SpecialMesh") or child:IsA("CharacterMesh") then
            table.insert(korbloxObs, child)
        end
    end
    local mesh = Instance.new("SpecialMesh")
    mesh.MeshType  = Enum.MeshType.FileMesh
    mesh.MeshId    = KORBLOX_MESH_ID
    mesh.TextureId = KORBLOX_TEXTURE_ID
    mesh.Scale     = Vector3.new(1,1,1)
    mesh.Parent    = rightLeg
    table.insert(korbloxObs, mesh)
end

local function applyKorbloxR15(ch)
    local rUpper = ch:FindFirstChild("RightUpperLeg")
    if not rUpper then return end
    rUpper.Transparency = 1
    table.insert(korbloxObs, rUpper)
    local rLower = ch:FindFirstChild("RightLowerLeg")
    local rFoot  = ch:FindFirstChild("RightFoot")
    if rLower then rLower.Transparency=1; table.insert(korbloxObs, rLower) end
    if rFoot  then rFoot.Transparency=1;  table.insert(korbloxObs, rFoot)  end
    local kLeg = Instance.new("Part")
    kLeg.Size=Vector3.new(1,2,1); kLeg.Anchored=false; kLeg.CanCollide=false; kLeg.Parent=ch
    table.insert(korbloxObs, kLeg)
    local mesh = Instance.new("SpecialMesh")
    mesh.MeshType=Enum.MeshType.FileMesh; mesh.MeshId=KORBLOX_MESH_ID
    mesh.TextureId=KORBLOX_TEXTURE_ID; mesh.Parent=kLeg
    table.insert(korbloxObs, mesh)
    local weld = Instance.new("Weld")
    weld.Part0=rUpper; weld.Part1=kLeg; weld.C0=CFrame.new(0,-0.8,0); weld.Parent=kLeg
    table.insert(korbloxObs, weld)
end

local function removeKorblox(ch)
    for _, obj in pairs(korbloxObs) do
        if obj and obj.Parent then pcall(function() obj:Destroy() end) end
    end
    korbloxObs = {}
    local rUpper = ch:FindFirstChild("RightUpperLeg")
    if rUpper then rUpper.Transparency=0 end
    local rLower = ch:FindFirstChild("RightLowerLeg")
    if rLower then rLower.Transparency=0 end
    local rFoot  = ch:FindFirstChild("RightFoot")
    if rFoot  then rFoot.Transparency=0  end
    local rLeg = ch:FindFirstChild("Right Leg")
    if rLeg then
        for _, child in ipairs(rLeg:GetChildren()) do
            if child:IsA("SpecialMesh") and child.MeshId==KORBLOX_MESH_ID then child:Destroy() end
        end
    end
end

local function applyKorbloxToChar(ch)
    task.wait(0.2)
    local hum = ch:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if korbloxActive then
        if hum.RigType==Enum.HumanoidRigType.R6 then applyKorbloxR6(ch)
        else applyKorbloxR15(ch) end
    else
        removeKorblox(ch)
    end
end

local HEADLESS_MESH_ID = "rbxassetid://1095708"
local origHeadTrans    = nil

local function removeFaceH(head)
    local face = head:FindFirstChild("face")
    if face then face:Destroy() end
end

local function restoreFaceH(head)
    if not head:FindFirstChild("face") then
        local newFace = Instance.new("Decal")
        newFace.Name    = "face"
        newFace.Texture = "rbxasset://textures/face.png"
        newFace.Parent  = head
    end
end

local function applyHeadless(head)
    if not head then return end
    origHeadTrans      = head.Transparency
    head.Transparency  = 1
    head.CanCollide    = false
    removeFaceH(head)
    if not head:FindFirstChild("HeadlessMesh") then
        local mesh = Instance.new("SpecialMesh")
        mesh.Name      = "HeadlessMesh"
        mesh.MeshType  = Enum.MeshType.FileMesh
        mesh.MeshId    = HEADLESS_MESH_ID
        mesh.Scale     = Vector3.new(0.001,0.001,0.001)
        mesh.Parent    = head
    end
end

local function removeHeadless(head)
    if not head then return end
    if origHeadTrans then head.Transparency=origHeadTrans end
    head.CanCollide = true
    restoreFaceH(head)
    local mesh = head:FindFirstChild("HeadlessMesh")
    if mesh then mesh:Destroy() end
end

local function applyHeadlessToChar(ch)
    task.wait(0.2)
    local head = ch:FindFirstChild("Head")
    if head then
        if headlessActive then applyHeadless(head)
        else removeHeadless(head) end
    end
end

player.CharacterAdded:Connect(function(ch)
    ch:WaitForChild("Humanoid")
    if korbloxActive  then applyKorbloxToChar(ch)  end
    if headlessActive then
        ch:WaitForChild("Head")
        applyHeadlessToChar(ch)
    end
end)

local AutoTargetEnabled    = false
local AutoTargetConnection = nil
local AutoTargetPrediction = 0.125

local function GetBestMurdererTarget()
    local target   = nil
    local shortest = math.huge
    local cam      = workspace.CurrentCamera
    local screenCenter = cam.ViewportSize / 2
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            local hasKnife = p.Character:FindFirstChild("Knife")
                or (p.Backpack and p.Backpack:FindFirstChild("Knife"))
            if hasKnife then
                local torso = p.Character:FindFirstChild("UpperTorso")
                    or p.Character:FindFirstChild("HumanoidRootPart")
                if torso then
                    local pos, onScreen = cam:WorldToViewportPoint(torso.Position)
                    if onScreen then
                        local dist = (Vector2.new(pos.X, pos.Y) - screenCenter).Magnitude
                        if dist < shortest then shortest=dist; target=torso end
                    end
                end
            end
        end
    end
    return target
end

local function StartAutoTarget()
    if AutoTargetConnection then AutoTargetConnection:Disconnect() end
    AutoTargetConnection = RunService.RenderStepped:Connect(function()
        if not AutoTargetEnabled then return end
        local target = GetBestMurdererTarget()
        if target then
            local cam      = workspace.CurrentCamera
            local velocity = target.AssemblyLinearVelocity or target.Velocity or Vector3.zero
            local predicted = target.Position + (velocity * AutoTargetPrediction)
            cam.CFrame = CFrame.new(cam.CFrame.Position, predicted)
        end
    end)
end

local function StopAutoTarget()
    if AutoTargetConnection then AutoTargetConnection:Disconnect(); AutoTargetConnection=nil end
end

        local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
        if hum and TrollState.SpinAutoRotate ~= nil then
            hum.AutoRotate = TrollState.SpinAutoRotate

    hum.AutoRotate = false
    RunService:BindToRenderStep(TrollState.SpinnerBind, Enum.RenderPriority.Last.Value, function(dt)
        if not TrollState.Spinner then
            pcall(function() RunService:UnbindFromRenderStep(TrollState.SpinnerBind) end)

        local c = player.Character
        local r = c and c:FindFirstChild("HumanoidRootPart")
        if not r then return end
        r.CFrame = r.CFrame * CFrame.Angles(0, TrollState.SpinSpeed * dt, 0)
    end)
end

local function setupFakeLag(state)
    TrollState.FakeLag = state
    if TrollState.LagConn then TrollState.LagConn:Disconnect(); TrollState.LagConn = nil end
    if not state then return end
    TrollState.LagConn = spawn(function()
        while TrollState.FakeLag do

            local hrp = c and c:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Anchored = true
                task.wait(TrollState.LagAnchored)
                hrp.Anchored = false
            end
            task.wait(TrollState.LagInterval)
        end
    end)
end

local function setupSpeedGlitch(state)
    TrollState.SpeedGlitch = state
    if TrollState.GlitchConn then TrollState.GlitchConn:Disconnect(); TrollState.GlitchConn = nil end
    if not state then return end
    local accel = 0.3
    TrollState.GlitchConn = RunService.Heartbeat:Connect(function()
        local c = player.Character
        local hum = c and c:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        local st = hum:GetState()
        if st == Enum.HumanoidStateType.Jumping or st == Enum.HumanoidStateType.Freefall then
            local md = hum.MoveDirection
            if md.Magnitude > 0 then
                accel = math.min(accel + 0.05, 1)
                c:TranslateBy(md.Unit * TrollState.GlitchPower * accel / 150)
            end
        end
    end)
end

local function setupSpamJump(state)
    TrollState.SpamJump = state
    if TrollState.SpamJumpConn then TrollState.SpamJumpConn:Disconnect(); TrollState.SpamJumpConn = nil end
    if not state then return end
    TrollState.SpamJumpConn = spawn(function()
        while TrollState.SpamJump do
            local c = player.Character
            local h = c and c:FindFirstChildOfClass("Humanoid")
            if h and h.FloorMaterial ~= Enum.Material.Air then
                h:ChangeState(Enum.HumanoidStateType.Jumping)
                task.wait(0.55)
            else
                task.wait(0.03)
            end
        end
    end)
end

local function setupFreeze(state)
    TrollState.Freeze = state
    local c = player.Character
    if not c then return end
    if state then
        for _, p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") then
                TrollState.FrozenParts[p] = p.Anchored
                p.Anchored = true
            end
        end
    else
        for p, was in pairs(TrollState.FrozenParts) do
            if p and p.Parent then p.Anchored = was or false end
        end
        TrollState.FrozenParts = {}
    end
end

        if cam then
            cam.CFrame = cam.CFrame * CFrame.new(0, 0, 0, 1, 0, 0, 0, 0.65, 0, 0, 0, 1)
        end
    end)
end
