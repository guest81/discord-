local function geniter(f, s, c)
	if type(f) ~= "function" then
		local mt = getmetatable(f)

		if type(mt) == "table" and mt.__iter then
			f, s, c = mt.__iter(f)
		elseif type(f) == "table" then
			f, s, c = next, f, nil
		end
	end

	return function()
		local r = table.pack(f(s, c))

		if r[1] == nil then
			return false
		end

		c = r[1]

		return true, table.unpack(r, 1, r.n)
	end
end

local function playSound(soundName)
	if not soundSettings.Enabled then
		return
	end

	local soundId = soundIds[soundName]

	if not soundId then
		return
	end

	pcall(function()
		local Sound = Instance.new("Sound")

		Sound.SoundId = soundId
		Sound.Volume = 1
		Sound.Parent = SoundService
		Sound:Play()
		Debris:AddItem(Sound, 5)
	end)
end

local function normalizeText(text)
	return tostring(text or ""):gsub("%s+", ""):lower()
end
local function readSavedKey()
	if type(isfile) ~= "function" or type(readfile) ~= "function" then
		return ""
	end

	local ok, result = pcall(readfile, "gscripts_timebomb_key.txt")

	if ok then
		return normalizeText(result)
	end

	return ""
end
local function writeSavedKey()
	if type(writefile) == "function" then
		return (pcall(writefile, "gscripts_timebomb_key.txt", "gscriptsboost"))
	end

	return false
end

					local entry = table.pack(iterator())

					if entry[1] then
						leaderstats = entry[3]

						continue
					end

					return
				end

				break
			end

			return leaderstats
		end
		local function findKillsValue(instance)
			for _, child in ipairs(instance:GetChildren()) do
				if child.Name:lower() == "kills" and child:IsA("ValueBase") then
					return child
				end
			end

			for _, descendant in ipairs(instance:GetDescendants()) do
				if descendant.Name:lower() == "kills" and descendant:IsA("ValueBase") then
					return descendant
				end
			end
		end
		local function parseNumber(rawValue)
			if type(rawValue) == "string" then
				rawValue = rawValue:gsub(",", "")
			end

			return tonumber(rawValue)
		end
		local function getKillCount()
			local leaderstats = findLeaderstats()

			if not leaderstats then
				return nil, nil
			end

			local killsValue = findKillsValue(leaderstats)

			if killsValue then
				return killsValue, parseNumber(killsValue.Value)
			end

			local attribute = leaderstats:GetAttribute("Kills")

			if attribute ~= nil then
				return leaderstats, parseNumber(attribute)
			end

			local attribute2 = LocalPlayer:GetAttribute("Kills")

			if attribute2 == nil then
				return
			end

			return LocalPlayer, parseNumber(attribute2)
		end
		local function checkKillCount()
			if not state.killMessageOn then
				return
			end

			local killSource, killCount = getKillCount()

			if not killSource or not killCount then
				state.killMessageTrackedSource = nil
				state.killMessageLastKills = nil

				return
			end

			if state.killMessageTrackedSource ~= killSource then
				state.killMessageTrackedSource = killSource
				state.killMessageLastKills = killCount

				return
			end

			local killMessageLastKills = state.killMessageLastKills

			state.killMessageLastKills = killCount

			if killMessageLastKills and killMessageLastKills < killCount then
				sendChatMessage(killMessages[math.random(1, #killMessages)])
			end
		end
		local function refreshKillMessage()
			if not state.killMessageOn then
				return
			end

			disconnectKillMessage()

			local leaderstats = findLeaderstats()

			if not leaderstats then
				state.killMessageLastKills = nil

				return
			end

			local killsValue = findKillsValue(leaderstats)

			if killsValue then
				state.killMessageLastKills = parseNumber(killsValue.Value)
				state.killMessageValueConnection = killsValue.Changed:Connect(function()
					if not state.killMessageOn then
						return
					end

					local killCount = parseNumber(killsValue.Value)
					local killMessageLastKills = state.killMessageLastKills

					if not killCount then
						return
					end

					state.killMessageLastKills = killCount

					if killMessageLastKills then
						if killMessageLastKills < killCount then
							sendChatMessage(killMessages[math.random(1, #killMessages)])
						end
					end
				end)
			else
				state.killMessageLastKills = nil
			end

			state.killMessageStatsConnection = leaderstats.ChildAdded:Connect(function(child)
				if child.Name:lower() == "kills" then
					refreshKillMessage()
				end
			end)
		end

		stopKillMessage = function()
			state.killMessageOn = false
			disconnectKillMessage()

			if state.killMessagePlayerConnection then
				state.killMessagePlayerConnection:Disconnect()
				state.killMessagePlayerConnection = nil
			end

			if state.killMessagePlayerRemovedConnection then
				state.killMessagePlayerRemovedConnection:Disconnect()
				state.killMessagePlayerRemovedConnection = nil
			end

			state.killMessageLastKills = nil
			state.killMessageTrackedSource = nil
			state.killMessagePollTimer = 0

			if state.killMessagePollConnection then
				state.killMessagePollConnection:Disconnect()
				state.killMessagePollConnection = nil
			end
		end
		startKillMessage = function()
			stopKillMessage()
			state.killMessageOn = true
			state.killMessagePlayerConnection = LocalPlayer.ChildAdded:Connect(function(child)
				if child.Name:lower() == "leaderstats" then
					refreshKillMessage()
				end
			end)
			state.killMessagePlayerRemovedConnection = LocalPlayer.ChildRemoved:Connect(function(child)
				if child.Name:lower() == "leaderstats" then
					disconnectKillMessage()
					state.killMessageTrackedSource = nil
					state.killMessageLastKills = nil
				end
			end)
			state.killMessagePollConnection = RunService.Heartbeat:Connect(function(dt)
				state.killMessagePollTimer += dt

				if state.killMessagePollTimer >= 0.1 then
					state.killMessagePollTimer = 0
					checkKillCount()
				end
			end)
			refreshKillMessage()
		end
	end

	local function findNearestPlayer(maxRange, excludePlayer, skipZone, requireSameArena)
		local localRoot = getRootPart(LocalPlayer)

		if not localRoot then
			return nil, nil
		end

		local nearestPlayer
		local huge = maxRange or math.huge

		for _, player in ipairs(Players:GetPlayers()) do
			if player ~= LocalPlayer and player ~= excludePlayer and player.Character then
				if isAlive(player) then
					local HumanoidRootPart = player.Character:FindFirstChild("HumanoidRootPart")

					if HumanoidRootPart then
						if skipZone then
							if isInsideZone(HumanoidRootPart.Position) then
								continue
							end
						end

						if not requireSameArena or isInSameArena(player) then
							local Magnitude = (localRoot.Position - HumanoidRootPart.Position).Magnitude

							if Magnitude < huge and (not state.teamCheckOn or not hasHighlight(player.Character)) then
								nearestPlayer = player
								huge = Magnitude
							end
						end
					end
				end
			end
		end

		return nearestPlayer, huge
	end
	local function getBombTool()
		local Character = LocalPlayer.Character
		local backpackOwner, FindFirstChild, s1

		if Character then
			for _, child in ipairs(Character:GetChildren()) do
				if child:IsA("Tool") then
					if child.Name:lower():find("bomb") then
						return child, true
					end
				end
			end

			backpackOwner = LocalPlayer
			FindFirstChild = LocalPlayer.FindFirstChild
			s1 = "Backpack"
		else
			backpackOwner = LocalPlayer
			FindFirstChild = LocalPlayer.FindFirstChild
			s1 = "Backpack"
		end

		local backpack = FindFirstChild(backpackOwner, s1)

		if backpack then
			for _, child in ipairs(backpack:GetChildren()) do
				if child:IsA("Tool") then
					if child.Name:lower():find("bomb") then
						return child, false
					end
				end
			end
		end

		return nil, false
	end

	local autoGiveBombStep

	do
		local function isPathBlocked(origin, destination)
			local Lobby = workspace:FindFirstChild("Lobby")

			if not Lobby then
				return false
			end

			local part = Lobby:FindFirstChild("part")

			if not part then
				return false
			end

			local distance = destination - origin
			local Magnitude = distance.Magnitude

			if Magnitude < 0.01 then
				return false
			end

			local raycastParams = RaycastParams.new()

			raycastParams.FilterType = Enum.RaycastFilterType.Include
			raycastParams.FilterDescendantsInstances = { part }

			return workspace:Raycast(origin, distance.Unit * Magnitude, raycastParams) ~= nil
		end
		local function clearBombBusy()
			state.bombBusy = false
		end
		local function giveBomb(targetPlayer, localRoot, originCFrame, excludedPlayer)
			if not targetPlayer then
				clearBombBusy()

				return
			end

			if not isAlive(targetPlayer) or not isInSameArena(targetPlayer) then
				local giveTarget = findNearestPlayer(state.GIVE_BOMB_RANGE, excludedPlayer, false, true)

				if giveTarget then
					giveBomb(giveTarget, localRoot, originCFrame, targetPlayer)
				else
					clearBombBusy()
				end

				return
			end

			local targetRoot = getRootPart(targetPlayer)

			if not targetRoot then
				local giveTarget = findNearestPlayer(state.GIVE_BOMB_RANGE, excludedPlayer, false, true)

				if giveTarget then
					giveBomb(giveTarget, localRoot, originCFrame, targetPlayer)
				else
					clearBombBusy()
				end

				return
			end

			if isPathBlocked(localRoot.Position, targetRoot.Position) then
				clearBombBusy()

				return
			end

			local bombTool, bombEquipped = getBombTool()
			local Humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")

			if bombTool and Humanoid and not bombEquipped then
				Humanoid:EquipTool(bombTool)
				RunService.Heartbeat:Wait()
				getBombTool()
			end

			local Attachment = Instance.new("Attachment")

			Attachment.Parent = localRoot

			local LinearVelocity = Instance.new("LinearVelocity")

			LinearVelocity.Attachment0 = Attachment
			LinearVelocity.MaxForce = math.huge
			LinearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
			LinearVelocity.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
			LinearVelocity.VectorVelocity = Vector3.zero
			LinearVelocity.Parent = localRoot

			local n3 = 0
			local n4 = 0
			local connection

			connection = RunService.Heartbeat:Connect(function(dt)
				n3 += dt
				n4 = math.min(n4 + dt * 7, 1)

				local localRoot2 = getRootPart(LocalPlayer)

				if not localRoot2 then
					connection:Disconnect()
					LinearVelocity:Destroy()
					Attachment:Destroy()
					clearBombBusy()

					return
				end

				localRoot2.AssemblyLinearVelocity = Vector3.zero
				localRoot2.AssemblyAngularVelocity = Vector3.zero

				local Humanoid2 = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")

				if Humanoid2 and Humanoid2.Health <= 0 then
					connection:Disconnect()
					LinearVelocity:Destroy()
					Attachment:Destroy()
					clearBombBusy()

					return
				end

				if not isAlive(targetPlayer) then
					connection:Disconnect()
					LinearVelocity:Destroy()
					Attachment:Destroy()

					local localRoot3 = getRootPart(LocalPlayer)

					if localRoot3 then
						localRoot3.CFrame = originCFrame
					end

					local giveTarget = findNearestPlayer(state.GIVE_BOMB_RANGE, targetPlayer, false, true)

					if giveTarget then
						giveBomb(giveTarget, localRoot, originCFrame, targetPlayer)
					else
						clearBombBusy()
					end

					return
				end

				local targetRoot = getRootPart(targetPlayer)

				if not targetRoot or not isInSameArena(targetPlayer) then
					connection:Disconnect()
					LinearVelocity:Destroy()
					Attachment:Destroy()

					local localRoot3 = getRootPart(LocalPlayer)

					if localRoot3 then
						localRoot3.CFrame = originCFrame
					end

					local giveTarget = findNearestPlayer(state.GIVE_BOMB_RANGE, targetPlayer, false, true)

					if giveTarget then
						giveBomb(giveTarget, localRoot, originCFrame, targetPlayer)
					else
						clearBombBusy()
					end

					return
				end

				if not (state.HOLD_TIME <= n3) then
					local lerpedPosition = localRoot2.Position:Lerp(targetRoot.Position, n4)

					localRoot2.CFrame = CFrame.new(lerpedPosition)

					return
				end

				connection:Disconnect()
				LinearVelocity:Destroy()
				Attachment:Destroy()

				local localRoot = getRootPart(LocalPlayer)

				if localRoot then
					localRoot.CFrame = originCFrame
				end

				task.delay(0.1, clearBombBusy)
			end)
		end

		autoGiveBombStep = function()
			if state.bombBusy then
				return
			end

			local localRoot = getRootPart(LocalPlayer)

			if not localRoot then
				return
			end

			local giveTarget = findNearestPlayer(state.GIVE_BOMB_RANGE, nil, false, true)

			end)
			giveBomb(giveTarget, localRoot, localRoot.CFrame, nil)
		end
	end

	do
		local function scheduleToolCheck(toolName)
			task.delay(0.25, function()
				for _, player in ipairs(Players:GetPlayers()) do
					if player == LocalPlayer then
						continue
					end

					local Character = player.Character
					local Backpack = player:FindFirstChild("Backpack")

					local function hasTool(container)
						if not container then
							return false
						end

						local iterFn, iterState, iterKey = ipairs(container:GetChildren())
						local iterator = geniter(iterFn, iterState, iterKey)
						local entry = table.pack(iterator())

						if entry[1] then
							local child = entry[3]

							while true do
								if child:IsA("Tool") and child.Name == toolName then
									return true
								end

								local entry = table.pack(iterator())

								if not entry[1] then
									break
								end

								child = entry[3]
							end
						end

						return false
					end

					if not hasTool(Character) and not hasTool(Backpack) then
						continue
					end

					emptyFunction(player.Name .. ": You gave them the bomb")

					return
				end
			end)
		end
		local function watchToolRemoval(character)
			if not character then
				return
			end

			character.ChildRemoved:Connect(function(child)
				if child:IsA("Tool") then
					if child.Name:lower():find("bomb") then
						scheduleToolCheck(child.Name)
					end
				end
			end)
		end

			if Character then
				watchToolRemoval(Character)
			end

			watchToolRemoval(LocalPlayer:FindFirstChild("Backpack"))
			LocalPlayer.CharacterAdded:Connect(function(character)
				watchToolRemoval(character)
				task.delay(0.1, function()
					watchToolRemoval(LocalPlayer:FindFirstChild("Backpack"))
				end)
			end)
		end

		watchLocalCharacter()
	end

			end

			if not state.tracerAtt0 or not state.tracerAtt0.Parent then
				if state.tracerBeam then
					state.tracerBeam:Destroy()
					state.tracerBeam = nil
				end

				state.tracerAtt0 = Instance.new("Attachment")
				state.tracerAtt0.Parent = localRoot
				state.tracerBeam = Instance.new("Beam")
				state.tracerBeam.Attachment0 = state.tracerAtt0
				state.tracerBeam.Attachment1 = state.tracerAtt1
				state.tracerBeam.Color = ColorSequence.new(Color3.fromRGB(255, 50, 50))
				state.tracerBeam.Width0 = 0.08
				state.tracerBeam.Width1 = 0.08
				state.tracerBeam.FaceCamera = true
				state.tracerBeam.Transparency = NumberSequence.new(0.2)
				state.tracerBeam.LightEmission = 0.5
				state.tracerBeam.Parent = localRoot
			end

			local closePlayer = findNearestPlayer(500)

			if not closePlayer then
				if state.tracerBeam then
					state.tracerBeam.Enabled = false
				end

				return
			end

			local closeRoot = getRootPart(closePlayer)

			if not closeRoot then
				if state.tracerBeam then
					state.tracerBeam.Enabled = false
				end

				return
			end

			state.tracerHolder.CFrame = CFrame.new(closeRoot.Position)

			if state.tracerBeam then
				state.tracerBeam.Enabled = true
			end
		end)
	end

	do

		local function getLocalCharacterInfo()
			local Character = LocalPlayer.Character

			if not Character then
				return nil
			end

			local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

			if not HumanoidRootPart then
				return nil
			end

			raycastParams.FilterDescendantsInstances = { Character }

			local groundHit
			local n5 = 3
			local CFrame_ = HumanoidRootPart.CFrame

			for i = 0, 7 do
				local hit = workspace:Raycast(HumanoidRootPart.Position, (CFrame_ * CFrame.Angles(0, math.rad(i * 45), 0)).LookVector * 2, raycastParams)

				if hit and hit.Instance and hit.Distance < n5 then
					n5 = hit.Distance
					groundHit = hit
				end
			end

			local blockcastResult = workspace:Blockcast(CFrame_ * CFrame.new(0, -1, -0.5), Vector3.new(1.5, 1, 0.5), CFrame_.LookVector * 1.5, raycastParams)

			if blockcastResult and blockcastResult.Instance and blockcastResult.Distance < n5 then
				groundHit = blockcastResult
			end

			return groundHit
		end
		local function performWallhop(part)
			if not state.jumpReady then
				return
			end

			local Character = LocalPlayer.Character
			local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
			local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")

			if not Humanoid or not HumanoidRootPart or not workspace.CurrentCamera or not part then
				return
			end

			if Humanoid:GetState() == Enum.HumanoidStateType.Dead then
				return
			end

			state.jumpReady = false

			local Normal = part.Normal
			local vector3 = Vector3.new(Normal.X, 0, Normal.Z)

			if vector3.Magnitude < 0.1 then
				vector3 = -(part.Position - HumanoidRootPart.Position) * Vector3.new(1, 0, 1)
			end

			local Unit = Vector3.new(vector3.X, 0, vector3.Z).Unit

			if Unit.Magnitude < 0.1 then
				Unit = Vector3.new(0, 0, 1)
			end

			local vector32 = Vector3.new(workspace.CurrentCamera.CFrame.LookVector.X, 0, workspace.CurrentCamera.CFrame.LookVector.Z)
			local Unit2 = vector32.Magnitude > 0.1 and vector32.Unit or Unit
			local dotProduct = math.clamp(Unit:Dot(Unit2), -1, 1)
			local angleBetween = math.acos(dotProduct)
			local turnDirection = -math.sign(Unit:Cross(Unit2).Y)

			if turnDirection == 0 then
				angleBetween = 0
			end

			HumanoidRootPart.CFrame = CFrame.lookAt(HumanoidRootPart.Position, HumanoidRootPart.Position + CFrame.Angles(0, (turnDirection == 1 and math.min(angleBetween, math.rad(20)) or turnDirection == -1 and math.min(angleBetween, math.rad(100)) or 0) * turnDirection, 0) * Unit)
			RunService.Heartbeat:Wait()

			if Humanoid:GetState() ~= Enum.HumanoidStateType.Dead then
				Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
				HumanoidRootPart.CFrame *= CFrame.Angles(0, -1, 0)
				task.wait(0.15)
				HumanoidRootPart.CFrame *= CFrame.Angles(0, 1, 0)
				task.wait(0.05)
				HumanoidRootPart.CFrame = CFrame.lookAt(HumanoidRootPart.Position, HumanoidRootPart.Position + -Unit)
			end

			task.wait(0.1)
			state.jumpReady = true
		end

		UserInputService.JumpRequest:Connect(function()

			end

			local localInfo = getLocalCharacterInfo()

			end
		end)

	local stopAutoTrack, startAutoTrack

	do
		local function extractNumber(text)
			local numberText = string.match(text, "%d+")

			return numberText and tonumber(numberText) or nil
		end
		local function findNumberLabel(container)
			if not container then
				return false
			end

			for _, descendant in ipairs(container:GetDescendants()) do
				if descendant:IsA("TextLabel") then
					local labelNumber = extractNumber(descendant.Text)

					if labelNumber ~= nil and labelNumber <= 2 then
						return true
					end
				end
			end

			return false
		end
		local function steerToward(part, targetPosition)
			local toTarget = targetPosition - part.Position

			if toTarget.Magnitude < 0.01 then
				return
			end

			part.CFrame = CFrame.lookAt(part.Position, part.Position + toTarget.Unit)
		end
		local function steerAwayFrom(part, position)
			local fromTarget = part.Position - position

			if fromTarget.Magnitude < 0.01 then
				return
			end

			part.CFrame = CFrame.lookAt(part.Position, part.Position + fromTarget.Unit)
		end
		local function stopHumanoidMovement(humanoid)
			humanoid:Move(Vector3.zero)

			local HumanoidRootPart = humanoid.Parent and humanoid.Parent:FindFirstChild("HumanoidRootPart")

			if HumanoidRootPart then
				humanoid:MoveTo(HumanoidRootPart.Position)
			end
		end
		local function moveCharacter(targetPosition, character, targetCharacter)
			local HumanoidRootPart = character:FindFirstChild("HumanoidRootPart")
			local Humanoid = character:FindFirstChildOfClass("Humanoid")

			if not HumanoidRootPart or not Humanoid then
				return
			end

			local offsetToTarget = targetPosition - HumanoidRootPart.Position
			local vector3 = Vector3.new(offsetToTarget.X, 0, offsetToTarget.Z)

			if vector3.Magnitude < 0.1 then
				stopHumanoidMovement(Humanoid)

				return
			end

			local raycastParams = RaycastParams.new()

			raycastParams.FilterType = Enum.RaycastFilterType.Exclude
			raycastParams.FilterDescendantsInstances = {
				character,
				targetCharacter
			}

			local hit = workspace:Raycast(HumanoidRootPart.Position, vector3, raycastParams)

			if not hit then
				Humanoid:MoveTo(targetPosition)

				return
			end

			local hitPoint = hit.Position - vector3.Unit * 3
			local vector32 = Vector3.new(-vector3.Unit.Z, 0, vector3.Unit.X)
			local rightProbe = hitPoint + vector32 * 5
			local leftProbe = hitPoint - vector32 * 5
			local hit2 = workspace:Raycast(HumanoidRootPart.Position, rightProbe - HumanoidRootPart.Position, raycastParams)
			local hit3 = workspace:Raycast(HumanoidRootPart.Position, leftProbe - HumanoidRootPart.Position, raycastParams)

			if not hit2 then
				Humanoid:MoveTo(rightProbe)
			elseif hit3 then
				Humanoid:MoveTo(hitPoint)
			else
				Humanoid:MoveTo(leftProbe)
			end
		end

		stopAutoTrack = function()
			state.autoTrackOn = false

			if state.autoTrackConn then
				state.autoTrackConn:Disconnect()
				state.autoTrackConn = nil
			end

			state.autoTrackTarget = nil
			state.autoTrackHadBomb = false
			state.autoTrackFleeing = false
			state.autoTrackFleeStart = 0
			state.autoTrackCharacter = nil
			state.autoTrackTimer = 0

			local Character = LocalPlayer.Character
			local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")

			if Humanoid then
				Humanoid.AutoRotate = state.autoTrackOriginalAutoRotate
				stopHumanoidMovement(Humanoid)
			end
		end
		startAutoTrack = function()
			stopAutoTrack()
			state.autoTrackOn = true
			state.autoTrackOriginalAutoRotate = true
			state.autoTrackTimer = 0

			local Character = LocalPlayer.Character
			local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")

			if Humanoid then
				state.autoTrackOriginalAutoRotate = Humanoid.AutoRotate
			end

			state.autoTrackConn = RunService.Heartbeat:Connect(function(dt)
				if not state.autoTrackOn then
					return
				end

				state.autoTrackTimer += dt

				if state.autoTrackTimer < state.AT_UPDATE_RATE then
					return
				end

				state.autoTrackTimer = 0

				local Character2 = LocalPlayer.Character
				local Humanoid2 = Character2 and Character2:FindFirstChildOfClass("Humanoid")
				local HumanoidRootPart = Character2 and Character2:FindFirstChild("HumanoidRootPart")

				if not Character2 or not Humanoid2 or not HumanoidRootPart or Humanoid2.Health <= 0 then
					return
				end

				if state.autoTrackCharacter ~= Character2 then
					state.autoTrackCharacter = Character2
					state.autoTrackOriginalAutoRotate = Humanoid2.AutoRotate
					state.autoTrackHadBomb = false
					state.autoTrackFleeing = false
					state.autoTrackTarget = nil
				end

				if isInsideZone(HumanoidRootPart.Position) then
					stopHumanoidMovement(Humanoid2)
					Humanoid2.AutoRotate = state.autoTrackOriginalAutoRotate

					return
				end

				local bombTool = getBombTool()
				local bombTarget = findNearestPlayer(state.BOMB_RANGE, nil, true)

				if not bombTool then
					if state.autoTrackHadBomb then
						state.autoTrackHadBomb = false
						state.autoTrackFleeing = true
						state.autoTrackFleeStart = tick()
					end

					if not state.autoTrackFleeing then
						Humanoid2.AutoRotate = state.autoTrackOriginalAutoRotate
						state.autoTrackTarget = nil
						stopHumanoidMovement(Humanoid2)

						return
					end

					Humanoid2.AutoRotate = false

					if state.AT_FLEE_TIME < tick() - state.autoTrackFleeStart then
						state.autoTrackFleeing = false
						state.autoTrackTarget = nil
						Humanoid2.AutoRotate = state.autoTrackOriginalAutoRotate
						stopHumanoidMovement(Humanoid2)

						return
					end

					if not bombTarget or not bombTarget.Character then
						state.autoTrackFleeing = false
						Humanoid2.AutoRotate = state.autoTrackOriginalAutoRotate
						stopHumanoidMovement(Humanoid2)

						return
					end

					state.autoTrackTarget = bombTarget

					local HumanoidRootPart2 = bombTarget.Character:FindFirstChild("HumanoidRootPart")

					if not HumanoidRootPart2 then
						return
					end

					local awayVector = HumanoidRootPart.Position - HumanoidRootPart2.Position

					if awayVector.Magnitude < 0.01 then
						awayVector = Vector3.new(1, 0, 0)
					end

					local retreatPosition = HumanoidRootPart.Position + awayVector.Unit * 20

					steerAwayFrom(HumanoidRootPart, HumanoidRootPart2.Position)
					moveCharacter(retreatPosition, Character2, bombTarget.Character)

					return
				end

				state.autoTrackHadBomb = true
				state.autoTrackFleeing = false
				Humanoid2.AutoRotate = false

				if bombTarget then
					if bombTarget.Character then
						state.autoTrackTarget = bombTarget

						local HumanoidRootPart2 = bombTarget.Character:FindFirstChild("HumanoidRootPart")

						if not HumanoidRootPart2 then
							stopHumanoidMovement(Humanoid2)

							return
						end

						local Magnitude = (HumanoidRootPart.Position - HumanoidRootPart2.Position).Magnitude

						if findNumberLabel(bombTool) then
							steerToward(HumanoidRootPart, HumanoidRootPart2.Position)

							if state.AT_TOUCH_DISTANCE < Magnitude then
								if state.AT_FOLLOW_DISTANCE + 2 < Magnitude then
									moveCharacter(HumanoidRootPart2.Position, Character2, bombTarget.Character)
								else
									Humanoid2:MoveTo(HumanoidRootPart2.Position)
								end
							else
								stopHumanoidMovement(Humanoid2)
							end
						else
							steerAwayFrom(HumanoidRootPart, HumanoidRootPart2.Position)

							if state.AT_FOLLOW_DISTANCE + 1 < Magnitude then
								moveCharacter(HumanoidRootPart2.Position + (HumanoidRootPart.Position - HumanoidRootPart2.Position).Unit * state.AT_FOLLOW_DISTANCE, Character2, bombTarget.Character)
							elseif Magnitude < state.AT_FOLLOW_DISTANCE - 1 then
								moveCharacter(HumanoidRootPart.Position + (HumanoidRootPart.Position - HumanoidRootPart2.Position).Unit * 4, Character2, bombTarget.Character)
							else
								stopHumanoidMovement(Humanoid2)
							end
						end

						return
					end
				end

				state.autoTrackTarget = nil
				stopHumanoidMovement(Humanoid2)
			end)
		end
	end

	local stopFixHitboxJumps, startFixHitboxJumps

	do
		local function expandCharacterHitbox(character)
			for _, r6Limb in ipairs(state.r6Limbs) do
				local limb = character:FindFirstChild(r6Limb)

				if limb then
					if limb:IsA("BasePart") then
						local originalLimbSize = state.hbOriginalSizes[limb]

						if originalLimbSize then
							pcall(function()
								limb.Size = originalLimbSize
								limb.CanCollide = true
								limb.Massless = false

								local VisualSizerMesh = limb:FindFirstChild("VisualSizerMesh")

								if VisualSizerMesh then
									VisualSizerMesh:Destroy()
								end
							end)
							state.hbOriginalSizes[limb] = nil
						end
					end
				end
			end
		end

		stopFixHitboxJumps = function()
			if state.fixHitboxConnection then
				state.fixHitboxConnection:Disconnect()
				state.fixHitboxConnection = nil
			end

			if state.fixHitboxCharacterAdded then
				state.fixHitboxCharacterAdded:Disconnect()
				state.fixHitboxCharacterAdded = nil
			end

			if state.fixHitboxCharacterRemoving then
				state.fixHitboxCharacterRemoving:Disconnect()
				state.fixHitboxCharacterRemoving = nil
			end

			for k, fixHitboxOriginalCollision in pairs(state.fixHitboxOriginalCollisions) do
				if k and k.Parent then
					k.CanCollide = fixHitboxOriginalCollision
				end
			end

			state.fixHitboxOriginalCollisions = {}
		end

		local function connectFixHitboxStepped()
			if state.fixHitboxConnection then
				state.fixHitboxConnection:Disconnect()
			end

			state.fixHitboxConnection = RunService.Stepped:Connect(function()
				if not LocalPlayer.Character then
					return
				end

				for _, player in ipairs(Players:GetPlayers()) do
					if player ~= LocalPlayer and player.Character then
						for _, child in ipairs(player.Character:GetChildren()) do
							if child:IsA("BasePart") then
								if state.fixHitboxOriginalCollisions[child] == nil then
									state.fixHitboxOriginalCollisions[child] = child.CanCollide
								end

								child.CanCollide = false
								child.Velocity = Vector3.new(0, 0, 0)
								child.RotVelocity = Vector3.new(0, 0, 0)
							end
						end
					end
				end
			end)
		end

			stopFixHitboxJumps()
			connectFixHitboxStepped()
			state.fixHitboxCharacterRemoving = LocalPlayer.CharacterRemoving:Connect(function()
				if state.fixHitboxConnection then
					state.fixHitboxConnection:Disconnect()
					state.fixHitboxConnection = nil
				end
			end)
			state.fixHitboxCharacterAdded = LocalPlayer.CharacterAdded:Connect(function()
				task.wait(0.5)

				if state.fixHitboxOn then
					connectFixHitboxStepped()
				end
			end)
		end
		RunService.Heartbeat:Connect(function()
			if not state.hitboxOn then
				return
			end

			local INCREASE_FACTOR = state.INCREASE_FACTOR
			local shrinkFactor = 1 / INCREASE_FACTOR

			for _, player in ipairs(Players:GetPlayers()) do
				if player ~= LocalPlayer and player.Character then
					local Character = player.Character
					local Humanoid = Character:FindFirstChildOfClass("Humanoid")

					if Humanoid and not (Humanoid.Health <= 0) then
						local Torso = Character:FindFirstChild("Torso") or Character:FindFirstChild("HumanoidRootPart")

						if Torso then
							Torso.CanCollide = false
						end

						for _, r6Limb in ipairs(state.r6Limbs) do
							local limbPart = Character:FindFirstChild(r6Limb)

							if limbPart then
								if limbPart:IsA("BasePart") then
									if not state.hbOriginalSizes[limbPart] then
										state.hbOriginalSizes[limbPart] = limbPart.Size
									end

									pcall(function()
										limbPart.Size = state.hbOriginalSizes[limbPart] * INCREASE_FACTOR
										limbPart.CanCollide = false
										limbPart.Massless = true

										local VisualSizerMesh = limbPart:FindFirstChild("VisualSizerMesh")

										if not VisualSizerMesh then
											VisualSizerMesh = Instance.new("SpecialMesh")
											VisualSizerMesh.Name = "VisualSizerMesh"
											VisualSizerMesh.MeshType = Enum.MeshType.FileMesh
											VisualSizerMesh.Parent = limbPart
										end

										VisualSizerMesh.Scale = Vector3.new(shrinkFactor, shrinkFactor, shrinkFactor)
									end)
								end
							end
						end
					else
						expandCharacterHitbox(Character)
					end
				end
			end
		end)
	end

	local stopPlayerHitboxes, startPlayerHitboxes

	do
		local function clearPartAdornments(part)
			if not part or not part:IsA("BasePart") then
				return
			end

			for _, child in ipairs(part:GetChildren()) do
				if child.Name == "PlayerHitboxFill" or child.Name == "PlayerHitboxEdges" then
					child:Destroy()
				end
			end
		end
		local function getValidCharacter(player)
			if not player then
				return false
			end

			local Character = player.Character

			if Character and Character:FindFirstChildOfClass("Tool") then
				return true
			end

			local Backpack = player:FindFirstChildOfClass("Backpack")

			if Backpack and Backpack:FindFirstChildOfClass("Tool") then
				return true
			end

			return false
		end
		local function addPartOutline(part, isTarget)
			if not part or not part:IsA("BasePart") then
				return
			end

			local color3 = isTarget and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(0, 255, 0)
			local PlayerHitboxEdges = part:FindFirstChild("PlayerHitboxEdges")

			if PlayerHitboxEdges and PlayerHitboxEdges:IsA("SelectionBox") then
				PlayerHitboxEdges.Color3 = color3
			end

			local PlayerHitboxFill = part:FindFirstChild("PlayerHitboxFill")

			if PlayerHitboxFill and PlayerHitboxFill:IsA("BoxHandleAdornment") then
				PlayerHitboxFill.Color3 = color3
			end
		end
		local function addPlayerPartHitbox(part, isTarget)
			if state.playerHitboxesOn and part then
				if part:IsA("BasePart") then
					clearPartAdornments(part)

					local color3 = isTarget and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(0, 255, 0)
					local SelectionBox = Instance.new("SelectionBox")

					SelectionBox.Name = "PlayerHitboxEdges"
					SelectionBox.Color3 = color3
					SelectionBox.LineThickness = 0.04
					SelectionBox.SurfaceTransparency = 1
					SelectionBox.Transparency = 0
					pcall(function()
						SelectionBox.CartographicMode = true
					end)
					SelectionBox.Adornee = part
					SelectionBox.Parent = part

					local BoxHandleAdornment = Instance.new("BoxHandleAdornment")

					BoxHandleAdornment.Name = "PlayerHitboxFill"
					BoxHandleAdornment.Size = part.Size
					BoxHandleAdornment.Color3 = color3
					BoxHandleAdornment.Transparency = 0.6
					BoxHandleAdornment.AlwaysOnTop = false
					BoxHandleAdornment.ZIndex = 9
					BoxHandleAdornment.Adornee = part
					BoxHandleAdornment.Parent = part

					return
				end
			end
		end
		local function trackPlayerHitbox(character, player)
			if not character or not player then
				return
			end

			task.spawn(function()
				local Head = character:FindFirstChild("Head") or character:WaitForChild("Head", 5)
				local Torso = character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso") or character:FindFirstChild("HumanoidRootPart")

				local function updatePlayerHitbox()
					if state.playerHitboxesOn then
						if character.Parent then
							local character2 = getValidCharacter(player)

							if Head then
								addPartOutline(Head, character2)
							end

							if Torso then
								addPartOutline(Torso, character2)
							end

							return
						end
					end
				end

				local character2 = getValidCharacter(player)

				if Head then
					addPlayerPartHitbox(Head, character2)
				end

				if Torso then
					addPlayerPartHitbox(Torso, character2)
				end

				table.insert(state.playerHitboxConnections, character.ChildAdded:Connect(function(child)
					if child:IsA("Tool") then
						updatePlayerHitbox()
					end
				end))
				table.insert(state.playerHitboxConnections, character.ChildRemoved:Connect(function(child)
					if child:IsA("Tool") then
						task.defer(updatePlayerHitbox)
					end
				end))

				local Backpack = player:FindFirstChildOfClass("Backpack")

				if Backpack then
					table.insert(state.playerHitboxConnections, Backpack.ChildAdded:Connect(function(child)
						if child:IsA("Tool") then
							updatePlayerHitbox()
						end
					end))
					table.insert(state.playerHitboxConnections, Backpack.ChildRemoved:Connect(function(child)
						if child:IsA("Tool") then
							task.defer(updatePlayerHitbox)
						end
					end))
				end
			end)
		end
		local function disconnectPlayerHitboxes()
			for _, playerHitboxConnection in ipairs(state.playerHitboxConnections) do
				playerHitboxConnection:Disconnect()
			end

			state.playerHitboxConnections = {}
		end

		stopPlayerHitboxes = function()
			disconnectPlayerHitboxes()

			local iterFn, iterState, iterKey = ipairs(Players:GetPlayers())
			local iterator = geniter(iterFn, iterState, iterKey)

			while true do
				local entry = table.pack(iterator())

				if not entry[1] then
					break
				end

				local Character = (entry[3]).Character

				if Character then
					local exitTo

					repeat
						for _, v in ipairs({
							"Head",
							"Torso",
							"UpperTorso",
							"HumanoidRootPart"
						}) do
							clearPartAdornments(Character:FindFirstChild(v))
						end

						exitTo = nil

						while true do
							local entry = table.pack(iterator())

							if entry[1] then
								Character = (entry[3]).Character

								if Character then
									exitTo = 1

									break
								end
							else
								break
							end
						end
					until exitTo ~= 1

					return
				end
			end
		end
		startPlayerHitboxes = function()
			stopPlayerHitboxes()

			for _, player in ipairs(Players:GetPlayers()) do
				if player.Character then
					trackPlayerHitbox(player.Character, player)
				end

				table.insert(state.playerHitboxConnections, player.CharacterAdded:Connect(function(character)
					if state.playerHitboxesOn then
						trackPlayerHitbox(character, player)
					end
				end))
			end

			table.insert(state.playerHitboxConnections, Players.PlayerAdded:Connect(function(player)
				table.insert(state.playerHitboxConnections, player.CharacterAdded:Connect(function(character)
					if state.playerHitboxesOn then
						trackPlayerHitbox(character, player)
					end
				end))

				if player.Character then
					trackPlayerHitbox(player.Character, player)
				end
			end))
		end
	end

	local function disableWalkAnimations(character)
		if not character then
			return
		end

		local Animate = character:FindFirstChild("Animate") or character:WaitForChild("Animate", 5)

		if Animate then
			if state.noWalkAnimateStates[character] == nil then
				state.noWalkAnimateStates[character] = Animate.Enabled
			end

			Animate.Enabled = false
		end

		local Humanoid = character:FindFirstChildOfClass("Humanoid")

		if not Humanoid then
			return
		end

		local Animator = Humanoid:FindFirstChildOfClass("Animator") or Humanoid:WaitForChild("Animator", 5)

		if not Animator then
			return
		end

			v:Stop()
		end

		table.insert(state.noWalkAnimationConnections, Animator.AnimationPlayed:Connect(function(track)
			if state.noWalkAnimationsOn then
				track:Stop()
			end
		end))
	end

		disableWalkAnimations(LocalPlayer.Character)
		table.insert(state.noWalkAnimationConnections, LocalPlayer.CharacterAdded:Connect(function(character)
			if state.noWalkAnimationsOn then
				disableWalkAnimations(character)
			end
		end))
	end

		local function startEspDraw()
			stopEspDraw()

			for _, player in ipairs(Players:GetPlayers()) do
				createEspForPlayer(player)
			end

			state.espDrawConn = RunService.RenderStepped:Connect(function()
				if not state.espOn then
					for _, espDrawing in pairs(state.espDrawings) do
						hideEspDrawing(espDrawing)
					end

					return
				end

				local Character = LocalPlayer.Character
				local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")

				if not HumanoidRootPart or not workspace.CurrentCamera then
					for _, espDrawing in pairs(state.espDrawings) do
						hideEspDrawing(espDrawing)
					end

					return
				end

				for k, espDrawing in pairs(state.espDrawings) do

						local Character2 = k.Character
						local Humanoid = Character2 and Character2:FindFirstChildOfClass("Humanoid")
						local HumanoidRootPart2 = Character2 and Character2:FindFirstChild("HumanoidRootPart")

						if Character2 and Humanoid and HumanoidRootPart2 and Humanoid.Health > 0 then
							local screenPosition, onScreen = workspace.CurrentCamera:WorldToViewportPoint(HumanoidRootPart2.Position)

							if onScreen then
								local Magnitude = (HumanoidRootPart2.Position - HumanoidRootPart.Position).Magnitude
								local boxHeight = math.abs(workspace.CurrentCamera:WorldToViewportPoint(HumanoidRootPart2.Position + Vector3.new(0, 2.8, 0)).Y - workspace.CurrentCamera:WorldToViewportPoint(HumanoidRootPart2.Position + Vector3.new(0, -3.2, 0)).Y)
								local boxWidth = boxHeight / 1.8
								local boxLeft = screenPosition.X - boxWidth / 2
								local boxTop = screenPosition.Y - boxHeight / 2

								espDrawing.BoxOutline.Size = Vector2.new(boxWidth, boxHeight)
								espDrawing.BoxOutline.Position = Vector2.new(boxLeft, boxTop)
								espDrawing.BoxOutline.Visible = true
								espDrawing.Box.Size = Vector2.new(boxWidth, boxHeight)
								espDrawing.Box.Position = Vector2.new(boxLeft, boxTop)
								espDrawing.Box.Visible = true
								espDrawing.Name.Text = k.DisplayName or k.Name
								espDrawing.Name.Position = Vector2.new(screenPosition.X, boxTop - 15)
								espDrawing.Name.Visible = true

								espDrawing.Kills.Text = tostring(playerKills) .. " Kills"
								espDrawing.Kills.Position = Vector2.new(boxLeft + boxWidth + 5, boxTop)
								espDrawing.Kills.Visible = true
								espDrawing.Dist.Text = tostring(math.floor(Magnitude)) .. " studs"
								espDrawing.Dist.Position = Vector2.new(screenPosition.X, boxTop + boxHeight + 2)
								espDrawing.Dist.Visible = true
							else
								hideEspDrawing(espDrawing)
							end

							hideEspDrawing(espDrawing)
						end

					end
				end
			end)

			if not state.espPlayerAddedConn then
				state.espPlayerAddedConn = Players.PlayerAdded:Connect(function(player)
					createEspForPlayer(player)
				end)
			end

			if not state.espPlayerRemovingConn then
				state.espPlayerRemovingConn = Players.PlayerRemoving:Connect(function(player)
					updateEspForPlayer(player)
				end)

	local applySkyTheme

	do
		local function toColor3(rgb)
			return Color3.fromRGB(rgb[1], rgb[2], rgb[3])
		end

		local savedLighting

		local function saveLighting()
			if savedLighting then
				return
			end

			savedLighting = {
				FogEnd = Lighting.FogEnd,
				FogStart = Lighting.FogStart,
				FogColor = Lighting.FogColor,
				Brightness = Lighting.Brightness,
				ClockTime = Lighting.ClockTime,
				GlobalShadows = Lighting.GlobalShadows,
				OutdoorAmbient = Lighting.OutdoorAmbient,
				Ambient = Lighting.Ambient
			}
		end
		local function restoreLighting()
			if not savedLighting then
				return
			end

			for k, v in pairs(savedLighting) do
				pcall(function()
					Lighting[k] = v
				end)
			end

			savedLighting = nil
		end
		local function clearCustomSky()
			for _, child in ipairs(Lighting:GetChildren()) do
				if child:GetAttribute("_V4MPSky") then
					pcall(function()
						child:Destroy()
					end)
				end
			end

			local Terrain = workspace:FindFirstChildOfClass("Terrain")

			if Terrain then
				for _, child in ipairs(Terrain:GetChildren()) do
					if child:GetAttribute("_V4MPSky") then
						pcall(function()
							child:Destroy()
						end)
					end
				end
			end
		end

		applySkyTheme = function(themeName)
			local skyTheme2 = skyThemes[themeName]

			if not skyTheme2 or skyTheme2.kind == "off" then
				clearCustomSky()
				restoreLighting()
				state.skyTheme = "Off"

				return
			end

			saveLighting()
			clearCustomSky()
			Lighting.FogEnd = 100000
			Lighting.FogStart = 0
			Lighting.FogColor = Color3.fromRGB(200, 200, 200)
			Lighting.GlobalShadows = true
			Lighting.ClockTime = skyTheme2.clock or 14
			Lighting.Brightness = skyTheme2.brightness or 2

			if skyTheme2.outAmb then
				Lighting.OutdoorAmbient = toColor3(skyTheme2.outAmb)
			end

			if skyTheme2.ambient then
				Lighting.Ambient = toColor3(skyTheme2.ambient)
			end

			if skyTheme2.sky then
				local Sky = Instance.new("Sky")

				Sky.Name = "_V4MPSky"
				Sky:SetAttribute("_V4MPSky", true)

				if skyTheme2.sky.stars then
					Sky.StarCount = skyTheme2.sky.stars
				end

				if skyTheme2.sky.moon then
					Sky.MoonAngularSize = skyTheme2.sky.moon
				end

				if skyTheme2.sky.sun then
					Sky.SunAngularSize = skyTheme2.sky.sun
				end

				if skyTheme2.sky.moonTex then
					Sky.MoonTextureId = "rbxasset://sky/moon.jpg"
				end

				Sky.Parent = Lighting
			end

			if skyTheme2.atm then
				local Atmosphere = Instance.new("Atmosphere")

				Atmosphere.Name = "_V4MPAtmosphere"
				Atmosphere:SetAttribute("_V4MPSky", true)
				Atmosphere.Density = skyTheme2.atm.dens or 0.3
				Atmosphere.Color = toColor3(skyTheme2.atm.color)
				Atmosphere.Decay = toColor3(skyTheme2.atm.decay)
				Atmosphere.Glare = skyTheme2.atm.glare or 1
				Atmosphere.Haze = skyTheme2.atm.haze or 1
				Atmosphere.Parent = Lighting
			end

			local Terrain = workspace:FindFirstChildOfClass("Terrain")

			if skyTheme2.clouds and Terrain then
				local Clouds = Instance.new("Clouds")

				Clouds.Name = "_V4MPSkyClouds"
				Clouds:SetAttribute("_V4MPSky", true)
				Clouds.Cover = skyTheme2.clouds.cover or 0.5
				Clouds.Density = skyTheme2.clouds.dens or 0.5
				Clouds.Color = toColor3(skyTheme2.clouds.color)
				Clouds.Parent = Terrain
			end

			state.skyTheme = themeName
		end
	end

	local function findSafeSpot(character, targetPart)
		local raycastParams = RaycastParams.new()

		raycastParams.FilterType = Enum.RaycastFilterType.Exclude
		raycastParams.FilterDescendantsInstances = {
			character,
			state.antiBombFolder
		}
		raycastParams.IgnoreWater = true

		for i = 1, 15 do
			local randomAngle = math.rad(math.random(0, 360))
			local randomDistance = math.random(8, 16)
			local candidatePosition = targetPart.Position + Vector3.new(math.cos(randomAngle) * randomDistance, 0, math.sin(randomAngle) * randomDistance)
			local originPosition = targetPart.Position + Vector3.new(0, 1, 0)
			local rayDirection = candidatePosition + Vector3.new(0, 1, 0) - originPosition

			if not workspace:Raycast(originPosition, rayDirection, raycastParams) then
				return CFrame.new(candidatePosition.X, targetPart.Position.Y, candidatePosition.Z)
			end
		end

		return targetPart.CFrame * CFrame.new(0, 8, 0)
	end
	local function startAntiBomb()
		stopAntiBomb()

		if not getRootPart(LocalPlayer) then
			return
		end

		local Folder = Instance.new("Folder")

		Folder.Name = "GS_AntiBombAura"

		Part.Transparency = 1
		Part.Anchored = false
		Part.CanCollide = false
		Part.Parent = Folder

		local antiBombMarkers = {}
		local markerCount = 6.2831853071795862 * state.antiBombRadius / 48

		for i = 1, 48 do
			local markerAngle = i / 48 * math.pi * 2
			local Part2 = Instance.new("Part")

			Part2.Name = "RingSegment"
			Part2.Size = Vector3.new(0.15, 0.05, markerCount + 0.05)

			Part2.Color = Color3.fromRGB(255, 255, 255)
			Part2.CanCollide = false
			Part2.CastShadow = false
			Part2.Parent = Folder
			Part2.CFrame = Part.CFrame * CFrame.new(math.cos(markerAngle) * state.antiBombRadius, 0, math.sin(markerAngle) * state.antiBombRadius) * CFrame.Angles(0, -markerAngle, 0)

			local WeldConstraint = Instance.new("WeldConstraint")

			WeldConstraint.Part0 = Part
			WeldConstraint.Part1 = Part2
			WeldConstraint.Parent = Part2
			table.insert(antiBombMarkers, {
				Part = Part2,
				Index = i
			})
		end

		Part.Anchored = true
		state.antiBombConn = RunService.PostSimulation:Connect(function()
			if not state.antiBombOn then
				return
			end

			if not localRoot or not localRoot.Parent then
				stopAntiBomb()

				return
			end

			Part.CFrame = localRoot.CFrame * CFrame.new(0, -2.8, 0)

			for i, v in ipairs(antiBombMarkers) do
				local spinAngle = i / 48 * math.pi * 2
				local spinX = math.cos(spinAngle) * state.antiBombRadius
				local spinZ = math.sin(spinAngle) * state.antiBombRadius

				v.Part.Transparency = 0
				v.Part.CFrame = Part.CFrame * CFrame.new(spinX, 0, spinZ) * CFrame.Angles(0, -spinAngle, 0)

				local pulseDistance = math.abs(i - tick() * 0.7 * 48 / 2 % 48)

				if pulseDistance > 24 then
					pulseDistance = 48 - pulseDistance
				end

				if pulseDistance < 14 then
					v.Part.Color = Color3.fromRGB(255, 255, 255):Lerp(Color3.fromRGB(180, 0, 255), 1 - pulseDistance / 14)
				else
					v.Part.Color = Color3.fromRGB(255, 255, 255)
				end
			end

			local Parent = localRoot.Parent

			if Parent and Parent:FindFirstChildOfClass("Tool") then
				return
			end

			for _, player in ipairs(Players:GetPlayers()) do
				if player ~= LocalPlayer and isAlive(player) then
					local playerRoot = getRootPart(player)
					local Character = player.Character
					local Tool = Character and Character:FindFirstChildOfClass("Tool")
					local Backpack = player:FindFirstChildOfClass("Backpack")
					local Tool2 = Tool or Backpack and Backpack:FindFirstChildOfClass("Tool")

					if playerRoot and Tool2 and isInSameArena(player) and (not state.teamCheckOn or not hasHighlight(Character)) and (Vector3.new(localRoot.Position.X, 0, localRoot.Position.Z) - Vector3.new(playerRoot.Position.X, 0, playerRoot.Position.Z)).Magnitude <= state.antiBombRadius then
						localRoot.CFrame = findSafeSpot(localRoot.Parent, localRoot)

						break
					end
				end
			end
		end)
	end

		sanitizeConfigName = function(text)
			return string.sub(string.gsub(string.gsub(string.gsub(string.gsub(string.gsub(tostring(text or ""), "^%s+", ""), "%s+$", ""), "%.json$", ""), "[<>:\"/|%?%*]", "_"), "\\", "_"), 1, 64)

		local function listConfigs()
			local configNames = {}

			ensureConfigFolder()

			if type(listfiles) ~= "function" then
				return configNames
			end

			local ok, result = pcall(listfiles, "gscripts_configs")

			if not ok or type(result) ~= "table" then
				return configNames
			end

			for _, v in ipairs(result) do
				local fileName = string.match(tostring(v), "[^\\/]+$")

				if fileName and string.match(fileName, "%.json$") then
					local configBaseName = string.gsub(fileName, "%.json$", "")

					if configBaseName ~= "" then
						table.insert(configNames, configBaseName)
					end
				end
			end

			table.sort(configNames)

			return configNames
		end

		local function saveConfig(name)
			ensureConfigFolder()

			local sanitizedName = sanitizeConfigName(name)

			if sanitizedName == "" then
				return false
			end

			if type(writefile) ~= "function" then
				return false
			end

			local ok, result = pcall(function()
				return game:GetService("HttpService"):JSONEncode(collectConfig())
			end)

			if not ok then
				return false
			end

			return (pcall(writefile, "gscripts_configs" .. "/" .. sanitizedName .. ".json", result))
		end
		local function loadConfig(name)
			ensureConfigFolder()

			local sanitizedLoadName = sanitizeConfigName(name)

			if sanitizedLoadName == "" then
				return false
			end

			if type(readfile) ~= "function" then
				return false
			end

			local ok, result = pcall(readfile, "gscripts_configs" .. "/" .. sanitizedLoadName .. ".json")

			if not ok then
				return false
			end

			local ok2, result2 = pcall(function()
				return game:GetService("HttpService"):JSONDecode(result)
			end)

			if ok2 and type(result2) == "table" then
				applyConfig(result2)

				return true
			end

			return false
		end

		setUiValue = function(flag, value)
			if suppressUiCallbacks then
				return
			end

			local control = uiControls[flag]

			if not control then
				return
			end

			suppressUiCallbacks = true
			pcall(function()
				if control.SetValue then
					control:SetValue(value)
				elseif control.Set then
					control:Set(value)
				elseif control.SetState then
					control:SetState(value)
				end
			end)
			suppressUiCallbacks = false
		end

		local function setUiValueSafe(control, value)
			if not control then
				return
			end

			suppressUiCallbacks = true
			pcall(function()
				if control.SetValue then
					control:SetValue(value)
				elseif control.Set then
					control:Set(value)
				elseif control.SetState then
					control:SetState(value)
				end
			end)
			suppressUiCallbacks = false
		end

		addToggle = function(section, label, flag, default, callback)
			local toggleControl = section:AddToggle({
				Text = label,
				Flag = flag,
				Default = default or false,
				Callback = function(enabled)
					if callback then
						callback(enabled)
					end

					if not suppressUiCallbacks and flag ~= "settings_ui_sounds" then
						if enabled then
							playSound(soundSettings.EnableSound)
						else
							playSound(soundSettings.DisableSound)
						end
					end
				end
			})

			uiControls[flag] = toggleControl

			return toggleControl
		end
		addSlider = function(section, label, flag, min, max, default, suffix, decimals, callback)
			local sliderControl = section:AddSlider({
				Text = label,
				Flag = flag,
				Min = min,
				Max = max,
				Default = default,
				Suffix = suffix or "",
				Decimal = decimals or 1,
				Callback = callback or function()
				end
			})

			uiControls[flag] = sliderControl

			return sliderControl
		end
		addDropdown = function(section, label, flag, options, default, callback)
			local dropdownLabel = section
			local AddDropdown = section.AddDropdown
			local dropdownOptions = {
				Text = label,
				Flag = flag,
				Options = options,
				Default = default
			}
			local dropdownCallback = callback
			local dropdownControl

			if dropdownCallback then
				dropdownOptions.Callback = dropdownCallback
				dropdownControl = AddDropdown(dropdownLabel, dropdownOptions)
				uiControls[flag] = dropdownControl
			else
				dropdownOptions.Callback = function()
				end
				dropdownControl = AddDropdown(dropdownLabel, dropdownOptions)
				uiControls[flag] = dropdownControl
			end

			return dropdownControl
		end

					end

					end

		})

	local function copyToClipboard(text)
		local setclipboard_ = setclipboard or toclipboard

		if setclipboard_ then
			if pcall(function()
				setclipboard_(text)
			end) then
				return true
			end
		end

		if syn and syn.write_clipboard and pcall(function()
			syn.write_clipboard(text)
		end) then
			return true
		end

		return false
	end
	local function getExecutorName()
		local ok, result = pcall(function()

		end

			return
		end
