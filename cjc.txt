    local function T(key)
        return Translations[_G.SelectedLanguage][key] or key
    end

    local function TF(key, vars)
        local s = T(key)
        if vars then
            for k, v in pairs(vars) do
                s = s:gsub("{"..k.."}", function() return tostring(v) end)
            end
        end
        return s
    end

        -- ── Callback value safety (WindUI may pass value or table) ──
        local function pick(v)
            if type(v)=="table" then return v.Value or v[1] or v end
            return v
        end
        local function numVal(v)
            if type(v)=="table" then v = v.Value or v[1] or 0 end
            return tonumber(v) or 0
        end

        -- ══════════════════════════════════════════════════════════
        --  TURN BOOST
        -- ══════════════════════════════════════════════════════════
        RunService.Heartbeat:Connect(function(delta)
            if not active or not P.s.turnBoost then return end
            if P.s.assistEnabled and P.s.assistHasTarget then return end
            local c = P.lp.Character
            local hrp = c and c:FindFirstChild("HumanoidRootPart")
            local hum = c and c:FindFirstChild("Humanoid")
            if hrp and hum and hum.MoveDirection.Magnitude > 0.1 then
                local factor = P.vals.turnBoostSpeeds[P.s.turnBoostSpeedIndex]
                local lerp = hrp.CFrame.LookVector:Lerp(hum.MoveDirection, factor * delta)
                hrp.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + lerp)
            end
        end)

        -- ══════════════════════════════════════════════════════════
        --  SPEED SYSTEM
        -- ══════════════════════════════════════════════════════════
        do
            local spConn = nil
            local function valToVel(v) return 16 + ((v-1)/9)*19 end
            local function valToCF(v)  return 0.025 + (v-1)*0.025 end

            local function setupVel(char)
                local hum  = char:WaitForChild("Humanoid", 5)
                local root = char:WaitForChild("HumanoidRootPart", 5)
                if not hum or not root then return end
                if spConn then spConn:Disconnect() spConn = nil end
                spConn = RunService.RenderStepped:Connect(function()
                    if not active or not P.s.speedActive or not root or not root.Parent then return end
                    local mDir = hum.MoveDirection
                    if mDir.Magnitude > 0.01 then
                        local dSpeed = valToVel(math.clamp(P.s.speedLevel,1,10))
                        if P.s.speedLevel >= 7 then dSpeed = dSpeed*1.2
                        elseif P.s.speedLevel >= 5 then dSpeed = dSpeed*1.1 end
                        local hDir = Vector3.new(mDir.X, 0, mDir.Z)
                        if hDir.Magnitude > 0 then
                            local nVel = hDir.Unit * dSpeed
                            root.Velocity = Vector3.new(nVel.X, root.Velocity.Y, nVel.Z)
                        end
                    else
                        local v = root.Velocity
                        root.Velocity = Vector3.new(v.X*0.8, v.Y, v.Z*0.8)
                    end
                end)
            end

            local function setupCF(char)
                local hum  = char:WaitForChild("Humanoid", 5)
                local root = char:WaitForChild("HumanoidRootPart", 5)
                if not hum or not root then return end
                if spConn then spConn:Disconnect() spConn = nil end
                spConn = RunService.RenderStepped:Connect(function()
                    if not active or not P.s.speedActive or not root or not root.Parent then return end
                    local mDir = hum.MoveDirection
                    if mDir.Magnitude > 0.01 then
                        root.CFrame = root.CFrame + mDir * valToCF(math.clamp(P.s.speedLevel,1,10))
                    end
                end)
            end

            P.fn.speedSetup = function()
                local char = P.lp.Character
                if not char then return end
                if P.s.speedType == "Speed 1" then setupVel(char) else setupCF(char) end
            end
            P.fn.speedStop = function()
                if spConn then spConn:Disconnect() spConn = nil end
            end
            P.lp.CharacterAdded:Connect(function(char)
                if not active then return end
                task.wait(1)
                if P.s.speedActive then P.fn.speedSetup() end
            end)

        -- ══════════════════════════════════════════════════════════
        --  WALL SLIDE
        -- ══════════════════════════════════════════════════════════
        do
            local wsChar, wsHum, wsHRP, wsAtt, wsVel, wsCharConn, wsLoopConn = nil,nil,nil,nil,nil,nil,nil
            local rayParams = RaycastParams.new()
            pcall(function() rayParams.FilterType = Enum.RaycastFilterType.Exclude end)
            rayParams.IgnoreWater = true

            local function flatUnit(v)
                local f = Vector3.new(v.X,0,v.Z)
                return f.Magnitude > 0.001 and f.Unit or Vector3.new(0,0,0)
            end
            local function safeUnit(v, fb) return v.Magnitude > 0.001 and v.Unit or fb end

            local function destroySlideVel()
                if wsVel then pcall(function() wsVel:Destroy() end) wsVel = nil end
            end
            local function restoreWS()
                destroySlideVel()
                if wsHum and wsHum.Parent then wsHum.AutoRotate = true end
            end
            local function createSlideVel()
                if not wsHRP or not wsAtt then return nil end
                local ok, vel = pcall(function()
                    local lv = Instance.new("LinearVelocity")
                    lv.Name = "SuiteSlideVel" lv.Attachment0 = wsAtt
                    lv.RelativeTo = Enum.ActuatorRelativeTo.World
                    lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
                    lv.MaxForce = 100000
                    pcall(function()
                        lv.ForceLimitsEnabled = true lv.ForceLimitMode = Enum.ForceLimitMode.PerAxis
                        lv.MaxAxesForce = Vector3.new(100000,0,100000)
                    end)
                    lv.Parent = wsHRP
                    return lv
                end)
                if ok and vel then return vel end
                local ok2, bv = pcall(function()
                    local b = Instance.new("BodyVelocity")
                    b.Name = "SuiteSlideVel" b.MaxForce = Vector3.new(100000,0,100000) b.P = 1250 b.Parent = wsHRP
                    return b
                end)
                return ok2 and bv or nil
            end
            local function getWallNormal(moveDir)
                if not wsHRP then return nil end
                local normals = {}
                local side = Vector3.new(-moveDir.Z,0,moveDir.X)
                local pos = wsHRP.Position
                local origins = {pos, pos+side*0.25, pos-side*0.25}
                local dirs = {moveDir, safeUnit(moveDir+side*0.25,moveDir), safeUnit(moveDir-side*0.25,moveDir)}
                for i=1,3 do
                    local res = workspace:Raycast(origins[i], dirs[i]*1.35, rayParams)
                    if res and res.Instance and res.Instance:IsA("BasePart") and res.Instance.CanCollide and math.abs(res.Normal.Y)<=0.3 then
                        local n = flatUnit(res.Normal)
                        if n.Magnitude > 0.001 and moveDir:Dot(n) < -0.01 then table.insert(normals,n) end
                    end
                end
                if #normals == 0 then return nil end
                local sum = Vector3.new(0,0,0)
                for _,n in ipairs(normals) do sum = sum+n end
                return safeUnit(sum, normals[1])
            end
            local function onChar(char)
                restoreWS()
                if wsAtt then pcall(function() wsAtt:Destroy() end) wsAtt = nil end
                wsChar = char wsHum = char:WaitForChild("Humanoid",5) wsHRP = char:WaitForChild("HumanoidRootPart",5)
                if not wsHum or not wsHRP then return end
                for _,n in ipairs({"SuiteSlideVel","SuiteSlideAtt"}) do
                    local old = wsHRP:FindFirstChild(n) if old then pcall(function() old:Destroy() end) end
                end
                wsAtt = Instance.new("Attachment") wsAtt.Name = "SuiteSlideAtt" wsAtt.Parent = wsHRP
            end
            local function step()
                if not active or not P.s.wallSlideEnabled then return end
                if not wsChar or not wsChar.Parent or not wsHum or wsHum.Health<=0 or not wsHRP or not wsHRP.Parent then restoreWS() return end
                local chars = {} for _,p in ipairs(Players:GetPlayers()) do if p.Character then table.insert(chars,p.Character) end end
                rayParams.FilterDescendantsInstances = chars
                local moveDir = flatUnit(wsHum.MoveDirection)
                if moveDir.Magnitude < 0.08 then restoreWS() return end
                local normal = getWallNormal(moveDir)
                if not normal then restoreWS() return end
                local vel = wsHRP.Velocity
                local yT = wsHum.FloorMaterial ~= Enum.Material.Air and 0 or math.min(vel.Y,1.5)
                if math.abs(yT-vel.Y)>0.01 then wsHRP.Velocity = Vector3.new(vel.X,yT,vel.Z) end
                if not wsVel or not wsVel.Parent then wsVel = createSlideVel() end
                if not wsVel then return end
                local slideDir = flatUnit(moveDir-normal*moveDir:Dot(normal))
                local finalDir = slideDir.Magnitude>0.05 and slideDir or safeUnit(Vector3.new(-normal.Z,0,normal.X),Vector3.new(1,0,0))
                local target = finalDir*wsHum.WalkSpeed + normal*0.01
                local vv = Vector3.new(target.X,0,target.Z)
                if wsVel:IsA("LinearVelocity") then wsVel.VectorVelocity = vv
                elseif wsVel:IsA("BodyVelocity") then wsVel.Velocity = vv end
                wsHum.AutoRotate = true
            end
            P.fn.stopWallSlide = function()
                P.s.wallSlideEnabled = false
                if wsCharConn then pcall(function() wsCharConn:Disconnect() end) wsCharConn = nil end
                if wsLoopConn then pcall(function() wsLoopConn:Disconnect() end) wsLoopConn = nil end
                restoreWS()
                if wsAtt then pcall(function() wsAtt:Destroy() end) wsAtt = nil end
                wsChar,wsHum,wsHRP = nil,nil,nil
            end
            P.fn.startWallSlide = function()
                P.s.wallSlideEnabled = true
                if P.lp.Character then onChar(P.lp.Character) end
                wsCharConn = P.lp.CharacterAdded:Connect(function(char)
                    pcall(function() task.wait(0.15) if active and P.s.wallSlideEnabled then onChar(char) end end)
                end)
                wsLoopConn = RunService.Heartbeat:Connect(function()
                    if not active then return end
                    local ok,err = pcall(step)
                    if not ok then P.fn.stopWallSlide() warn("[WallSlide] stopped: "..tostring(err)) end
                end)
            end
        end

        -- ══════════════════════════════════════════════════════════
        --  ENEMY DETECTION (used by Assist, GrabBomb, AutoHold)
        -- ══════════════════════════════════════════════════════════
        do
            local cache = {enemies={}, arena=nil, mySide=nil, time=0}
            local function detectEnemies()
                local now = tick()
                if now-cache.time < 0.1 then return cache.enemies, cache.arena, cache.mySide end
                local enemies,myTeam,myArena = {},nil,nil
                if workspace:FindFirstChild("Arenas") then
                    for aName, arena in pairs(workspace.Arenas:GetChildren()) do
                        if aName=="Arena5" or aName=="Arena5ICED" then continue end
                        if arena:FindFirstChild("Slots") then
                            for _,side in pairs({"Left","Right"}) do
                                if not myArena and arena.Slots:FindFirstChild(side) then
                                    for i=1,4 do
                                        local slot = arena.Slots[side]:FindFirstChild(tostring(i))
                                        if slot and slot:FindFirstChild("Data") and slot.Data:FindFirstChild("Player") then
                                            if slot.Data.Player.Value == P.lp then myTeam=side myArena=arena break end
                                        end
                                    end
                                end
                            end
                            if myTeam and myArena then
                                local et = (myTeam=="Left") and "Right" or "Left"
                                if myArena.Slots:FindFirstChild(et) then
                                    for i=1,4 do
                                        local slot = myArena.Slots[et]:FindFirstChild(tostring(i))
                                        if slot and slot:FindFirstChild("Data") and slot.Data:FindFirstChild("Player") then
                                            local e = slot.Data.Player.Value
                                            if e and e ~= P.lp then table.insert(enemies,e) end
                                        end
                                    end
                                end
                                break
                            end
                        end
                    end
                end
                cache.enemies,cache.arena,cache.mySide,cache.time = enemies,myArena,myTeam,now
                return enemies,myArena,myTeam
            end
            P.fn.detectEnemies = detectEnemies

            P.fn.isPlayerAlive = function(plr)
                return plr and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0
            end
            P.fn.isPlayerOnFloor = function(plr, arena)
                if not plr or not arena then return false end
                if arena:FindFirstChild("Slots") then
                    for _,side in pairs({"Left","Right"}) do
                        if arena.Slots:FindFirstChild(side) then
                            for i=1,4 do
                                local s = arena.Slots[side]:FindFirstChild(tostring(i))
                                if s and s:FindFirstChild("Data") and s.Data:FindFirstChild("Player") then
                                    if s.Data.Player.Value == plr then
                                        if s:FindFirstChild("Pad") then
                                            local pad = s.Pad local c = pad.Color
                                            return not (math.abs(c.R-1)<0.01 and math.abs(c.G-89/255)<0.01 and math.abs(c.B-89/255)<0.01)
                                        end
                                        return true
                                    end
                                end
                            end
                        end
                    end
                end
                return false
            end
        end

        -- ══════════════════════════════════════════════════════════
        --  MOVEMENT ASSIST + REACTION TIME
        -- ══════════════════════════════════════════════════════════
        do
            local function hasTargetLOS(hrp, character, targetHRP)
                local params = RaycastParams.new()
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = {P.lp.Character}
                params.IgnoreWater = true
                local origin = hrp.Position + Vector3.new(0,1.5,0)
                local points = {targetHRP.Position, targetHRP.Position+Vector3.new(0,1.5,0), targetHRP.Position+Vector3.new(0,-1.5,0)}
                local head = character:FindFirstChild("Head")
                if head then table.insert(points, head.Position) end
                for _,point in ipairs(points) do
                    local res = workspace:Raycast(origin, point-origin, params)
                    if res and res.Instance and res.Instance:IsDescendantOf(character) then return true end
                end
                return false
            end
            local function getAssistTarget(hrp, maxDist)
                local enemies,arena = P.fn.detectEnemies()
                local bestDist,best = maxDist,nil
                for _,plr in ipairs(enemies) do
                    if plr.Character then
                        local char = plr.Character
                        local tHRP = char:FindFirstChild("HumanoidRootPart")
                        if tHRP and P.fn.isPlayerAlive(plr) and P.fn.isPlayerOnFloor(plr,arena) then
                            local dist = (tHRP.Position-hrp.Position).Magnitude
                            if dist < bestDist and hasTargetLOS(hrp,char,tHRP) then bestDist=dist best=plr end
                        end
                    end
                end
                return best, bestDist
            end
            local function getPredictedPos(targetHrp, myHrp)
                local targetPos = targetHrp.Position
                local vel = targetHrp.AssemblyLinearVelocity
                vel = Vector3.new(vel.X,0,vel.Z)
                local speed = vel.Magnitude
                if speed > 50 then vel = vel.Unit*50 elseif speed < 1.5 then return targetPos end
                local dist = (targetPos-myHrp.Position).Magnitude
                local predTime = math.clamp(dist/42, 0.12, 0.42)
                local predictedPos = targetPos + vel*predTime
                local params = RaycastParams.new()
                params.FilterDescendantsInstances = {targetHrp.Parent, P.lp.Character}
                params.FilterType = Enum.RaycastFilterType.Exclude
                local res = workspace:Raycast(targetPos, predictedPos-targetPos, params)
                if res then return res.Position-(vel.Unit*1.2) end
                return predictedPos
            end
            local fastTouchBusy = false
            local function fastTouch(targetChar, tool)
                if fastTouchBusy then return end
                if not targetChar or not tool then return end
                local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildOfClass("Part")
                if not handle then return end
                local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
                if not targetHRP then return end
                local myHRP = P.lp.Character and P.lp.Character:FindFirstChild("HumanoidRootPart")
                if not myHRP then return end
                if (myHRP.Position-targetHRP.Position).Magnitude < 4.5 then
                    fastTouchBusy = true
                    task.spawn(function()
                        for _=1,15 do
                            pcall(function()
                                firetouchinterest(targetHRP, handle, 0)
                                firetouchinterest(targetHRP, handle, 1)
                            end)
                        end
                        fastTouchBusy = false
                    end)
                end
            end

            RunService.RenderStepped:Connect(function(delta)
                if not active then return end
                if not P.s.assistEnabled then
                    P.s.assistHasTarget = false P.s.reactionTargetId = nil P.s.reactionStartTime = 0 return
                end
                if P.s.grabBombActive or P.s.autoHoldActive then return end
                local character = P.lp.Character
                if not character then P.s.assistHasTarget = false return end
                local tool = character:FindFirstChildOfClass("Tool")
                if not tool then
                    P.s.assistHasTarget = false P.s.reactionTargetId = nil
                    if P.ui.assistStatusBtn then P.ui.assistStatusBtn:SetTitle(TF("assistWaitingTool")) end
                    return
                end
                local hum = character:FindFirstChildOfClass("Humanoid")
                local hrp = character:FindFirstChild("HumanoidRootPart")
                if not hum or not hrp or hum.Health <= 0 then P.s.assistHasTarget = false P.s.reactionTargetId = nil return end
                local target, dist = getAssistTarget(hrp, P.vals.rangeValues[P.s.rangeIndex])
                if not target or not target.Character then
                    P.s.assistHasTarget = false P.s.reactionTargetId = nil P.s.reactionStartTime = 0
                    if P.ui.assistStatusBtn then P.ui.assistStatusBtn:SetTitle(TF("assistNoTarget",{range=P.vals.rangeValues[P.s.rangeIndex]})) end
                    return
                end
                local targetHRP = target.Character:FindFirstChild("HumanoidRootPart")
                if not targetHRP then P.s.assistHasTarget = false P.s.reactionTargetId = nil return end
                local targetId = target.UserId
                if P.s.reactionTargetId ~= targetId then
                    P.s.reactionTargetId = targetId P.s.reactionStartTime = tick()
                end
                P.s.assistHasTarget = true
                if P.s.reactionTime > 0 then
                    local elapsed = tick()-P.s.reactionStartTime
                    if elapsed < P.s.reactionTime then
                        if P.ui.assistStatusBtn then P.ui.assistStatusBtn:SetTitle(TF("assistWaiting",{t=string.format("%.1f",P.s.reactionTime-elapsed),name=target.Name})) end
                        return
                    end
                end
                if P.ui.assistStatusBtn then P.ui.assistStatusBtn:SetTitle(TF("assistTarget",{name=target.Name,d=math.floor(dist+0.5)})) end
                local predictedPos = getPredictedPos(targetHRP, hrp)
                if hum.MoveDirection.Magnitude > 0 then
                    local diff = predictedPos-hrp.Position
                    local flat = Vector3.new(diff.X,0,diff.Z)
                    if flat.Magnitude > 0.1 then
                        local power = P.vals.powerValues[P.s.powerIndex]
                        local blend = hum.MoveDirection.Unit*(1-power) + flat.Unit*power
                        if blend.Magnitude > 0 then hum:Move(blend.Unit, false) end
                    end
                end
                local function smoothLookAt(h, targetPos, d)
                    if P.UIS.MouseBehavior == Enum.MouseBehavior.LockCenter then return end
                    local flatTarget = Vector3.new(targetPos.X, h.Position.Y, targetPos.Z)
                    local dir = flatTarget-h.Position
                    if dir.Magnitude < 0.5 then return end
                    h.CFrame = h.CFrame:Lerp(CFrame.lookAt(h.Position,flatTarget), math.clamp(d*10,0,1))
                end
                smoothLookAt(hrp, predictedPos, delta)
                if dist < 4.5 then fastTouch(target.Character, tool) end
            end)
        end

            local function startShake()
                if P.s.shakeActive then return end
                P.s.shakeActive = true
                if P.ui.shakeStatusBtn then P.ui.shakeStatusBtn:SetTitle(TF("shakeActiveSt")) end
                shakeConn = RunService.RenderStepped:Connect(function(delta)
                    if not active or not hasTool() then stopShake() return end
                    local c = P.lp.Character local hrp = c and c:FindFirstChild("HumanoidRootPart")
                    if not hrp or P.UIS.MouseBehavior ~= Enum.MouseBehavior.LockCenter then return end
                    local intensity = P.vals.shakeSpeedValues[P.s.shakeSpeedIndex]
                    P.s.shakeTime = P.s.shakeTime + delta*2*intensity
                    P.s.shakeIndex = math.floor(P.s.shakeTime*60)+1
                    if P.s.shakeIndex > #shakeOffsets then P.s.shakeIndex=1 P.s.shakeTime=0 end
                    local pos = hrp.CFrame.Position
                    workspace.CurrentCamera.CFrame = CFrame.new(pos)*CFrame.Angles(0,shakeOffsets[P.s.shakeIndex]*3,0)*CFrame.new(-pos)*workspace.CurrentCamera.CFrame
                end)
            end

            RunService.RenderStepped:Connect(function()
                if not active then return end
                if not P.s.shakeEnabled then
                    if P.s.shakeActive then stopShake() end
                    P.vals.recentChanges = {} P.s.lastDir = 0
                    if P.ui.shakeStatusBtn then P.ui.shakeStatusBtn:SetTitle(TF("shakeDisabled")) end
                    local cam0 = workspace.CurrentCamera
                    if cam0 then local _,y0 = cam0.CFrame:ToEulerAnglesYXZ() lastCamYaw = y0 end
                    return
                end
                local cam = workspace.CurrentCamera if not cam then return end
                local _,yaw = cam.CFrame:ToEulerAnglesYXZ()
                if not hasTool() then
                    if P.s.shakeActive then stopShake() end
                    P.vals.recentChanges = {} P.s.lastDir = 0
                    if P.ui.shakeStatusBtn then P.ui.shakeStatusBtn:SetTitle(TF("shakeWaitingTool")) end
                    lastCamYaw = yaw return
                end
                if lastCamYaw == nil then lastCamYaw = yaw return end
                local diff = yaw-lastCamYaw
                if diff > math.pi then diff=diff-2*math.pi end if diff < -math.pi then diff=diff+2*math.pi end
                if math.abs(diff) > shakeThreshold then
                    P.s.lastChangeTime = tick()
                    if P.s.lastDir ~= 0 and diff*P.s.lastDir < 0 then table.insert(P.vals.recentChanges, tick()) end
                    P.s.lastDir = diff
                    local now = tick() local filtered = {} local count = 0
                    for _,t in ipairs(P.vals.recentChanges) do if now-t<=shakeWindow then count=count+1 table.insert(filtered,t) end end
                    P.vals.recentChanges = filtered
                    if P.ui.shakeStatusBtn then P.ui.shakeStatusBtn:SetTitle(TF("shakeSwipes",{n=count})) end
                    if count >= 3 and not P.s.shakeActive then startShake() end
                elseif P.s.shakeActive and tick()-P.s.lastChangeTime >= 0.5 then
                    stopShake() P.vals.recentChanges = {}
                end
                lastCamYaw = yaw
            end)
        end

                if not tool then return nil end
                for _,obj in ipairs(tool:GetDescendants()) do
                    if obj:IsA("TextLabel") or obj:IsA("TextBox") then
                        local n = tonumber(string.match(obj.Text or "","%d+%.?%d*")) if n then return n end

                end
                for _,obj in ipairs(tool:GetDescendants()) do
                    if (obj:IsA("NumberValue") or obj:IsA("IntValue")) and string.lower(obj.Name):find("time") then return obj.Value end

                if antiLagSaved[inst][prop]==nil then antiLagSaved[inst][prop]=orig end
                pcall(function() inst[prop]=value end)
            end
            local function detachTemp(inst)
                if antiLagDetached[inst] then return end antiLagDetached[inst]=inst.Parent pcall(function() inst.Parent=nil end)
            end
            local function process(v)
                if v.Name and v.Name:sub(1,3)=="PX_" then return end
                pcall(function()
                    if v:IsA("Model") then saveAndSet(v,"LevelOfDetail","Disabled") saveAndSet(v,"ModelStreamingMode","Nonatomic")
                    elseif v:IsA("MeshPart") then saveAndSet(v,"CastShadow",false) saveAndSet(v,"DoubleSided",false) saveAndSet(v,"RenderFidelity",Enum.RenderFidelity.Performance)
                    elseif v:IsA("BasePart") then saveAndSet(v,"CastShadow",false) saveAndSet(v,"Material",Enum.Material.Plastic) saveAndSet(v,"Reflectance",0) saveAndSet(v,"MaterialVariant","")
                    elseif v:IsA("Decal") or v:IsA("Texture") then saveAndSet(v,"Transparency",1)
                    elseif v:IsA("SpecialMesh") then saveAndSet(v,"TextureId",0)
                    elseif v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") or v:IsA("SpotLight") then saveAndSet(v,"Enabled",false)
                    elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then saveAndSet(v,"Enabled",false)
                    elseif v:IsA("Attachment") then saveAndSet(v,"Visible",false)
                    elseif v:IsA("Explosion") then saveAndSet(v,"BlastPressure",1) saveAndSet(v,"BlastRadius",1)
                    elseif v:IsA("Sky") then detachTemp(v)
                    elseif v:IsA("Atmosphere") then saveAndSet(v,"Density",0) saveAndSet(v,"Haze",0)
                    elseif v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("SunRaysEffect") or v:IsA("DepthOfFieldEffect") or v:IsA("ColorCorrectionEffect") then saveAndSet(v,"Enabled",false) end
                end)
            end

                P.s.antiLagActive = st
                if st then
                    pcall(function() setfpscap(999999999) end)
                    for _,v in next,workspace:GetDescendants() do process(v) end
                    for _,v in next,Lighting:GetDescendants() do process(v) end
                    antiLagLightingSaved = { GlobalShadows=Lighting.GlobalShadows, FogEnd=Lighting.FogEnd, Brightness=Lighting.Brightness, Ambient=Lighting.Ambient, OutdoorAmbient=Lighting.OutdoorAmbient }
                    Lighting.GlobalShadows=false Lighting.FogEnd=9e9 Lighting.Brightness=0
                    local terrain=workspace:FindFirstChildOfClass("Terrain")
                    if terrain then
                        antiLagTerrainSaved = {WaterReflectance=terrain.WaterReflectance,WaterTransparency=terrain.WaterTransparency,WaterWaveSize=terrain.WaterWaveSize,WaterWaveSpeed=terrain.WaterWaveSpeed}
                        pcall(function() terrain.WaterReflectance=0 terrain.WaterTransparency=0.7 terrain.WaterWaveSize=0 terrain.WaterWaveSpeed=0 end)
                    end
                    local wC=workspace.DescendantAdded:Connect(function(v) if P.s.antiLagActive then process(v) end end)
                    local lC=Lighting.ChildAdded:Connect(function(v) if P.s.antiLagActive then process(v) end end)
                    table.insert(antiLagConns,wC) table.insert(antiLagConns,lC)
                else
                    for _,conn in pairs(antiLagConns) do if conn then conn:Disconnect() end end antiLagConns={}
                    for inst,props in pairs(antiLagSaved) do
                        if typeof(inst)=="Instance" and inst.Parent then for prop,value in pairs(props) do pcall(function() inst[prop]=value end) end end
                    end
                    antiLagSaved={} antiLagSavedCount=0
                    for inst,parent in pairs(antiLagDetached) do if typeof(inst)=="Instance" and parent then pcall(function() inst.Parent=parent end) end end
                    antiLagDetached={}
                    if antiLagLightingSaved then
                        Lighting.GlobalShadows=antiLagLightingSaved.GlobalShadows Lighting.FogEnd=antiLagLightingSaved.FogEnd
                        Lighting.Brightness=antiLagLightingSaved.Brightness Lighting.Ambient=antiLagLightingSaved.Ambient
                        Lighting.OutdoorAmbient=antiLagLightingSaved.OutdoorAmbient antiLagLightingSaved=nil
                    end
                    local terrain=workspace:FindFirstChildOfClass("Terrain")
                    if terrain and antiLagTerrainSaved then
                        pcall(function()
                            terrain.WaterReflectance=antiLagTerrainSaved.WaterReflectance terrain.WaterTransparency=antiLagTerrainSaved.WaterTransparency
                            terrain.WaterWaveSize=antiLagTerrainSaved.WaterWaveSize terrain.WaterWaveSpeed=antiLagTerrainSaved.WaterWaveSpeed
                        end)
                        antiLagTerrainSaved=nil
                    end
                    pcall(function() setfpscap(60) end)
                end
            end

            local hooked,played = setmetatable({},{__mode="k"}), setmetatable({},{__mode="k"})
            local function snapshot(char,cf)
                local clone; pcall(function() local old=char.Archivable char.Archivable=true clone=char:Clone() char.Archivable=old end)
                if not clone then return nil end
                for _,d in ipairs(clone:GetDescendants()) do
                    pcall(function()
                        if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("Humanoid") or d:IsA("Sound") or d:IsA("BillboardGui") or d:IsA("ForceField") or d:IsA("Tool") or d:IsA("Highlight") or d:IsA("ParticleEmitter") or d:IsA("Fire") or d:IsA("Smoke") or d:IsA("Sparkles") then d:Destroy() end
                    end)
                end
                for _,p in ipairs(H.parts(clone)) do p.Anchored=true p.CanCollide=false p.CanTouch=false p.CanQuery=false p.Massless=true end
                if cf then clone:PivotTo(cf) end clone.Parent=Folder Debris:AddItem(clone,12) return clone
            end
            local function hideOriginal(char)
                local function hide()
                    for _,d in ipairs(char:GetDescendants()) do
                        if d:IsA("BasePart") then d.LocalTransparencyModifier=1 d.Transparency=1
                        elseif d:IsA("Decal") or d:IsA("Texture") then d.Transparency=1
                        elseif d:IsA("BillboardGui") then d.Enabled=false end
                    end
                end
                hide() task.spawn(function() local t=os.clock() while os.clock()-t<8 and char.Parent do hide() RunService.Heartbeat:Wait() end end)
            end
            local function playFX(id,char,cf)
                local fn=FX[id] if not fn or not char then return end
                local m=snapshot(char,cf) if not m then return end
                if not cf then hideOriginal(char) end
                local hrp=m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
                if not hrp then m:Destroy() return end local ok=pcall(fn,m,hrp) if not ok then m:Destroy() end
            end
            local function onDeath(char)
                if played[char] then return end played[char]=true
                if P.kfx.current<=1 then return end
                local mR=P.lp.Character and P.lp.Character:FindFirstChild("HumanoidRootPart")
                local rt=char:FindFirstChild("HumanoidRootPart")
                if not (mR and rt) then return end
                if (rt.Position-mR.Position).Magnitude > P.kfx.range then return end
                playFX(P.kfx.list[P.kfx.current].id, char)
            end
            local function watchChar(char)
                if hooked[char] then return end hooked[char]=true
                local hum=char:WaitForChild("Humanoid",5) if not hum then return end
                local fired=false
                local function died() if fired then return end fired=true pcall(onDeath,char) end
                hum.Died:Connect(died) hum.HealthChanged:Connect(function(h) if h<=0 then died() end end)
            end
            local function watchPlayer(p)
                if p==P.lp then return end
                if p.Character then task.spawn(watchChar,p.Character) end
                p.CharacterAdded:Connect(watchChar)
            end
            for _,p in ipairs(Players:GetPlayers()) do watchPlayer(p) end
            Players.PlayerAdded:Connect(watchPlayer)
        end
