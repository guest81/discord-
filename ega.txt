local function travelTo(position, instant)
    local root = getRoot()
    if not root then return false end
    if instant then
        root.CFrame = CFrame.new(position)
        return true
    else
        local tween = TweenService:Create(root, TweenInfo.new(0.5, Enum.EasingStyle.Linear), {CFrame = CFrame.new(position)})
        tween:Play()
        tween.Completed:Wait()
        return true
    end
end

local function recordMutations(eggData)
    local list = {}
    if eggData and eggData.Mutations then
        for _, mut in pairs(eggData.Mutations) do
            if type(mut) == "string" then
                table.insert(list, mut)
            end
        end
    end
    if eggData and eggData.BaseMutation and type(eggData.BaseMutation) == "string" then
        table.insert(list, eggData.BaseMutation)
    end
    return list
end

local function matchesEggFilters(egg, areas, rarities, mutations)
    if areas and #areas > 0 then
        local areaId = egg.AreaId
        if not areaId then return false end
        local allowed = false
        for _, a in ipairs(areas) do
            if tostring(a) == tostring(areaId) then
                allowed = true
                break
            end
        end
        if not allowed then return false end
    end
    if rarities and #rarities > 0 then
        local rarity = resolveRarity(egg.AssetCategory)
        if not rarity then return false end
        local allowed = false
        for _, r in ipairs(rarities) do
            if tostring(r) == tostring(rarity) or tostring(r) == tostring(egg.AssetCategory) then
                allowed = true
                break
            end
        end
        if not allowed then return false end
    end
    if mutations and #mutations > 0 then
        local eggMutations = recordMutations(egg)
        local found = false
        for _, mut in ipairs(eggMutations) do
            for _, selected in ipairs(mutations) do
                if tostring(selected) == tostring(mut) then
                    found = true
                    break
                end
            end
            if found then break end
        end
        if not found then return false end
    end
    return true
end

local function priorityOrder()
    local order = {}
    local added = {}
    for _, key in ipairs({"PrioritySlot1", "PrioritySlot2", "PrioritySlot3", "PrioritySlot4"}) do
        local val = getOption(key)
        if val and not added[val] then
            table.insert(order, val)
            added[val] = true
        end
    end
    for _, name in ipairs(StealPriorityList) do
        if not added[name] then
            table.insert(order, name)
        end
    end
    return order
end

local function sortEggsByPriority(eggs, priorityMode)
    if priorityMode == "Nearest" then
        local root = getRoot()
        if root then
            table.sort(eggs, function(a, b)
                local distA = (a.Position - root.Position).Magnitude
                local distB = (b.Position - root.Position).Magnitude
                return distA < distB
            end)
        end
    elseif priorityMode == "Furthest" then
        local root = getRoot()
        if root then
            table.sort(eggs, function(a, b)
                local distA = (a.Position - root.Position).Magnitude
                local distB = (b.Position - root.Position).Magnitude
                return distA > distB
            end)
        end
    elseif priorityMode == "Biggest Size" then
        table.sort(eggs, function(a, b)
            return (a.AssetScale or 1) > (b.AssetScale or 1)
        end)
    elseif priorityMode == "Rarest" then
        table.sort(eggs, function(a, b)
            local rA = resolveRarity(a.AssetCategory) or 0
            local rB = resolveRarity(b.AssetCategory) or 0
            return rA > rB
        end)
    end
    return eggs
end

local function getAvailableEggs(areas, rarities, mutations, bigEggMinScale)
    local snapshot = Remotes.GetAreaEggSnapshot()
    local valid = {}
    if not snapshot then return valid end
    for _, egg in pairs(snapshot) do
        if not egg.IsCarried then
            if matchesEggFilters(egg, areas, rarities, mutations) then
                local scale = egg.AssetScale or 1
                if bigEggMinScale and bigEggMinScale > 1 then
                    if scale >= bigEggMinScale then
                        table.insert(valid, egg)
                    end
                else
                    table.insert(valid, egg)
                end
            end
        end
    end
    return valid
end

local function runAutoSteal()
    if isUnloaded then return end
    if isCarryingEgg() then return end
    if isInventoryFull() then return end
    local areas = getOption("StealZones")
    local rarities = getOption("StealRarities")
    local mutations = getOption("StealMutations")
    local priority = getOption("StealPriority") or "Nearest"
    local bigEggs = isOn("StealBigEggs")
    local bigScale = getOption("StealBigEggScale") or 1.5
    local eggs = getAvailableEggs(areas, rarities, mutations, bigEggs and bigScale or nil)
    if #eggs == 0 then return false end
    eggs = sortEggsByPriority(eggs, priority)
    local target = eggs[1]
    if target then
        local success = Remotes.RequestCarryAreaEgg(target.Uid)
        if success then
            currentStealCount = currentStealCount + 1
            hopThresholdCounter = hopThresholdCounter + 1
            return true
        end
    end
    return false
end

local function runAutoFusePets(force)
    if isUnloaded then return end
    if isCarryingEgg() then return end
    if not isOn("AutoFusePets") and not force then return false end
    if isFusingNow then return false end
    local rarities = getOption("FuseRarities")
    local mutations = getOption("FuseMutations")
    local targetGroup = getOption("FuseTarget") or "Rarity"
    local keepMutated = isOn("FuseKeepMutated")
    local keepEquipped = isOn("FuseKeepEquipped")
    local maxScale = getOption("FuseMaxScale") or 10
    local keepPerCategory = getOption("FuseKeepPerCategory") or 0
    isFusingNow = true
    local inventory = GameModules.Save.Get().PetInventory
    if not inventory then isFusingNow = false return false end
    local petsByGroup = {}
    for id, pet in pairs(inventory) do
        local groupKey
        if targetGroup == "Rarity" then
            groupKey = pet.Rarity or pet.AssetCategory
        else
            groupKey = pet.AssetName or pet.AssetCategory
        end
        if not petsByGroup[groupKey] then petsByGroup[groupKey] = {} end
        table.insert(petsByGroup[groupKey], {id = id, data = pet})
    end
    local selectedPets = {}
    for group, list in pairs(petsByGroup) do
        table.sort(list, function(a, b)
            local rA = resolveRarity(a.data.AssetCategory) or 0
            local rB = resolveRarity(b.data.AssetCategory) or 0
            return rA < rB
        end)
        local keepCount = 0
        for _, entry in ipairs(list) do
            local pet = entry.data
            local scale = pet.AssetScale or 1
            local isMutated = pet.Mutations and #pet.Mutations > 0
            local isEquipped = pet.Equipped == true
            if keepMutated and isMutated then keepCount = keepCount + 1 end
            if keepEquipped and isEquipped then keepCount = keepCount + 1 end
            if scale > maxScale then keepCount = keepCount + 1 end
        end
        local toKeep = math.max(keepCount, keepPerCategory)
        for i, entry in ipairs(list) do
            if i > toKeep then
                table.insert(selectedPets, entry.id)
            end
        end
    end
    if #selectedPets == 0 then isFusingNow = false return false end
    for _, id in ipairs(selectedPets) do
        if isOn("FuseAutoReveal") then
            Remotes.ConfirmBriefing()
            Remotes.FinishReveal()
        end
        Remotes.LoadPet(id)
        Remotes.StartFuse()
        task.wait(0.5)
        isFusingNow = false
        return true
    end
    isFusingNow = false
    return false
end

local function serverHop(reason)
    if isHopingNow then return end
    isHopingNow = true
    local jobId = game.JobId
    local placeId = game.PlaceId
    TeleportService:TeleportToPlaceInstance(placeId, jobId, LocalPlayer)
    isHopingNow = false
end

local function runServerHop()
    if isUnloaded then return end
    if not isOn(1289) then return false end
    local mode = getOption("HopMode") or "No Matching Eggs"
    local threshold = getOption(921) or 15
    if mode == "No Matching Eggs" then
        local areas = getOption("StealZones")
        local rarities = getOption("StealRarities")
        local mutations = getOption("StealMutations")
        local eggs = getAvailableEggs(areas, rarities, mutations, nil)
        if #eggs == 0 then
            serverHop("No eggs found")
            return true
        end
    elseif mode == "Timed Interval" then
        if os.clock() - sessionStart > threshold * 60 then
            serverHop("Time interval")
            return true
        end
    elseif mode == "After Steal Count" then
        if hopThresholdCounter >= threshold then
            hopThresholdCounter = 0
            serverHop("Steal count")
            return true
        end
    end
    return false
end

local function buildConfigPayload()
    local config = {}
    for _, library in ipairs({Toggles, Options}) do
        for key, obj in pairs(library) do
            if type(obj) == "table" and obj.Type then
                local entry = {idx = key, type = obj.Type}
                if obj.Type == "Toggle" then
                    entry.value = obj.Value == true
                elseif obj.Type == "Dropdown" then
                    entry.value = obj.Value
                    entry.multi = obj.Multi == true
                elseif obj.Type == "Input" then
                    entry.text = tostring(obj.Value or "")
                elseif obj.Type == "ColorPicker" then
                    entry.value = obj.Value:ToHex()
                    entry.transparency = obj.Transparency
                elseif obj.Type == "KeyPicker" then
                    entry.key = obj.Value
                    entry.mode = obj.Mode
                    entry.modifiers = obj.Modifiers
                    entry.toggled = obj.Toggled
                end
                table.insert(config, entry)
            end
        end
    end
    return config
end

local function applyConfigObject(obj)
    local target
    if obj.type == "Toggle" then
        target = Toggles[obj.idx]
        if target and target.Type == "Toggle" then
            target:SetValue(obj.value)
            return true
        end
    elseif obj.type == "Dropdown" then
        target = Options[obj.idx]
        if target and target.Type == "Dropdown" then
            target:SetValue(obj.value)
            return true
        end
    elseif obj.type == "Input" then
        target = Options[obj.idx]
        if target and target.Type == "Input" then
            target:SetValue(obj.text)
            return true
        end
    elseif obj.type == "ColorPicker" then
        target = Options[obj.idx]
        if target and target.Type == "ColorPicker" then
            target:SetValueRGB(Color3.fromHex(obj.value))
            target.Transparency = obj.transparency
            return true
        end
    elseif obj.type == "KeyPicker" then
        target = Options[obj.idx]
        if target and target.Type == "KeyPicker" then
            target:SetValue(obj.key)
            target.Mode = obj.mode
            target.Modifiers = obj.modifiers
            target.Toggled = obj.toggled
            return true
        end
    end
    return false
end

local function runEsp()
    if isUnloaded then return end
    clearAllEsp()
    local distance = getOption("EspDistance") or 2000
    local root = getRoot()
    if not root then return end
    local function createEsp(part, color, text)
        if (part.Position - root.Position).Magnitude > distance then return end
        local billboard = Instance.new("BillboardGui")
        billboard.Size = UDim2.new(0, 200, 0, 50)
        billboard.StudsOffset = Vector3.new(0, 3, 0)
        billboard.AlwaysOnTop = true
        billboard.Parent = part
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = text or "?"
        label.TextColor3 = color or Color3.new(1, 1, 1)
        label.TextStrokeTransparency = 0
        label.TextScaled = true
        label.Font = Enum.Font.GothamBold
        label.Parent = billboard
        table.insert(espObjects, billboard)
    end
    if isOn("EspWorldEggs") then
        local eggs = Remotes.GetAreaEggSnapshot()
        if eggs then
            for _, egg in pairs(eggs) do
                if not egg.IsCarried and egg.Position then
                    local color = Color3.new(1, 0.8, 0.2)
                    local text = getAssetName(egg.AssetCategory)
                    createEsp(egg.Position, color, text)
                end
            end
        end
    end
end

local function sendWebhookEmbed(data, important)
    local url = getOption("WebhookUrl")
    if not url or url == "" then return false end
    local ping = getOption("WebhookPingId")
    if ping and ping ~= "" then
        data.content = "<@" .. ping .. ">"
    end
    local success, err = pcall(function()
        HttpService:PostAsync(url, HttpService:JSONEncode(data))
    end)
    return success
end

local function trackWebhookEvents()
    if not isOn("WebhookEnabled") then return end
    local snapshot = Remotes.GetAreaEggSnapshot()
    if not snapshot then return end
    for _, egg in pairs(snapshot) do
        if egg.IsCarried and not eggHistory[egg.Uid] then
            eggHistory[egg.Uid] = true
            if isOn("WebhookEggSpawns") then
                local rarities = getOption("WebhookRarities")
                if rarities and #rarities > 0 then
                    local r = resolveRarity(egg.AssetCategory)
                    local allowed = false
                    for _, val in ipairs(rarities) do
                        if tostring(val) == tostring(r) or tostring(val) == tostring(egg.AssetCategory) then
                            allowed = true
                            break
                        end
                    end
                    if allowed then
                        sendWebhookEmbed({
                            title = "Egg Stolen",
                            description = string.format("**%s** | Rarity: %s", getAssetName(egg.AssetCategory), tostring(resolveRarity(egg.AssetCategory))),
                            color = 65280,
                            footer = {text = "Ouroboros Hub"},
                            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
                        })
                    end
                end
            end
        end
    end
end

local function main()
    bypassAll()
    buildTrailData()

    Options = {}
    Toggles = {}
    task.spawn(function()
        while not isUnloaded do
            task.wait(0.5)
            local hum = getHumanoid()
            local root = getRoot()
            if hum and root then
                if isOn("WalkSpeedEnabled") then
                    hum.WalkSpeed = getOption("WalkSpeed") or 32
                end
                if isOn("JumpPowerEnabled") then
                    hum.UseJumpPower = true
                    hum.JumpPower = getOption("JumpPower") or 50
                end
                if isOn("NoClip") then
                    for _, part in ipairs(Character:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.CanCollide = false
                        end
                    end
                end
                if isOn("Fly") then
                    hum.PlatformStand = true
                    local vel = Vector3.zero
                    local cam = CurrentCamera
                    if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                        vel = vel + cam.CFrame.LookVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                        vel = vel - cam.CFrame.LookVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                        vel = vel - cam.CFrame.RightVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                        vel = vel + cam.CFrame.RightVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                        vel = vel + Vector3.new(0, 1, 0)
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                        vel = vel - Vector3.new(0, 1, 0)
                    end
                    if vel.Magnitude > 0 then
                        root.CFrame = root.CFrame + vel.Unit * (getOption("FlySpeed") or 60) * 0.016
                    end
                else
                    hum.PlatformStand = false
                end
            end
        end
    end)
    task.spawn(function()
        while not isUnloaded do
            task.wait(0.2)
            if isOn("AutoDropEgg") and isCarryingEgg() then
                runAutoDropEgg()
            end
        end
    end)
    task.spawn(function()
        while not isUnloaded do
            task.wait(0.5)
            if stealingEnabled() and not isCarryingEgg() and not isInventoryFull() then
                local success = runAutoSteal()
                if success then
                    task.wait(getOption("StealSpeed") / 1000)
                end
            end
        end
    end)
    task.spawn(function()

            task.wait(2)
