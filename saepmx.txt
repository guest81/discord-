local function FNum(n)
    n = tonumber(n) or 0
    local u = {{1e12,"T"},{1e9,"B"},{1e6,"M"},{1e3,"K"}}
    for _, x in ipairs(u) do
        if n >= x[1] then
            local v = n/x[1]
            return (v < 10 and string.format("%.2f",v) or string.format("%.1f",v)):gsub("%.?0+$","")..x[2]
        end
    end
    return tostring(math.floor(n))
end
local function fmtTime(s)
    s = math.max(0, math.floor(s))
    local h, m = math.floor(s/3600), math.floor(s/60) % 60
    if h > 0 then return ("%dh %02dm"):format(h, m) end
    if m > 0 then return ("%dm %02ds"):format(m, s % 60) end
    return ("%ds"):format(s)
end

function KH.fixRagdoll(char)
    char = char or LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.Physics, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
        if hum.PlatformStand then hum.PlatformStand = false end
        if hum.Sit then hum.Sit = false end
    end
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("BallSocketConstraint") or d:IsA("HingeConstraint") or d:IsA("NoCollisionConstraint") then
            pcall(function() d:Destroy() end)
        end
    end
    for _, j in ipairs(char:GetDescendants()) do
        if j:IsA("Motor6D") and j.Part0 and j.Part1 then
            j.Enabled = true
            local wn = "RigidJointWeld_" .. j.Name
            if not j.Part1:FindFirstChild(wn) then
                local w = Instance.new("WeldConstraint")
                w.Name = wn; w.Part0 = j.Part0; w.Part1 = j.Part1
                w.Parent = j.Part1
            end
        end
    end
end

function KH.applyPerf(item)
    pcall(function()
        if item:IsA("BasePart") then
            item.Material = Enum.Material.SmoothPlastic; item.Reflectance = 0; item.CastShadow = false
            if item:IsA("MeshPart") then
                item.TextureID = ""
                pcall(function() item.RenderFidelity = Enum.RenderFidelity.Performance end)
                pcall(function() item.CollisionFidelity = Enum.CollisionFidelity.Box end)
            end
        elseif item:IsA("SpecialMesh") then item.TextureId = ""
        elseif item:IsA("Decal") or item:IsA("Texture") or item:IsA("SurfaceAppearance") then item.Transparency = 1
        elseif item:IsA("ParticleEmitter") or item:IsA("Trail") or item:IsA("Smoke")
            or item:IsA("Fire") or item:IsA("Sparkles") or item:IsA("Beam") then item.Enabled = false
        elseif item:IsA("Light") or item:IsA("PointLight") or item:IsA("SpotLight") or item:IsA("SurfaceLight") then
            item.Enabled = false
        end
    end)
end

        if t then pcall(function()
            t.Decoration = false; t.WaveSize = 0; t.WaterWaveSize = 0
            t.WaterWaveSpeed = 0; t.WaterReflectance = 0; t.WaterTransparency = 0

function CRZS.applyClone(char)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local cam = WS.CurrentCamera
    local anim = char:FindFirstChild("Animate")
    local jp, jh, hp, mhp = hum.JumpPower, hum.JumpHeight, hum.Health, hum.MaxHealth
    if anim and anim:IsA("LocalScript") then anim.Disabled = true end
    local an = hum:FindFirstChildOfClass("Animator")
    if an then for _, t in ipairs(an:GetPlayingAnimationTracks()) do t:Stop(0) end end
    hum.Archivable = true
    local nh = hum:Clone()
    for _, c in ipairs(nh:GetChildren()) do if c:IsA("Animator") then c:Destroy() end end
    hum.Name = "_OldHumanoid"; nh.Name = "Humanoid"; nh.Parent = char
    local na = Instance.new("Animator"); na.Parent = nh
    nh.WalkSpeed = CRZS.desired; nh.JumpPower = jp; nh.JumpHeight = jh
    nh.MaxHealth = mhp; nh.Health = math.min(hp, mhp)
    if cam then cam.CameraSubject = nh end
    hum:Destroy()
    if anim and anim:IsA("LocalScript") then
        task.wait(); anim.Disabled = false
        task.defer(function()
            if anim.Parent then anim.Disabled = true; task.wait(); anim.Disabled = false end
        end)
    end
    task.defer(function() if nh.Parent then nh:ChangeState(Enum.HumanoidStateType.Running) end end)
    CRZS.forceWS(nh)
end

local function ReadField()
    if not EggStateModule or not EggStateModule.ReadFieldEggs then return nil end
    local ok, d = pcall(EggStateModule.ReadFieldEggs)
    if not ok or type(d) ~= "table" then return nil end
    return d.Records or d
end
local function GetValue(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if not d then return 0 end
    return math.max(1, math.floor((tonumber(d.EarningRate) or 0) * (tonumber(rec.AssetScale) or 1)))
end
local function GetName(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.DisplayName ~= "" then return d.DisplayName end
    return tostring(rec.AssetCategory or "Egg")
end
local function GetRarity(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.Rarity then return d.Rarity.DisplayName or d.Rarity._id or "?" end
    return "?"
end
local function GetRarityColor(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.Rarity and typeof(d.Rarity.Color) == "Color3" then return d.Rarity.Color end
    return Color3.fromRGB(200, 200, 220)
end
local function GetIcon(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    return (d and tostring(d.Icon or "")) or ""
end

function SAE.Evaluate()
    local s = SAE.State
    local now = os.clock()
    if s.Cache and (now - s.CacheAt) < 3 then return s.Cache end
    local recs = ReadField()
    if not recs then return s.Cache or {} end
    local hrp = getHRP()
    local mp = hrp and hrp.Position or Vector3.new()
    local out = {}
    for _, r in ipairs(recs) do
        if type(r) == "table" and r.Uid and typeof(r.BoundsCFrame) == "CFrame"
            and (r.State == "Slot" or r.State == "Dropped") then
            local p = r.BoundsCFrame.Position
            table.insert(out, {
                Uid = r.Uid, Name = GetName(r), Value = GetValue(r),
                Rarity = GetRarity(r), RarityColor = GetRarityColor(r),
                Icon = GetIcon(r), Position = p, Distance = (p - mp).Magnitude,
                Area = r.AreaId, Record = r,
            })
        end
    end
    table.sort(out, function(a, b) return a.Value > b.Value end)
    s.Cache, s.CacheAt = out, now
    return out
end

function SAE.TweenTo(target, speed, sid)
    local hrp = getHRP()
    if not hrp then return false end
    speed = math.clamp(speed or SAE.State.FlightSpeed, 100, 1000)
    local d = (target - hrp.Position).Magnitude
    if d < 3 then return true end
    local dir = (target - hrp.Position).Unit
    local goal = CFrame.lookAt(target, target + dir)
    local tw = TweenService:Create(hrp, TweenInfo.new(d/speed, Enum.EasingStyle.Linear), {CFrame = goal})
    local hb
    hb = RunService.Heartbeat:Connect(function()
        if sid and sid ~= SAE.State.Session then
            pcall(function() tw:Cancel() end)
            if hb then hb:Disconnect() end
            return
        end
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    tw:Play()
    tw.Completed:Wait()
    if hb then hb:Disconnect() end
    return true
end

function SAE.GetInventory()
    local out = {}
    local c = LP.Character
    local bp = LP:FindFirstChild("Backpack")
    for _, cont in ipairs({c, bp}) do

            for _, t in ipairs(cont:GetChildren()) do
                if t:IsA("Tool") and (t:GetAttribute("ItemType") == "AssetEgg"
                    or t:GetAttribute("UID") ~= nil or t.Name:lower():find("egg")) then
                    table.insert(out, t)
                end
            end
        end
    end
    return out
end

function SAE.PlaceAll()
    local s = SAE.State
    if s.Placing then return 0 end
    s.Placing = true

local stealTask
local function StealLoop()
    local s = SAE.State
    while s.Enabled do
        local sid = s.Session
        task.wait(0.15)
        if not s.Enabled or sid ~= s.Session then break end
        if s.Ragdolled then task.wait(0.3); continue end
        local eggs = SAE.Evaluate()
        local target
        if s.SelectedUid then
            for _, e in ipairs(eggs) do if e.Uid == s.SelectedUid then target = e; break end end
        end
        if not target then
            for _, e in ipairs(eggs) do if not s.Unreachable[e.Uid] then target = e; break end end
        end
        if not target then task.wait(2); continue end
        local hrp = getHRP()
        if not hrp then task.wait(1); continue end
        if s.Treadmilling then
            local d = GetRemote("RF/Treadmill/AskDoff")
            if d then pcall(function() d:InvokeServer() end) end
            s.Treadmilling = false
            task.wait(0.3)
        end
        SAE.TweenTo(target.Position + Vector3.new(0, 1.5, 0), s.FlightSpeed, sid)
        if sid ~= s.Session or not s.Enabled then break end
        local ok = SAE.Carry(target.Uid)
        if not ok then s.Unreachable[target.Uid] = os.clock(); task.wait(0.5); continue end
        task.wait(2.5)
        if sid ~= s.Session or not s.Enabled then break end
        for _ = 1, 6 do
            if sid ~= s.Session then break end
            if SAE.IsCarrying() then break end
            SAE.Carry(target.Uid)
            task.wait(0.25)
        end
        if s.AutoReturn then SAE.TweenTo(SAFE_ZONE_POS, s.FlightSpeed, sid); task.wait(0.6) end
        if s.AutoPlace5 then
            s.BatchCount = s.BatchCount + 1
            if s.BatchCount >= 5 then
                s.BatchCount = 0
                task.spawn(function()
                    SAE.PlaceAll()
                    if s.AutoHatch then SAE.HatchAll() end
                end)
            end
        end
        task.wait(0.4)
    end
end

function SAE.Start(mode)
    local s = SAE.State
    if s.Enabled and s.Mode == mode then return end
    if s.Enabled then s.Session = s.Session + 1; task.wait(0.1) end
    s.Enabled = true; s.Mode = mode; _G.PhantomX_Stealing = true
    s.Session = s.Session + 1
    stealTask = task.spawn(StealLoop)
end

function SAE.Stop()
    local s = SAE.State
    s.Enabled = false; s.Mode = "NONE"; _G.PhantomX_Stealing = false
    s.Session = s.Session + 1
    if stealTask then pcall(task.cancel, stealTask) end
end

function ESP.Update(eggs)
    if type(eggs) ~= "table" then return end
    local n = math.min(#eggs, ESP.Max)
    for i = 1, n do
        local e = eggs[i]
        local c = ESP.Pool[i]
        if not c or not c.Anchor.Parent then c = MakeCard(160); ESP.Pool[i] = c end
        c.Anchor.CFrame = CFrame.new(e.Position)
        local key = table.concat({e.Uid or "", tostring(e.Value or 0), tostring(e.Rarity or ""), tostring(e.Icon or "")}, "|")
        if c.LastKey ~= key then
            c.LastKey = key
            c.Name.Text = tostring(e.Name or "Egg")
            c.Price.Text = "$" .. FNum(e.Value or 0) .. "/s"
            c.Strip.BackgroundColor3 = e.RarityColor or Color3.fromRGB(150,150,180)
            c.Icon.Image = (e.Icon and e.Icon ~= "") and e.Icon or ""
        end
        c.Dist.Text = math.floor(e.Distance or 0) .. "m"
        if not c.Billboard.Enabled then c.Billboard.Enabled = true end
    end
    for i = n + 1, #ESP.Pool do
        if ESP.Pool[i] and ESP.Pool[i].Billboard then ESP.Pool[i].Billboard.Enabled = false end
    end
end

function ESP.Clear()
    for _, c in ipairs(ESP.Pool) do pcall(function() c.Anchor:Destroy() end) end
    ESP.Pool = {}; ESP.LastHash = ""
end

local function ReadPlotEggs()
    local out = {}
    local rendered = WS:FindFirstChild("PlacedEggRenders")
    if not rendered then return out end
    local me = LP.UserId
    local prefix = tostring(me) .. "_"
    local recs = {}
    pcall(function()
        if EggStateModule and EggStateModule.ReadOwnerEggs then recs = EggStateModule.ReadOwnerEggs(me) or {} end
    end)
    for _, m in ipairs(rendered:GetChildren()) do
        if m:IsA("Model") and m.Name:sub(1, #prefix) == prefix then
            local uid = m.Name:sub(#prefix + 1)
            local rec = recs[uid]
            local e = {Uid = uid, Pos = nil, Name = "Egg", Rarity = "", Color = Color3.fromRGB(200, 200, 200), Icon = "", Muts = "", ReadyAt = nil, Ready = false, Value = 0}
            pcall(function() e.Pos = m:GetPivot().Position end)
            if rec then
                local dir
                pcall(function() dir = AssetsDirectory and AssetsDirectory[rec.AssetCategory] end)
                if dir then
                    e.Name = (dir.DisplayName and dir.DisplayName ~= "") and dir.DisplayName or tostring(rec.AssetCategory)
                    e.Icon = tostring(dir.Icon or "")
                    if dir.Rarity then
                        e.Rarity = tostring(dir.Rarity.DisplayName or dir.Rarity._id or "")
                        if typeof(dir.Rarity.Color) == "Color3" then e.Color = dir.Rarity.Color end
                    end
                    local grow = dir.Egg and tonumber(dir.Egg.GrowthTime)
                    local placed = rec.Placement and tonumber(rec.Placement.PlacedAt)
                    local mult = math.max(tonumber(rec.GrowthSpeedMultiplier) or 1, 0.01)
                    if grow and placed then e.ReadyAt = placed + grow / mult end
                end
                pcall(function()
                    e.Value = math.max(1, math.floor((tonumber(dir and dir.EarningRate) or 0) * (tonumber(rec.AssetScale) or 1)))
                end)
                pcall(function()
                    if type(rec.Mutations) == "table" and #rec.Mutations > 0 then
                        local names = {}
                        for _, mu in ipairs(rec.Mutations) do
                            names[#names + 1] = tostring(type(mu) == "table" and (mu.DisplayName or mu._id or "?") or mu)
                        end
                        e.Muts = table.concat(names, " · ")
                    end
                end)
            end
            pcall(function()
                if EggStateModule and EggStateModule.IsReadyToHatch then e.Ready = EggStateModule.IsReadyToHatch(uid) == true end
            end)
            if e.Pos then out[#out + 1] = e end
        end
    end
    return out
end

local function UpdatePlotESP()
    local now = WS:GetServerTimeNow()
    local seen = {}
    for _, e in ipairs(PlotESP.Latest) do
        seen[e.Uid] = true
        local c = PlotESP.Cards[e.Uid]
        if not c or not c.Anchor.Parent then c = MakePlotCard(); PlotESP.Cards[e.Uid] = c end
        c.Anchor.CFrame = CFrame.new(e.Pos)
        c.Strip.BackgroundColor3 = e.Color
        c.Icon.Image = e.Icon
        c.Name.Text = e.Name
        local bits = {}
        if (e.Value or 0) > 0 then bits[#bits + 1] = FNum(e.Value) .. "/s" end
        if e.Rarity ~= "" then
            local c3 = e.Color
            if 0.299*c3.R + 0.587*c3.G + 0.114*c3.B < 0.35 then c3 = c3:Lerp(Color3.new(1,1,1), 0.6) end
            bits[#bits + 1] = ('<font color="%s"><b>%s</b></font>'):format(PlotHex(c3), e.Rarity)
        end
        local left = e.ReadyAt and (e.ReadyAt - now)
        if e.Ready or (left and left <= 0) then
            bits[#bits + 1] = ('<font color="%s"><b>Ready</b></font>'):format(PlotHex(PlotESP.Green))
        elseif left then bits[#bits + 1] = fmtTime(left) end
        local txt = table.concat(bits, "  ·  ")
        if e.Muts ~= "" then txt = txt .. "\n" .. e.Muts end
        c.Sub.Text = txt
    end
    for uid, c in pairs(PlotESP.Cards) do
        if not seen[uid] then pcall(function() c.Anchor:Destroy() end); PlotESP.Cards[uid] = nil end
    end
end

local function weightKg(rec)
    local w
    pcall(function()
        if type(rec) ~= "table" then return end
        local itemData = type(rec.ItemData) == "table" and rec.ItemData or rec
        local category = itemData.AssetCategory or itemData.Category or rec.AssetCategory or rec.Category
        local rawScale = itemData.AssetScale or itemData.Scale or rec.AssetScale or rec.Scale
        local entry = category and AssetsDirectory and AssetsDirectory[category]
        local modelWeight = tonumber(entry and entry.ModelWeight)
        local scale = tonumber(rawScale)
        if modelWeight == nil or scale == nil then return end
        w = modelWeight * math.max(scale, 0) ^ 3
    end)
    return w
end

local function itemValue(item)
    if type(item) ~= "table" then return 0, false end
    local cat = item.Category or item.AssetCategory
    local scale = tonumber(item.Scale or item.AssetScale) or 1
    local entry = cat and AssetsDirectory and AssetsDirectory[cat]
    if not entry then return 0, false end
    local base = tonumber(entry.EarningRate) or 0
    local mult = 1
    if type(item.Mutations) == "table" then
        for _, m in pairs(item.Mutations) do if m ~= "None" then mult = mult * 1.5 end end
    end
    return base * scale * mult, true
end

    return (syn and syn.request) or (http and http.request) or http_request or request

local function normalizeURL(v)
    return tostring(v or ""):gsub("[%c%s]", ""):gsub("^<", ""):gsub(">$", "")
end

local function validURL(u)
    return tostring(u):match("^https://[%w%.%-]+/api/webhooks/%d+/%S+$") ~= nil
end

local function resolveThumb(icon)
    local raw = tostring(icon or "")
    if raw:match("^https://[%w%-]+%.rbxcdn%.com/") then return raw end
    local id = raw:match("(%d+)")
    if not id then return nil end
    if Webhook.ThumbCache[id] then return Webhook.ThumbCache[id] end
    local req = getRequester()
    if type(req) ~= "function" then return nil end
    local placeholder = "e5bef3179d5ce82a42fdc8ddc83a2ba9"
    local url = ("https://thumbnails.roblox.com/v1/assets?assetIds=%s&size=420x420&format=Png&isCircular=false"):format(id)
    for attempt = 1, 2 do
        local ok, r = pcall(req, {Url = url, Method = "GET", Headers = {["Accept"] = "application/json"}})
        local code = ok and type(r) == "table" and tonumber(r.StatusCode or r.Status) or nil
        local body = ok and type(r) == "table" and (r.Body or r.body) or nil
        if code and (code < 200 or code >= 300) then return nil end
        if type(body) ~= "string" then return nil end
        local ok2, dec = pcall(HttpService.JSONDecode, HttpService, body)
        local item = ok2 and type(dec) == "table" and type(dec.data) == "table" and dec.data[1] or nil
        local img = item and item.imageUrl
        if type(img) == "string" and img:find(placeholder, 1, true) then return nil end
        if type(img) == "string" and img:match("^https://[%w%-]+%.rbxcdn%.com/") then
            Webhook.ThumbCache[id] = img
            return img
        end
        if not item or item.state ~= "Pending" or attempt == 2 then return nil end
        task.wait(0.2)
    end
    return nil
end

    Categories = {}, MinValue = 0, KGMode = "Any", KGThreshold = 0,
}

function Fuse.isReady()
    return FuseMods and FuseMods.Remotes and FuseMods.Remotes.LoadPet
        and FuseMods.Remotes.BeginFuse and FuseMods.Remotes.FinishReveal and FuseMods.Remotes.EjectPet
end

function Fuse.finite(v)
    return type(v) == "number" and v == v and v >= 0 and v < math.huge
end

function Fuse.eligible(uid, raw, save, loaded)
    if not FuseMods or type(raw) ~= "table" then return false end
    if Fuse.Categories[raw.Category] ~= true then return false end
    if not Fuse.finite(Fuse.MinValue) or not Fuse.finite(Fuse.KGThreshold) then return false end
    if raw.IsFavorite == true or (raw.InFuse == true) ~= loaded then return false end
    for _, equipped in ipairs(save.EquippedAssets or {}) do
        if tostring(equipped) == tostring(uid) then return false end
    end
    local ok, allowed = pcall(FuseMods.Kernel.MayEnterFuse, uid, raw, nil, loaded)
    if not ok or allowed ~= true then return false end
    local ok2, item = pcall(FuseMods.AssetItems.Decode, raw)
    if not ok2 or type(item) ~= "table" then return false end
    if Fuse.KGMode ~= "Any" and Fuse.KGThreshold > 0 then
        local w = weightKg(item)
        if not Fuse.finite(w) then return false end
        local displayed = math.round(math.max(w, 0))
        if Fuse.KGMode == "Below" and displayed <= Fuse.KGThreshold then return false end
        if Fuse.KGMode == "Above" and displayed >= Fuse.KGThreshold then return false end
    end
    if Fuse.MinValue > 0 then
        local value, known = itemValue(item)
        if not known or not Fuse.finite(value) or value >= Fuse.MinValue then return false end
    end
    return true, item
end

function Fuse.plan(save)
    if not FuseMods or type(save) ~= "table" or type(save.Inventory) ~= "table" or type(save.FusionSlots) ~= "table" then
        return nil, "Waiting for save"
    end
    local loaded, category, seen = {}, nil, {}
    for i = 1, 3 do
        local uid = save.FusionSlots[i]
        if uid then
            if seen[uid] or not Fuse.Owned[uid] then return nil, "Machine loaded" end
            seen[uid] = true
            local raw = save.Inventory[uid]
            if not Fuse.eligible(uid, raw, save, true) then return nil, "Returning protected", uid end
            if category and raw.Category ~= category then return nil, "Mismatched pet", uid end
            category = raw.Category
            loaded[#loaded + 1] = uid
        end
    end
    local groups = {}
    if category then groups[category] = table.clone(loaded) end
    for uid, raw in pairs(save.Inventory) do
        if not seen[uid] and (not category or raw.Category == category) and Fuse.eligible(uid, raw, save, false) then
            groups[raw.Category] = groups[raw.Category] or {}
            table.insert(groups[raw.Category], uid)
        end
    end
    local cats = {}
    for name, uids in pairs(groups) do if #uids >= 3 then cats[#cats + 1] = name end end
    table.sort(cats)
    if #cats == 0 then return nil, "Waiting for 3 pets" end
    local uids = groups[cats[1]]
    local remaining = {}
    for _, uid in ipairs(uids) do if not seen[uid] then remaining[#remaining + 1] = uid end end
    table.sort(remaining, function(a, b) return tostring(a) < tostring(b) end)
    local result = table.clone(loaded)
    for _, uid in ipairs(remaining) do
        if #result == 3 then break end
        result[#result + 1] = uid
    end
    return {uids = result, loaded = #loaded, category = cats[1]}, "Ready"
end

function Fuse.tick()
    if not Fuse.Enabled or not Fuse.isReady() then return end
    if SAE.State.Carrying then Fuse.Status = "Busy" return end
    local save
    pcall(function() save = FuseMods.Save.Get() end)
    if type(save) ~= "table" or type(save.FusionSlots) ~= "table" then Fuse.Status = "Waiting for save"; return end
    if Fuse.Pending then
        local p = Fuse.Pending
        local consumed = p.action == "BeginFuse" and p.uids ~= nil
        if consumed then
            for _, uid in ipairs(p.uids) do if save.Inventory[uid] then consumed = false break end end
        end
        local observed = (p.action == "LoadPet" and save.Inventory[p.uid] and save.Inventory[p.uid].InFuse == true)
            or (p.action == "EjectPet" and (not save.Inventory[p.uid] or not save.Inventory[p.uid].InFuse))
            or (p.action == "BeginFuse" and (save.FusionLocked == true or type(save.FusionEggReward) == "table" or consumed))
            or (p.action == "FinishReveal" and save.FusionEggReward == false)
        if not observed and os.clock() - p.at < 8 then Fuse.Status = "Waiting" return end
        if not observed then Fuse.Enabled = false; Fuse.Status = "Timeout · stopped"; Fuse.Pending = nil; return end
        Fuse.Pending = nil
    end
    if save.FusionEggReward ~= false and save.FusionEggReward ~= nil then
        if type(save.FusionEndsAt) == "number" and save.FusionEndsAt > WS:GetServerTimeNow() then Fuse.Status = "Fusing"; return end
        Fuse.Status = "Claiming"
        Fuse.Pending = {action = "FinishReveal", at = os.clock()}
        pcall(function() FuseMods.Remotes.FinishReveal:InvokeServer() end)
        return
    end
    if save.FusionLocked == true then Fuse.Status = "Fusing" return end
    local slotSet = {}
    for _, uid in pairs(save.FusionSlots) do if uid then slotSet[uid] = true end end
    for uid in pairs(Fuse.Owned) do if not slotSet[uid] then Fuse.Owned[uid] = nil end end
    local plan, reason, eject = Fuse.plan(save)
    if eject then
        Fuse.Status = "Returning"
        Fuse.Pending = {action = "EjectPet", uid = eject, at = os.clock()}
        pcall(function() FuseMods.Remotes.EjectPet:InvokeServer(eject) end)
        return
    end
    if not plan then Fuse.Status = reason or "Waiting" return end
    local count, cap = getEggInventoryCount()
    if cap > 0 and count >= cap then Fuse.Status = "Waiting for space" return end
    local items = {}
    for _, uid in ipairs(plan.uids) do items[#items + 1] = FuseMods.AssetItems.Decode(save.Inventory[uid]) end
    local price
    pcall(function() price = FuseMods.Kernel.PriceFor(items) end)
    if not Fuse.finite(price) or not Fuse.finite(save.Money) or save.Money < price then Fuse.Status = "Waiting for cost"; return end
    if plan.loaded < 3 then
        local uid = plan.uids[plan.loaded + 1]
        local latest
        pcall(function() latest = FuseMods.Save.Get() end)
        local checked = Fuse.plan(latest)
        if not checked or checked.uids[checked.loaded + 1] ~= uid or checked.loaded ~= plan.loaded then return end
        Fuse.Owned[uid] = true
        Fuse.Pending = {action = "LoadPet", uid = uid, at = os.clock()}
        Fuse.Status = "Loading " .. plan.category .. " " .. tostring(plan.loaded + 1) .. "/3"
        pcall(function() FuseMods.Remotes.LoadPet:InvokeServer(uid) end)
    else
        Fuse.Status = "Fusing " .. plan.category
        Fuse.Pending = {action = "BeginFuse", at = os.clock(), uids = table.clone(plan.uids)}
        local ok = pcall(function() FuseMods.Remotes.BeginFuse:InvokeServer() end)
        if ok then Fuse.Completed = Fuse.Completed + 1 end
    end
end

    local function BuildOpts()
        local opts = {T("mainNone")}
        local eggs = SAE.Evaluate()

            local e = eggs[i]
            opts[#opts + 1] = string.format("%s | %s/s | %s", e.Name, FNum(e.Value), e.Rarity)
        end
        return opts
    end
    local function CurrentLabel()
        if not SAE.State.SelectedUid then return T("mainNone") end

            if e.Uid == SAE.State.SelectedUid then
