local function CleanupAllGUIs()

        if g:IsA("ScreenGui") and (g.Name:find("Wind") or g.Name:find("PhantomX") or g.Name:find("PX_")) then
            g:Destroy()
        end
    end
end

local function FNum(n)
    n = tonumber(n) or 0
    local u = {{1e12,"T"},{1e9,"B"},{1e6,"M"},{1e3,"K"}}
    for _, x in ipairs(u) do
        if n >= x[1] then
            local v = n/x[1]
            return (v < 10 and string.format("%.2f",v) or string.format("%.1f",v)):gsub("%.?0+$","")..x[2]
        end
    end
    return tostring(math.floor(n))
end
local function fmtTime(s)
    s = math.max(0, math.floor(s))
    local h, m = math.floor(s/3600), math.floor(s/60) % 60
    if h > 0 then return ("%dh %02dm"):format(h, m) end
    if m > 0 then return ("%dm %02ds"):format(m, s % 60) end
    return ("%ds"):format(s)
end

local function GetValue(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if not d then return 0 end
    return math.max(1, math.floor((tonumber(d.EarningRate) or 0) * (tonumber(rec.AssetScale) or 1)))
end
local function GetName(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.DisplayName ~= "" then return d.DisplayName end
    return tostring(rec.AssetCategory or "Egg")
end
local function GetRarity(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.Rarity then return d.Rarity.DisplayName or d.Rarity._id or "?" end
    return "?"
end
local function GetRarityColor(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.Rarity and typeof(d.Rarity.Color) == "Color3" then
        return d.Rarity.Color
    end
    return Color3.fromRGB(200, 200, 220)

    return (d and tostring(d.Icon or "")) or ""
end

function SAE.Evaluate()
    local s = SAE.State
    local now = os.clock()
    if s.Cache and (now - s.CacheAt) < 3 then return s.Cache end
    local recs = ReadField()
    if not recs then return s.Cache or {} end
    local hrp = getHRP()
    local mp = hrp and hrp.Position or Vector3.new()
    local out = {}
    for _, r in ipairs(recs) do
        if type(r) == "table" and r.Uid and typeof(r.BoundsCFrame) == "CFrame"
            and (r.State == "Slot" or r.State == "Dropped") then
            local p = r.BoundsCFrame.Position
            table.insert(out, {
                Uid = r.Uid,
                Name = GetName(r),
                Value = GetValue(r),
                Rarity = GetRarity(r),
                RarityColor = GetRarityColor(r),
                Icon = GetIcon(r),
                Position = p,
                Distance = (p - mp).Magnitude,
                Area = r.AreaId,
                Record = r,
            })
        end
    end
    table.sort(out, function(a, b) return a.Value > b.Value end)
    s.Cache, s.CacheAt = out, now
    return out
end

function SAE.TweenTo(target, speed, sid)
    local hrp = getHRP()
    if not hrp then return false end
    speed = math.clamp(speed or SAE.State.FlightSpeed, 100, 1000)
    local d = (target - hrp.Position).Magnitude
    if d < 3 then return true end
    local dir = (target - hrp.Position).Unit
    local goal = CFrame.lookAt(target, target + dir)
    local tw = TweenService:Create(hrp, TweenInfo.new(d/speed, Enum.EasingStyle.Linear), {CFrame = goal})
    local hb
    hb = RunService.Heartbeat:Connect(function()
        if sid and sid ~= SAE.State.Session then
            pcall(function() tw:Cancel() end)
            if hb then hb:Disconnect() end
            return
        end
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    tw:Play()
    tw.Completed:Wait()
    if hb then hb:Disconnect() end
    return true
end

local stealTask
local function StealLoop()
    local s = SAE.State
    while s.Enabled do
        local sid = s.Session
        task.wait(0.15)
        if not s.Enabled or sid ~= s.Session then break end
        if s.Ragdolled then task.wait(0.3); continue end

        local eggs = SAE.Evaluate()
        local target
        if s.SelectedUid then
            for _, e in ipairs(eggs) do
                if e.Uid == s.SelectedUid then target = e; break end
            end
        end
        if not target then
            for _, e in ipairs(eggs) do
                if not s.Unreachable[e.Uid] then target = e; break end
            end
        end
        if not target then task.wait(2); continue end

        local hrp = getHRP()
        if not hrp then task.wait(1); continue end

        if s.Treadmilling then
            local d = GetRemote("RF/Treadmill/AskDoff")
            if d then pcall(function() d:InvokeServer() end) end
            s.Treadmilling = false
            task.wait(0.3)
        end

        SAE.TweenTo(target.Position + Vector3.new(0, 1.5, 0), s.FlightSpeed, sid)
        if sid ~= s.Session or not s.Enabled then break end

        local ok = SAE.Carry(target.Uid)
        if not ok then
            s.Unreachable[target.Uid] = os.clock()
            task.wait(0.5)
            continue
        end

        task.wait(2.5)
        if sid ~= s.Session or not s.Enabled then break end

        for _ = 1, 6 do
            if sid ~= s.Session then break end
            if SAE.IsCarrying() then break end
            SAE.Carry(target.Uid)
            task.wait(0.25)
        end

        if s.AutoReturn then
            SAE.TweenTo(SAFE_ZONE_POS, s.FlightSpeed, sid)
            task.wait(0.6)
        end

        if s.AutoPlace5 then
            s.BatchCount = s.BatchCount + 1
            if s.BatchCount >= 5 then
                s.BatchCount = 0
                task.spawn(function()
                    SAE.PlaceAll()
                    if s.AutoHatch then SAE.HatchAll() end
                end)
            end
        end

        task.wait(0.4)
    end
end

function SAE.Start(mode)
    local s = SAE.State
    if s.Enabled and s.Mode == mode then return end
    if s.Enabled then
        s.Session = s.Session + 1
        task.wait(0.1)
    end
    s.Enabled = true
    s.Mode = mode
    _G.PhantomX_Stealing = true
    s.Session = s.Session + 1
    stealTask = task.spawn(StealLoop)
end

    s.Mode = "NONE"
    _G.PhantomX_Stealing = false
    s.Session = s.Session + 1
    if stealTask then pcall(task.cancel, stealTask) end
end

function ESP.Update(eggs)
    if type(eggs) ~= "table" then return end
    local n = math.min(#eggs, ESP.Max)
    for i = 1, n do
        local e = eggs[i]
        local c = ESP.Pool[i]
        if not c or not c.Anchor.Parent then
            c = MakeCard(160)
            ESP.Pool[i] = c
        end
        c.Anchor.CFrame = CFrame.new(e.Position)
        local key = table.concat({
            e.Uid or "", tostring(e.Value or 0),
            tostring(e.Rarity or ""), tostring(e.Icon or ""),
        }, "|")
        if c.LastKey ~= key then
            c.LastKey = key
            c.Name.Text = tostring(e.Name or "Egg")
            c.Price.Text = "$" .. FNum(e.Value or 0) .. "/s"
            local rc = e.RarityColor or Color3.fromRGB(150,150,180)
            c.Strip.BackgroundColor3 = rc
            c.Icon.Image = (e.Icon and e.Icon ~= "") and e.Icon or ""
        end
        c.Dist.Text = math.floor(e.Distance or 0) .. "m"
        if not c.Billboard.Enabled then c.Billboard.Enabled = true end
    end
    for i = n + 1, #ESP.Pool do
        if ESP.Pool[i] and ESP.Pool[i].Billboard then
            ESP.Pool[i].Billboard.Enabled = false
        end
    end
end

function ESP.Clear()
    for _, c in ipairs(ESP.Pool) do
        pcall(function() c.Anchor:Destroy() end)
    end
    ESP.Pool = {}
    ESP.LastHash = ""
end

                ESP.LastHash = hash
                ESP.Update(eggs)

        else

        end
    end
end)

local function PlotHex(c)
    return ("#%02X%02X%02X"):format(

local function ReadPlotEggs()
    local out = {}
    local rendered = WS:FindFirstChild("PlacedEggRenders")
    if not rendered then return out end
    local me = LP.UserId
    local prefix = tostring(me) .. "_"
    local recs = {}
    pcall(function()
        if EggStateModule and EggStateModule.ReadOwnerEggs then
            recs = EggStateModule.ReadOwnerEggs(me) or {}
        end
    end)
    for _, m in ipairs(rendered:GetChildren()) do
        if m:IsA("Model") and m.Name:sub(1, #prefix) == prefix then
            local uid = m.Name:sub(#prefix + 1)
            local rec = recs[uid]
            local e = {
                Uid = uid, Pos = nil, Name = "Egg", Rarity = "",
                Color = Color3.fromRGB(200, 200, 200), Icon = "",
                Muts = "", ReadyAt = nil, Ready = false, Value = 0,
            }
            pcall(function() e.Pos = m:GetPivot().Position end)
            if rec then
                local dir
                pcall(function() dir = AssetsDirectory and AssetsDirectory[rec.AssetCategory] end)
                if dir then
                    if dir.DisplayName and dir.DisplayName ~= "" then
                        e.Name = dir.DisplayName
                    else
                        e.Name = tostring(rec.AssetCategory)
                    end
                    e.Icon = tostring(dir.Icon or "")
                    if dir.Rarity then
                        e.Rarity = tostring(dir.Rarity.DisplayName or dir.Rarity._id or "")
                        if typeof(dir.Rarity.Color) == "Color3" then
                            e.Color = dir.Rarity.Color
                        end
                    end
                    local grow = dir.Egg and tonumber(dir.Egg.GrowthTime)
                    local placed = rec.Placement and tonumber(rec.Placement.PlacedAt)
                    local mult = math.max(tonumber(rec.GrowthSpeedMultiplier) or 1, 0.01)
                    if grow and placed then e.ReadyAt = placed + grow / mult end
                end
                pcall(function()
                    e.Value = math.max(1, math.floor(
                        (tonumber(dir and dir.EarningRate) or 0)
                        * (tonumber(rec.AssetScale) or 1)))
                end)
                pcall(function()
                    if type(rec.Mutations) == "table" and #rec.Mutations > 0 then
                        local names = {}
                        for _, mu in ipairs(rec.Mutations) do
                            names[#names + 1] = tostring(type(mu) == "table"
                                and (mu.DisplayName or mu._id or "?") or mu)
                        end
                        e.Muts = table.concat(names, " · ")
                    end
                end)
            end
            pcall(function()
                if EggStateModule and EggStateModule.IsReadyToHatch then
                    e.Ready = EggStateModule.IsReadyToHatch(uid) == true
                end
            end)
            if e.Pos then out[#out + 1] = e end
        end
    end
    return out
end

local function UpdatePlotESP()
    local now = WS:GetServerTimeNow()
    local seen = {}
    for _, e in ipairs(PlotESP.Latest) do
        seen[e.Uid] = true
        local c = PlotESP.Cards[e.Uid]
        if not c or not c.Anchor.Parent then
            c = MakePlotCard()
            PlotESP.Cards[e.Uid] = c
        end
        c.Anchor.CFrame = CFrame.new(e.Pos)
        c.Strip.BackgroundColor3 = e.Color
        c.Icon.Image = e.Icon
        c.Name.Text = e.Name

        local bits = {}
        if (e.Value or 0) > 0 then
            bits[#bits + 1] = FNum(e.Value) .. "/s"
        end
        if e.Rarity ~= "" then
            local c3 = e.Color
            if 0.299*c3.R + 0.587*c3.G + 0.114*c3.B < 0.35 then
                c3 = c3:Lerp(Color3.new(1,1,1), 0.6)
            end
            bits[#bits + 1] = ('<font color="%s"><b>%s</b></font>')
                :format(PlotHex(c3), e.Rarity)
        end
        local left = e.ReadyAt and (e.ReadyAt - now)
        if e.Ready or (left and left <= 0) then
            bits[#bits + 1] = ('<font color="%s"><b>Ready</b></font>')
                :format(PlotHex(PlotESP.Green))
        elseif left then
            bits[#bits + 1] = fmtTime(left)
        end
        local txt = table.concat(bits, "  ·  ")
        if e.Muts ~= "" then txt = txt .. "\n" .. e.Muts end
        c.Sub.Text = txt
    end
    for uid, c in pairs(PlotESP.Cards) do
        if not seen[uid] then
            pcall(function() c.Anchor:Destroy() end)
            PlotESP.Cards[uid] = nil
        end
    end
end

local function ClearPlotESP()
    for uid, c in pairs(PlotESP.Cards) do
        pcall(function() c.Anchor:Destroy() end)
    end
    PlotESP.Cards = {}
    local f = WS:FindFirstChild("PhantomX_PlotESP")
    if f then f:Destroy() end
    PlotESP.Folder = nil
end

local function DoPlaceAll()
    local save = GetSave()
    if not save or type(save.EggInventory) ~= "table" then return 0 end
    if not EggStateModule or not EggStateModule.PlantEgg then return 0 end
    local n = 0
    for uid, egg in pairs(save.EggInventory) do
        if type(egg) == "table" and egg.Placement == nil then
            pcall(function()
                EggStateModule.PlantEgg(uid, CFrame.new(math.random(-5,5), 0, math.random(-5,5)))
            end)
            n = n + 1
            task.wait(0.15)
        end
    end
    return n
end

local function DoHatch()
    local save = GetSave()
    if not save or type(save.EggInventory) ~= "table" then return 0 end
    if not EggStateModule then return 0 end
    local n = 0
    for uid, egg in pairs(save.EggInventory) do
        if type(egg) == "table" then
            local ready = true
            if EggStateModule.IsReadyToHatch then
                local ok, r = pcall(EggStateModule.IsReadyToHatch, uid)
                ready = ok and r == true
            end
            if ready and EggStateModule.BeginHatch then
                pcall(function()
                    EggStateModule.BeginHatch(uid)
                    if EggStateModule.FinishHatch then EggStateModule.FinishHatch(uid) end
                end)
                n = n + 1
                task.wait(0.2)
            end
        end
    end
    return n
end

local function DoUpTread()
    local TreadmillData
    pcall(function() TreadmillData = require(RS.Data.Treadmills) end)
    local save = GetSave()
    if TreadmillData and save then
        local level = tonumber(save.TreadmillUpgradeLevel) or 0
        local okCfg, nextCfg
        if type(TreadmillData.GetByUpgradeLevel) == "function" then
            okCfg, nextCfg = pcall(TreadmillData.GetByUpgradeLevel, level + 1)
        end
        if okCfg and type(nextCfg) == "table" and nextCfg._id then
            local price = tonumber(nextCfg.Price) or math.huge
            local money = tonumber(save.Money) or 0
            if money < price then return false end
            local net = RS:FindFirstChild("Packages") and RS.Packages:FindFirstChild("Networking")
            local r = net and (net:FindFirstChild("RF/Treadmill/AskTierRaise") or net:FindFirstChild("RF/Treadmill/AskTierRaise", true))
            if not r then r = GetRemote("RF/Treadmill/AskTierRaise") end
            if r then
                local ok, result = pcall(function() return r:InvokeServer(nextCfg._id) end)
                return ok and result == true
            end
        end
    end
    local r = GetRemote("RF/Treadmill/AskTierRaise")
    if r then return pcall(function() r:InvokeServer() end) end
    return false
end
local function DoEnterAbyss()
    local r = GetRemote("RF/BossEvent/AskEnter")
    if not r then return false, "No remote" end
    local ok, acc = pcall(function() return r:InvokeServer() end)
    if not ok then return false, tostring(acc) end
    return acc == true
end
