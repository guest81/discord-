--==============================================================
-- Roles (MM2 remote: ReplicatedStorage.GetPlayerData)
--==============================================================
local PlayerRoles = {}
local function UpdateRoles()
    local remote = ReplicatedStorage:FindFirstChild("GetPlayerData", true)
    if not remote then return end
    local ok, data = pcall(function() return remote:InvokeServer() end)
    if ok and type(data) == "table" then
        PlayerRoles = {}
        for name, info in pairs(data) do
            PlayerRoles[name] = info.Role
        end
    end
end
local function GetRole(plr)
    return PlayerRoles[plr.Name]
end
local function FindMurderer()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and (p.Character:FindFirstChild("Knife") or (p:FindFirstChild("Backpack") and p.Backpack:FindFirstChild("Knife"))) then
            return p
        end
    end
end
local function FindSheriff()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and (p.Character:FindFirstChild("Gun") or (p:FindFirstChild("Backpack") and p.Backpack:FindFirstChild("Gun"))) then
            return p
        end
    end
end

local OriginalAnimations = {}
local SelectedAnimation  = "Adidas Community"
local CurrentAnimPack    = nil

local function SaveOriginalAnimations(char)
    local animate = char:FindFirstChild("Animate")
    if not animate then return end
    for _, slot in ipairs(AnimSlots) do
        local folder = animate:FindFirstChild(slot[1])
        local anim = folder and folder:FindFirstChild(slot[2])
        if anim and anim:IsA("Animation") then
            OriginalAnimations[slot[1] .. ":" .. slot[2]] = anim.AnimationId
        end
    end
end

local function ApplyAnimationIds(ids) -- ids: map "folder:anim" -> id string
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local animate = char and char:FindFirstChild("Animate")
    if not (hum and animate) then return false end
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do track:Stop() end
    animate.Disabled = true
    task.wait(0.1)
    for key, id in pairs(ids) do
        local folderName, animName = string.match(key, "^(.-):(.+)$")
        local folder = animate:FindFirstChild(folderName)
        local anim = folder and folder:FindFirstChild(animName)
        if anim and anim:IsA("Animation") then
            anim.AnimationId = id
        end
    end
    animate.Disabled = false
    task.wait(0.1)
    hum:ChangeState(Enum.HumanoidStateType.Landed)
    task.wait(0.05)
    hum:ChangeState(Enum.HumanoidStateType.Running)
    return true
end

local function ApplyAnimationPack(name)
    local pack = AnimationPacks[name]
    if not pack then return false end
    local ids = {}
    for _, slot in ipairs(AnimSlots) do
        if pack[slot[3]] then
            ids[slot[1] .. ":" .. slot[2]] = "rbxassetid://" .. pack[slot[3]]
        end
    end
    CurrentAnimPack = name
    return ApplyAnimationIds(ids)
end

    -- picks a public server from the Roblox list; smallest = fewest players first
    local function HopServer(smallest)
        local url = ("https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=%s&limit=100"):format(game.PlaceId, smallest and "Asc" or "Desc")
        local ok, body = pcall(function() return game:HttpGet(url) end)
        if not ok then return false end
        local ok2, data = pcall(function() return game:GetService("HttpService"):JSONDecode(body) end)
        if not ok2 or type(data) ~= "table" or type(data.data) ~= "table" then return false end
        local pool = {}
        for _, s in ipairs(data.data) do
            if s.id ~= game.JobId and tonumber(s.playing) and tonumber(s.maxPlayers) and s.playing < s.maxPlayers then
                table.insert(pool, s)
            end
        end
        if #pool == 0 then return false end
        local pick = smallest and pool[1] or pool[math.random(1, math.min(#pool, 10))]
        local tp = pcall(function()
            game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId, pick.id, LocalPlayer)
        end)
        return tp
    end

    local function NewESP()
        local o = {

            Tracer = Drawing.new("Line"),

            Dist   = Drawing.new("Text"),

    local function DrawFor(o, pos3, color, cat, label)
        local cam = workspace.CurrentCamera
        local sp, onScreen = cam:WorldToViewportPoint(pos3)
        local show = onScreen
        local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local dist = myHRP and (myHRP.Position - pos3).Magnitude or 0
        local scale = 1 / (sp.Z * math.tan(math.rad(cam.FieldOfView / 2)) * 2) * 1000
        local w, h = 4 * scale, 6 * scale
        o.Box.Visible = show and ESPSettings.boxes[cat]
        o.Box.Color = color; o.Box.Size = Vector2.new(w, h); o.Box.Position = Vector2.new(sp.X - w / 2, sp.Y - h / 2)
        o.Tracer.Visible = show and ESPSettings.tracers[cat]
        o.Tracer.Color = color
        o.Tracer.From = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
        o.Tracer.To = Vector2.new(sp.X, sp.Y)
        o.Name.Visible = show and ESPSettings.names[cat]
        o.Name.Color = color; o.Name.Text = label; o.Name.Position = Vector2.new(sp.X, sp.Y - h / 2 - 16)
        o.Dist.Visible = show and ESPSettings.distance[cat]
        o.Dist.Color = color; o.Dist.Text = math.floor(dist) .. "m"; o.Dist.Position = Vector2.new(sp.X, sp.Y + h / 2 + 2)
    end

    -- load any animation pack straight from the Roblox catalog (bundle id)
    local BundleInput = ""
    local function ResolveBundle(id)
        local ok, body = pcall(function() return game:HttpGet("https://catalog.roblox.com/v1/bundles/" .. id .. "/details") end)
        if not ok then return nil end
        local ok2, data = pcall(function() return game:GetService("HttpService"):JSONDecode(body) end)
        if not ok2 or type(data) ~= "table" or type(data.items) ~= "table" then return nil end
        local pack = {}
        for _, item in ipairs(data.items) do
            local n, itemId = tostring(item.name):lower(), tonumber(item.id)
            if itemId then
                if n:find("%f[%a]swim%f[%A]") and n:find("%f[%a]idle%f[%A]") then pack.SwimIdle = itemId
                elseif n:find("%f[%a]swim%f[%A]") then pack.Swim = itemId
                elseif n:find("%f[%a]idle%f[%A]") then pack.Idle1 = pack.Idle1 or itemId; pack.Idle2 = pack.Idle2 or itemId
                elseif n:find("%f[%a]walk%f[%A]") then pack.Walk = itemId
                elseif n:find("%f[%a]run%f[%A]") then pack.Run = itemId
                elseif n:find("%f[%a]jump%f[%A]") then pack.Jump = itemId
                elseif n:find("%f[%a]fall%f[%A]") then pack.Fall = itemId
                elseif n:find("%f[%a]climb%f[%A]") then pack.Climb = itemId end
            end
        end
        if not (pack.Walk or pack.Run or pack.Idle1) then return nil end
        return pack, tostring(data.name or ("Bundle " .. id))
    end

                    local s = Samples[p]
                    if s and now - s.t > 0 then

                        if a.Magnitude > 150 then a = a.Unit * 150 end
                        s.a = s.a:Lerp(a, 0.4)

        local PingMul = 1
        -- lead time = base + ping; wall = longer lead used for the through-wall shot
        local function Predict(plr, wall)
            local char = plr and plr.Character
            local root = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso"))
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not (root and hum) then return nil end
            local ping = math.clamp(LocalPlayer:GetNetworkPing(), 0.005, 0.5)
            local t = wall and (0.145 + ping * 1.1 * PingMul) or (0.085 + ping * 1.05 * PingMul)
            local pos, vel = root.Position, root.AssemblyLinearVelocity
            local s = Samples[plr]
            local accel = s and s.a or Vector3.zero
            local out = pos + vel * t + 0.5 * accel * t * t
            local st = hum:GetState()
            if st == Enum.HumanoidStateType.Freefall or st == Enum.HumanoidStateType.Jumping then
                out += Vector3.new(0, -0.5 * workspace.Gravity * t * t, 0)
            end
            if hum.MoveDirection.Magnitude > 0 then
                local vdir = vel.Magnitude > 0 and vel.Unit or Vector3.zero
                if hum.MoveDirection:Dot(vdir) < 0.3 then
                    out = out:Lerp(pos + hum.MoveDirection * hum.WalkSpeed * t, 0.2)
                end
            end
            return out
        end

            local function DoDodge(dir)
                local root, hum = MyRoot(), MyHum()
                if not (root and hum) then return end
                Dodge.Last = os.clock()
                if Dodge.Mode == "Platform" then
                    if Dodge.Busy then return end
                    Dodge.Busy = true
                    task.spawn(function()
                        local back = root.CFrame
                        local plat = Instance.new("Part")
                        plat.Anchored = true
                        plat.Size = Vector3.new(20, 1, 20)
                        plat.Transparency = 1
                        plat.Position = root.Position - Vector3.new(1000, 3, 1000)
                        plat.Parent = workspace
                        root.CFrame = plat.CFrame * CFrame.new(0, 4, 0)
                        task.wait(0.7)
                        local r2 = MyRoot()
                        if r2 then
                            r2.CFrame = back
                            r2.AssemblyLinearVelocity = Vector3.zero
                        end
                        plat:Destroy()
                        Dodge.Busy = false
                    end)
                    return
                end
                hum:ChangeState(Enum.HumanoidStateType.Jumping)
                root.AssemblyLinearVelocity = Vector3.new(dir.X * 42, math.max(root.AssemblyLinearVelocity.Y, Dodge.Jump), dir.Z * 42)
                hum:Move(dir, false)
            end
            local function Track(obj)
                if not obj or not obj.Parent then return end
                local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
                if part then Knives[obj] = { part = part, p = part.Position, t = os.clock() } end
            end
            local function Step()
                if Dodge.Busy or os.clock() - Dodge.Last < 0.45 or IAmMurderer() then return end
                local root = MyRoot()
                if not root then return end
                local now = os.clock()
                for k, d in pairs(Knives) do
                    local part = d.part
                    if not (part and part.Parent) then
                        Knives[k] = nil
                    else
                        local dt = now - d.t
                        if dt > 0 then
                            local vel = (part.Position - d.p) / dt
                            d.p, d.t = part.Position, now
                            if vel.Magnitude > 8 then
                                local dir = vel.Unit
                                local rel = root.Position - part.Position
                                local along = rel:Dot(dir)
                                if along > 0 and along < 80 then
                                    local perp = rel - dir * along
                                    if perp.Magnitude < 5 and along / vel.Magnitude < 0.5 then
                                        local flat = Vector3.new(perp.X, 0, perp.Z)
                                        if flat.Magnitude < 0.4 then flat = Vector3.new(-dir.Z, 0, dir.X) end
                                        DoDodge(flat.Unit)
                                        return
                                    end
                                end
                            end
                        end
                    end
                end
            end
            local function DodgeStop()
                for _, c in ipairs(DConns) do pcall(function() c:Disconnect() end) end
                DConns = {}
                Knives = {}
            end

        -- downloads mm2meshes.lua from Lutosys/opensrc (github) and runs it, didn't review that file
        local function LoadData()
            if Skin.Data then return true end
            if not (httpRequest and isfile and writefile and loadfile) then return false end
            if not isfile("mm2data.lua") then
                local ok, res = pcall(httpRequest, { Url = MESH_URL, Method = "GET" })
                if ok and res and (res.Success or res.StatusCode == 200) and res.Body then
                    pcall(writefile, "mm2data.lua", res.Body)
                end
            end
            local ok, data = pcall(function() return loadfile("mm2data.lua")() end)
            if ok and type(data) == "table" then
                Skin.Data = data
                return true
            end
            return false
        end

        local function FindMesh(node)
            if type(node) ~= "table" then return nil end
            local props = node.Props
            if props then
                local meshId = props.MeshId or props.MeshID
                if meshId and meshId ~= "" then
                    return {
                        meshid = meshId,
                        textureid = props.TextureId or props.TextureID or "",
                        scale = props.Scale or Vector3.new(0.045, 0.045, 0.045),
                        size = props.Size or Vector3.new(0.045, 0.045, 0.045),
                    }
                end
            end
            if type(node.Display) == "table" then
                for _, child in ipairs(node.Display) do
                    local res = FindMesh(child)
                    if res then return res end
                end
            end
            return nil
        end
        local function WeaponData(name)
            local w = Skin.Data and Skin.Data[name]
            return w and FindMesh(w) or nil
        end

        local function ApplyMesh(refPart, info, kind)
            if not (info and info.meshid) then return end
            if kind == "Gun" then
                local tool = refPart:FindFirstAncestorOfClass("Tool")
                if tool then
                    if info.meshid:find("79401392") then
                        tool.Grip = CFrame.fromMatrix(Vector3.new(0, -0.7, -0.3), Vector3.new(1, 0, 0), Vector3.new(0, 0, 1), Vector3.new(0, -1, 0))
                    elseif info.meshid:find("6600918074") then
                        tool.Grip = CFrame.new(1, -0.36, 0, 0, 0, 1, 0, 1, 0, -1, 0, 0)
                    else
                        tool.Grip = CFrame.fromMatrix(Vector3.new(0, -0.5, 0.7), Vector3.new(1, 0, 0), Vector3.new(0, 1, 0), Vector3.new(0, 0, 1))
                    end
                end
            end
            if refPart:IsA("MeshPart") then
                local sm = refPart:FindFirstChildOfClass("SpecialMesh")
                if sm then sm:Destroy() end
                refPart.Size = info.size
                refPart.MeshId = info.meshid
                refPart.TextureID = info.textureid
            else
                local mesh = refPart:FindFirstChildOfClass("SpecialMesh")
                if not mesh then
                    mesh = Instance.new("SpecialMesh")
                    mesh.Name = "Mesh"
                    mesh.Parent = refPart
                end
                refPart.Size = info.size
                mesh.MeshId = info.meshid
                mesh.TextureId = info.textureid
                mesh.Scale = info.scale
            end
        end

        -- marks every weapon that has mesh data as owned in the local profile
        local function Unlock()
            if Skin.Unlocked then return true end
            local Modules = ReplicatedStorage:FindFirstChild("Modules")
            local Database = ReplicatedStorage:FindFirstChild("Database")
            if not (Modules and Database) then return false end
            local ok = pcall(function()
                Skin.Inventory = require(Modules.InventoryModule)
                Skin.Profile = require(Modules.ProfileData)
                Skin.Sync = require(Database.Sync)
            end)
            if not ok or not (Skin.Profile and Skin.Sync) then return false end
            for name, item in pairs(Skin.Sync.Weapons) do
                item.SortWithinGroup = item.SortWithinGroup or 0
                item.Name = item.Name or item.ItemName or name
                item.Rarity = item.Rarity or "Common"
                if Skin.Sync.Rarities[item.Rarity] then
                    local info = WeaponData(name)
                    if info and info.textureid and info.scale and info.size then
                        Skin.Profile.Weapons.Owned[name] = 1
                    end
                end
            end
            -- refresh the inventory screen (needs getgc/filtergc, line 122 fingerprint is from the original script)
            pcall(function()
                local upd = filtergc("table", { Keys = { "UpdateInventory" } }, true).UpdateInventory
                for _, f in pairs(getgc()) do
                    if typeof(f) == "function" and islclosure(f) and debug.info(f, "l") == 122 and #debug.getupvalues(f) == 2 then
                        upd(debug.getupvalue(f, 2), Skin.Inventory.MyInventory)
                    end
                end
            end)
            Skin.Unlocked = true
            return true
        end

        local function IAmMurderer()
            local c, bp = LocalPlayer.Character, LocalPlayer:FindFirstChild("Backpack")
            for _, holder in ipairs({ c, bp }) do
                if holder then
                    for _, v in ipairs(holder:GetChildren()) do
                        if v:IsA("Tool") and v:GetAttribute("ItemType") == "Knife" then return true end
                    end
                end
            end
            return false
        end

        local function SkinLoop()
            if Skin.Loop then return end
            Skin.Loop = true
            while Skin.On and UIAlive do
                local char = LocalPlayer.Character
                local equipped = Skin.Profile and Skin.Profile.Weapons and Skin.Profile.Weapons.Equipped
                if char and equipped then
                    local knifeInfo = equipped.Knife and WeaponData(equipped.Knife)
                    local gunInfo = equipped.Gun and WeaponData(equipped.Gun)
                    for _, tool in ipairs(char:GetChildren()) do
                        if tool:IsA("Tool") then
                            local kind = tool:GetAttribute("ItemType")
                            local handle = tool:FindFirstChild("Handle")
                            if handle then
                                if kind == "Knife" and knifeInfo then ApplyMesh(handle, knifeInfo, "Knife")
                                elseif kind == "Gun" and gunInfo then ApplyMesh(handle, gunInfo, "Gun") end
                            end
                        end
                    end
                    local refGun, refKnife = char:FindFirstChild("DisplayRefGun"), char:FindFirstChild("DisplayRefKnife")
                    local g = refGun and refGun.Value
                    local k = refKnife and refKnife.Value
                    if gunInfo and typeof(g) == "Instance" then ApplyMesh(g, gunInfo) end
                    if knifeInfo and typeof(k) == "Instance" then ApplyMesh(k, knifeInfo) end
                end
                task.wait()
            end
            Skin.Loop = false
        end

        SkinsTab:Toggle({ Title = T.skinSpawner, Desc = DS.skinSpawner, Value = false, Size = "medium", Callback = function(state)

                    Skin.On = false
                    return
                end
                if not Skin.Hooked then
                    Skin.Hooked = true
                    -- thrown knife sitting in the wall
                    table.insert(Connections, workspace.ChildAdded:Connect(function(ch)
                        if ch:IsA("BasePart") and ch.Name == "StuckKnife" and Skin.On then
                            local mesh = ch:WaitForChild("Mesh", 3)
                            local info = mesh and Skin.Profile.Weapons.Equipped.Knife and WeaponData(Skin.Profile.Weapons.Equipped.Knife)
                            if info and IAmMurderer() then
                                mesh.MeshId = info.meshid
                                mesh.TextureId = info.textureid
                                mesh.Scale = info.scale
                            end
                        end
                    end))
                end
                SkinLoop()
            end)
        end })
        table.insert(Cleanups, function() Skin.On = false end)
    end

        local function Decode(s)
            return (s:gsub("&amp;", "&"):gsub("&#39;", "'"):gsub("&quot;", '"'):gsub("<[^>]*>", ""))
        end

        local function ExtOf(url, default)
            local clean = url:lower():gsub("[%?#].*$", "")
            local e = clean:match("%.(%w+)$")
            if e and #e <= 4 then return e end
            return default
        end
        local function Download(url, ext)
            if not (httpRequest and writefile and getcustomasset and isfile) then return nil end
            pcall(function() if makefolder and not isfolder(DIR) then makefolder(DIR) end end)
            local key = 0
            for i = 1, #url do key = (key * 31 + url:byte(i)) % 4294967291 end
            local path = DIR .. "/a" .. key .. "." .. ext
            if not isfile(path) then
                local ok, res = pcall(httpRequest, { Url = url, Method = "GET" })
                if not (ok and res and res.Body and (res.Success or res.StatusCode == 200)) then return nil end
                local head = res.Body:sub(1, 32):lower()
                if head:find("<!doctype", 1, true) or head:find("<html", 1, true) then return nil end
                if not pcall(writefile, path, res.Body) then return nil end
            end
            local ok, asset = pcall(getcustomasset, path)
            if ok and type(asset) == "string" and asset ~= "" then return asset end
            return nil
        end

                if kind == "gun" then
                    Assets.Sound = asset
                    SoundWatch()
                elseif kind == "sky" then
                    for _, c in ipairs(Lighting:GetChildren()) do

                            table.insert(Assets.OrigSkies, c)
                            c.Parent = nil
                        end
