-- Safe loader
local function safeRequire(path, timeout)
    timeout = timeout or 2
    local parts = {}
    for part in string.gmatch(path, "[^%.]+") do
        table.insert(parts, part)
    end
    local current = ReplicatedStorage
    for i, part in ipairs(parts) do
        local child = current:FindFirstChild(part)
        if not child then
            if i == #parts then
                return nil
            end
            local success, result = pcall(function()
                return current:WaitForChild(part, timeout)
            end)
            if not success or not result then
                return nil
            end
            current = result
        else
            current = child
        end
    end
    if typeof(current) == "Instance" then
        local success, module = pcall(require, current)
        if success then
            return module
        end
    end
    return nil
end

local function resolveRarity(cat) return RarityMap[cat] end
local function getAssetName(cat) return tostring(cat) end

local function recordMutations(eggData)
    local list = {}
    if eggData and eggData.Mutations then
        for _, m in pairs(eggData.Mutations) do
            if type(m)=="string" then table.insert(list,m) end
        end
    end
    if eggData and eggData.BaseMutation and type(eggData.BaseMutation)=="string" then
        table.insert(list, eggData.BaseMutation)
    end
    return list
end

local function matchesEggFilters(egg, areas, rarities, mutations)
    if areas and #areas>0 then
        local areaId = egg.AreaId
        if not areaId then return false end
        local ok = false
        for _, a in ipairs(areas) do if tostring(a)==tostring(areaId) then ok=true; break end end
        if not ok then return false end
    end
    if rarities and #rarities>0 then
        local rarity = resolveRarity(egg.AssetCategory)
        if not rarity then return false end
        local ok = false
        for _, r in ipairs(rarities) do
            if tostring(r)==tostring(rarity) or tostring(r)==tostring(egg.AssetCategory) then ok=true; break end
        end
        if not ok then return false end
    end
    if mutations and #mutations>0 then
        local eggMuts = recordMutations(egg)
        local found = false
        for _, m in ipairs(eggMuts) do
            for _, s in ipairs(mutations) do
                if tostring(s)==tostring(m) then found=true; break end
            end
            if found then break end
        end
        if not found then return false end
    end
    return true
end

local function getAvailableEggs(areas, rarities, mutations, bigEggMinScale)
    local snapshot = Remotes.GetAreaEggSnapshot and Remotes.GetAreaEggSnapshot()
    local valid = {}
    if not snapshot then return valid end
    for _, egg in pairs(snapshot) do
        if not egg.IsCarried then
            if matchesEggFilters(egg, areas, rarities, mutations) then
                local scale = egg.AssetScale or 1
                if bigEggMinScale and bigEggMinScale>1 then
                    if scale>=bigEggMinScale then table.insert(valid, egg) end
                else
                    table.insert(valid, egg)
                end
            end
        end
    end
    return valid
end

local function sortEggsByPriority(eggs, priority)
    if priority=="Nearest" then
        local root = getRoot()
        if root then
            table.sort(eggs, function(a,b)
                return (a.Position-root.Position).Magnitude < (b.Position-root.Position).Magnitude
            end)
        end
    elseif priority=="Furthest" then
        local root = getRoot()
        if root then
            table.sort(eggs, function(a,b)
                return (a.Position-root.Position).Magnitude > (b.Position-root.Position).Magnitude
            end)
        end
    elseif priority=="Biggest Size" then
        table.sort(eggs, function(a,b) return (a.AssetScale or 1) > (b.AssetScale or 1) end)
    elseif priority=="Rarest" then
        table.sort(eggs, function(a,b)
            return (resolveRarity(a.AssetCategory) or 0) > (resolveRarity(b.AssetCategory) or 0)
        end)
    end
    return eggs
end

local function runAutoSteal()
    if isUnloaded or isCarryingEgg() or isInventoryFull() then return false end
    local areas = Options.StealZones and Options.StealZones.Value
    local rarities = Options.StealRarities and Options.StealRarities.Value
    local mutations = Options.StealMutations and Options.StealMutations.Value
    local priority = Options.StealPriority and Options.StealPriority.Value or "Nearest"
    local bigEggs = Options.StealBigEggs and Options.StealBigEggs.Value
    local bigScale = Options.StealBigEggScale and Options.StealBigEggScale.Value or 1.5
    local eggs = getAvailableEggs(areas, rarities, mutations, bigEggs and bigScale or nil)
    if #eggs==0 then return false end
    eggs = sortEggsByPriority(eggs, priority)
    local target = eggs[1]
    if target and Remotes.RequestCarryAreaEgg then
        local success = Remotes.RequestCarryAreaEgg(target.Uid)
        if success then
            currentStealCount = currentStealCount + 1
            hopThresholdCounter = hopThresholdCounter + 1
            return true
        end
    end
    return false
end

local function runAutoFusePets(force)
    if isUnloaded or isCarryingEgg() then return false end
    if not (Options.AutoFusePets and Options.AutoFusePets.Value) and not force then return false end
    if isFusingNow then return false end
    local rarities = Options.FuseRarities and Options.FuseRarities.Value
    local mutations = Options.FuseMutations and Options.FuseMutations.Value
    local targetGroup = Options.FuseTarget and Options.FuseTarget.Value or "Rarity"
    local keepMutated = Options.FuseKeepMutated and Options.FuseKeepMutated.Value
    local keepEquipped = Options.FuseKeepEquipped and Options.FuseKeepEquipped.Value
    local maxScale = Options.FuseMaxScale and Options.FuseMaxScale.Value or 10
    local keepPerCategory = Options.FuseKeepPerCategory and Options.FuseKeepPerCategory.Value or 0
    isFusingNow = true
    local save = GameModules.Save and GameModules.Save.Get()
    if not save or not save.PetInventory then isFusingNow=false; return false end
    local inventory = save.PetInventory
    local petsByGroup = {}
    for id, pet in pairs(inventory) do
        local groupKey = (targetGroup=="Rarity") and (pet.Rarity or pet.AssetCategory) or (pet.AssetName or pet.AssetCategory)
        if not petsByGroup[groupKey] then petsByGroup[groupKey]={} end
        table.insert(petsByGroup[groupKey], {id=id, data=pet})
    end
    local selectedPets = {}
    for group, list in pairs(petsByGroup) do
        table.sort(list, function(a,b)
            return (resolveRarity(a.data.AssetCategory) or 0) < (resolveRarity(b.data.AssetCategory) or 0)
        end)
        local keepCount = 0
        for _, entry in ipairs(list) do
            local pet = entry.data
            local scale = pet.AssetScale or 1
            local isMutated = pet.Mutations and #pet.Mutations > 0
            local isEquipped = pet.Equipped == true
            if keepMutated and isMutated then keepCount = keepCount + 1 end
            if keepEquipped and isEquipped then keepCount = keepCount + 1 end
            if scale > maxScale then keepCount = keepCount + 1 end
        end
        local toKeep = math.max(keepCount, keepPerCategory)
        for i, entry in ipairs(list) do
            if i > toKeep then table.insert(selectedPets, entry.id) end
        end
    end
    if #selectedPets == 0 then isFusingNow=false; return false end
    for _, id in ipairs(selectedPets) do
        if Options.FuseAutoReveal and Options.FuseAutoReveal.Value then
            if Remotes.ConfirmBriefing then Remotes.ConfirmBriefing() end
            if Remotes.FinishReveal then Remotes.FinishReveal() end
        end
        if Remotes.LoadPet then Remotes.LoadPet(id) end
        if Remotes.StartFuse then Remotes.StartFuse() end
        task.wait(0.5)
        isFusingNow = false
        return true
    end
    isFusingNow = false
    return false
end

local function serverHop(reason)
    if isHopingNow then return end
    isHopingNow = true

    TeleportService:TeleportToPlaceInstance(placeId, jobId, LocalPlayer)

end

local function runServerHop()
    if isUnloaded then return false end
    if not (Options.AutoServerHop and Options.AutoServerHop.Value) then return false end
    local mode = Options.HopMode and Options.HopMode.Value or "No Matching Eggs"
    local threshold = Options.HopThreshold and Options.HopThreshold.Value or 15
    if mode == "No Matching Eggs" then
        local areas = Options.StealZones and Options.StealZones.Value
        local rarities = Options.StealRarities and Options.StealRarities.Value
        local mutations = Options.StealMutations and Options.StealMutations.Value
        local eggs = getAvailableEggs(areas, rarities, mutations, nil)
        if #eggs == 0 then
            serverHop("No eggs")
            return true
        end
    elseif mode == "Timed Interval" then
        if os.clock() - sessionStart > threshold * 60 then
            serverHop("Time")
            return true
        end
    elseif mode == "After Steal Count" then
        if hopThresholdCounter >= threshold then
            hopThresholdCounter = 0
            serverHop("Count")
            return true
        end
    end
    return false
end

-- Main loop for auto functions
local function mainLoop()
    while not isUnloaded do
        task.wait(0.5)
        if stealingEnabled() and not isCarryingEgg() and not isInventoryFull() then
            runAutoSteal()
        end
        if placingEnabled() and not isCarryingEgg() then
            runAutoPlaceEggs()
        end
        if Options.AutoOpenReadyEggs and Options.AutoOpenReadyEggs.Value and not isCarryingEgg() then
            runAutoHatch()
        end
        if Options.AutoTreadmill and Options.AutoTreadmill.Value and not isCarryingEgg() then
            runAutoTreadmill()
        end
        if Options.ResolvePlot and Options.ResolvePlot.Value then
            runAutoEquipBest()
        end
        if Options.AutoFusePets and Options.AutoFusePets.Value and not isCarryingEgg() then
            task.wait(Options.FuseInterval and Options.FuseInterval.Value or 8)
            runAutoFusePets(false)
        end
        if Options.AutoSellPets and Options.AutoSellPets.Value and not isCarryingEgg() then
            runAutoSellPets(false)
        end
        if Options.AutoSellEggs and Options.AutoSellEggs.Value and not isCarryingEgg() then
            runAutoSellEggs(false)
        end
        if Options.AutoUpgrades and Options.AutoUpgrades.Value and not isCarryingEgg() then
            runAutoUpgrades()
        end
        if Options.AutoClaimIndex and Options.AutoClaimIndex.Value then
            runAutoClaimIndex()
        end
        if Options.AutoClaimGroupReward and Options.AutoClaimGroupReward.Value then
            runAutoClaimGroupReward()
        end
        if Options.AutoClaimOffline and Options.AutoClaimOffline.Value then
            runClaimOfflineEarnings()
        end
        if Options.AutoBuyTrail and Options.AutoBuyTrail.Value then
            runAutoBuyTrail()
        end
        if Options.AutoEquipBestTrail and Options.AutoEquipBestTrail.Value then
            runAutoEquipBestTrail()
        end
        if Options.AutoEquipBestGear and Options.AutoEquipBestGear.Value then
            runAutoEquipBestGear()
        end
        if Options.AutoServerHop and Options.AutoServerHop.Value then
            runServerHop()
        end
        if Options.AntiAfk and Options.AntiAfk.Value then
            antiAfkTap()
        end
        if Options.WalkSpeedEnabled and Options.WalkSpeedEnabled.Value then
            local hum = getHumanoid()
            if hum then hum.WalkSpeed = 32 end
        end
        if Options.JumpPowerEnabled and Options.JumpPowerEnabled.Value then
            local hum = getHumanoid()
            if hum then hum.JumpPower = 50 end
        end
        if Options.InfJump and Options.InfJump.Value then
            local hum = getHumanoid()
            if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
        end
        if Options.NoClip and Options.NoClip.Value then
            local root = getRoot()
            if root then root.CanCollide = false end
        end
        if Options.Fly and Options.Fly.Value then
            local hum = getHumanoid()
            local root = getRoot()
            if hum and root then
                hum.PlatformStand = true
                local vel = Vector3.zero
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then vel = vel + CurrentCamera.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then vel = vel - CurrentCamera.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then vel = vel - CurrentCamera.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then vel = vel + CurrentCamera.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vel = vel + Vector3.new(0,1,0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vel = vel - Vector3.new(0,1,0) end
                if vel.Magnitude > 0 then
                    root.CFrame = root.CFrame + vel.Unit * 60 * 0.016
                end
            end
        end
    end
end
