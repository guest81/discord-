local TrackedConnections = {}
local function TrackConnection(connection)
    table.insert(TrackedConnections, connection)
    return connection
end

_G.PhantomX_CleanupFunction = function()
    for _, connection in ipairs(TrackedConnections) do
        pcall(function() connection:Disconnect() end)
    end
    TrackedConnections = {}
end

local function CleanupAllGUIs()
    for _, guiElement in pairs(game.CoreGui:GetChildren()) do
        if guiElement:IsA("ScreenGui") and (
            guiElement.Name:find("Wind") or
            guiElement.Name:find("PhantomX") or
            guiElement.Name:find("PX_")
        ) then
            guiElement:Destroy()
        end
    end
end

local function FormatNumber(number)
    number = tonumber(number) or 0
    local units = {{1e12, "T"}, {1e9, "B"}, {1e6, "M"}, {1e3, "K"}}
    for _, unit in ipairs(units) do
        if number >= unit[1] then
            local value = number / unit[1]
            local text = value < 10 and string.format("%.2f", value) or string.format("%.1f", value)
            text = text:gsub("%.?0+$", "")
            return text .. unit[2]
        end
    end
    return tostring(math.floor(number))
end

-- Layer 1: Client Anti-Cheat Neutralizer
-- Neutralizes anti-cheat validation tables by blocking __newindex
pcall(function()
    if typeof(filtergc) ~= "function" then return end
    if typeof(debug) ~= "table" then return end

    local validatorsFound, validatorFunction = pcall(function()
        return filtergc("function", {Constants = {"gmatch", "GetFullName"}}, true)
    end)

    if not validatorsFound or type(validatorFunction) ~= "function" then return end

    local setMetaFunction = (typeof(setrawmetatable) == "function" and setrawmetatable) or setmetatable
    if not setMetaFunction then return end

    local upvaluesFound, upvalues = pcall(debug.getupvalues, validatorFunction)
    if not upvaluesFound or type(upvalues) ~= "table" then return end

    for _, tableValue in pairs(upvalues) do
        if typeof(tableValue) == "table" then
            pcall(setMetaFunction, tableValue, {__newindex = function() end})
        end
    end
end)

-- Layer 2: Runtime Anti-Cheat Table Freezer
-- Freezes anti-cheat state tables
pcall(function()
    local getGarbageCollector = getgc or (debug and debug.getgc)
    local setMeta = setrawmetatable or setmetatable
    local getMeta = getrawmetatable or getmetatable

    if not getGarbageCollector or not setMeta then return end

    for _, tableObject in ipairs(getGarbageCollector(true)) do
        if typeof(tableObject) == "table" and not (getMeta and getMeta(tableObject)) then
            local selfReferenceFound = false
            for _, tableValue in pairs(tableObject) do
                if tableValue == tableObject then
                    selfReferenceFound = true
                    break
                end
            end
            if selfReferenceFound then
                for _, tableValue in pairs(tableObject) do
                    if typeof(tableValue) == "number" and tableValue >= 1 and tableValue <= 3 and tableObject[tableValue] == nil then
                        pcall(setMeta, tableObject, {__newindex = function() end})
                        break
                    end
                end
            end
        end
    end
end)

-- Layer 3: UGI Constant Wiper
-- Removes "Humanoid" constant from ReplicatedFirst watchdog script
pcall(function()
    local getGarbageCollector = getgc or (debug and debug.getgc)
    local getConstants = getconstants or (debug and debug.getconstants)
    local setConstant = setconstant or (debug and debug.setconstant)

    if not getGarbageCollector or not getConstants or not setConstant then return end

    for _, functionObject in ipairs(getGarbageCollector(true)) do
        if typeof(functionObject) == "function" then
            local infoOk, source = pcall(debug.info, functionObject, "s")
            if infoOk and type(source) == "string"
                and source:find("ReplicatedFirst", 1, true)
                and source:find("UGI", 1, true) then
                local constantsOk, constants = pcall(getConstants, functionObject)
                if constantsOk and type(constants) == "table" then
                    for index, constant in next, constants do
                        if type(constant) == "string" and constant == "Humanoid" then
                            pcall(setConstant, functionObject, index, "")
                        end
                    end
                end
            end
        end
    end
end)

-- Layer 4: Anti-Ragdoll Hook
-- Blocks __newindex on Humanoid.PlatformStand and Motor6D.Enabled
pcall(function()
    if typeof(hookmetamethod) ~= "function" then return end
    if _G.PhantomX_AntiRagdollHooked then return end
    _G.PhantomX_AntiRagdollHooked = true

    local checkCallerFunction = typeof(checkcaller) == "function" and checkcaller or function() return false end
    local newCClosureFunction = typeof(newcclosure) == "function" and newcclosure or function(fn) return fn end

    local originalFunction
    originalFunction = hookmetamethod(game, "__newindex", newCClosureFunction(function(self, key, value)
        if not checkCallerFunction() and typeof(self) == "Instance" then
            if self:IsA("Motor6D") and key == "Enabled" and value == false then
                return nil
            end
            if self:IsA("Humanoid") then
                if key == "PlatformStand" and value == true then
                    return nil
                end
                if key == "Sit" and value == true and _G.PhantomX_AutoStealRunning then
                    return nil
                end
            end
        end
        return originalFunction(self, key, value)
    end))
end)

local function ReadFieldEggs()
    if not EggStateModule or not EggStateModule.ReadFieldEggs then return nil end
    local success, data = pcall(EggStateModule.ReadFieldEggs)
    if not success or type(data) ~= "table" then return nil end
    return data.Records or data
end

local function GetEggValue(record)
    local assetData = AssetsDirectory and AssetsDirectory[record.AssetCategory]
    if not assetData then return 0 end
    local baseRate = tonumber(assetData.EarningRate) or 0
    local scale = tonumber(record.AssetScale) or 1
    return math.max(1, math.floor(baseRate * scale))
end

local function GetEggName(record)
    local assetData = AssetsDirectory and AssetsDirectory[record.AssetCategory]
    if assetData and assetData.DisplayName ~= "" then
        return assetData.DisplayName
    end
    return tostring(record.AssetCategory or "Egg")
end

local function GetEggRarity(record)
    local assetData = AssetsDirectory and AssetsDirectory[record.AssetCategory]
    if assetData and assetData.Rarity then
        return assetData.Rarity.DisplayName or assetData.Rarity._id or "?"
    end
    return "?"
end

local function GetEggRarityColor(record)
    local assetData = AssetsDirectory and AssetsDirectory[record.AssetCategory]
    if assetData and assetData.Rarity and typeof(assetData.Rarity.Color) == "Color3" then
        return assetData.Rarity.Color
    end
    return Color3.fromRGB(200, 200, 220)
end

local function GetEggIcon(record)
    local assetData = AssetsDirectory and AssetsDirectory[record.AssetCategory]
    return (assetData and tostring(assetData.Icon or "")) or ""
end

function StealAnEgg.EvaluateEggList()
    local runtime = StealAnEgg.Runtime
    local currentTime = os.clock()
    if runtime.EggCache and (currentTime - runtime.EggCacheTimestamp) < 3 then
        return runtime.EggCache
    end

    local records = ReadFieldEggs()
    if not records then return runtime.EggCache or {} end

    local rootPart = GetHumanoidRootPart()
    local playerPosition = rootPart and rootPart.Position or Vector3.new()

    local eggList = {}
    for _, record in ipairs(records) do
        if type(record) == "table" and record.Uid
            and typeof(record.BoundsCFrame) == "CFrame"
            and (record.State == "Slot" or record.State == "Dropped") then
            local eggPosition = record.BoundsCFrame.Position
            table.insert(eggList, {
                Uid = record.Uid,
                Name = GetEggName(record),
                Value = GetEggValue(record),
                Rarity = GetEggRarity(record),
                RarityColor = GetEggRarityColor(record),
                Icon = GetEggIcon(record),
                Position = eggPosition,
                Distance = (eggPosition - playerPosition).Magnitude,
                AreaId = record.AreaId,
            })
        end
    end

    table.sort(eggList, function(a, b) return a.Value > b.Value end)

    runtime.EggCache = eggList
    runtime.EggCacheTimestamp = currentTime
    return eggList
end

local function CarryEgg(eggUid)
    if not EggStateModule then return false end

    for _ = 1, 15 do
        local success, result = pcall(function()
            return EggStateModule.CarryFieldEgg(eggUid, nil)
        end)
        if success and result then return true end
        task.wait(0.12)
    end

    local slotsFolder = WorkspaceService:FindFirstChild("AreaEggSlotsClient")
    local eggModel = slotsFolder and slotsFolder:FindFirstChild(eggUid)
    if eggModel then
        for _, descendant in ipairs(eggModel:GetDescendants()) do
            if descendant:IsA("ProximityPrompt") and descendant.Enabled then
                pcall(function()
                    descendant.HoldDuration = 0
                    if typeof(fireproximityprompt) == "function" then
                        fireproximityprompt(descendant, 0)
                        fireproximityprompt(descendant)
                    end
                end)
            end
        end
    end

    task.wait(0.25)
    local success, result = pcall(function()
        return EggStateModule.CarryFieldEgg(eggUid, nil)
    end)
    return success and result
end

local function IsCurrentlyCarryingEgg()
    local character = LocalPlayer.Character
    if character then
        for _, child in ipairs(character:GetChildren()) do
            if child:IsA("Tool") and (
                child:GetAttribute("ItemType") == "AssetEgg" or
                child:GetAttribute("UID")
            ) then
                return true
            end
        end
    end
    return false
end

local function GetBasePosition()
    local basePosition
    pcall(function()
        if PlotStateModule and PlotStateModule.ResolvePlot then
            local plot = PlotStateModule.ResolvePlot()
            if type(plot) == "table" and plot.RespawnPointCFrame then
                basePosition = plot.RespawnPointCFrame.Position
            end
        end
    end)
    if not basePosition then
        local spawnLocation = WorkspaceService:FindFirstChildOfClass("SpawnLocation")
        basePosition = spawnLocation and spawnLocation.Position or Vector3.new(516, 70, -364)
    end
    return basePosition
end

local function TweenToPosition(targetPosition, speed, sessionId)
    local rootPart = GetHumanoidRootPart()
    if not rootPart then return false end

    speed = math.clamp(speed or StealAnEgg.Settings.FlightSpeed, 100, 1000)
    local distance = (targetPosition - rootPart.Position).Magnitude
    if distance < 3 then return true end

    local direction = (targetPosition - rootPart.Position).Unit
    local goalCFrame = CFrame.lookAt(targetPosition, targetPosition + direction)
    local tween = TweenService:Create(
        rootPart,
        TweenInfo.new(distance / speed, Enum.EasingStyle.Linear),
        {CFrame = goalCFrame}
    )

    local heartbeatConnection
    heartbeatConnection = RunService.Heartbeat:Connect(function()
        if sessionId and sessionId ~= StealAnEgg.Runtime.CurrentSession then
            pcall(function() tween:Cancel() end)
            if heartbeatConnection then heartbeatConnection:Disconnect() end
            return
        end
        rootPart.AssemblyLinearVelocity = Vector3.zero
        rootPart.AssemblyAngularVelocity = Vector3.zero
    end)

    tween:Play()
    tween.Completed:Wait()
    if heartbeatConnection then heartbeatConnection:Disconnect() end
    return true
end

local function GetInventoryEggTools()
    local eggTools = {}
    local character = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")

    for _, container in ipairs({character, backpack}) do
        if container then
            for _, tool in ipairs(container:GetChildren()) do
                if tool:IsA("Tool") and (
                    tool:GetAttribute("ItemType") == "AssetEgg" or
                    tool:GetAttribute("UID") ~= nil or
                    tool.Name:lower():find("egg")
                ) then
                    table.insert(eggTools, tool)
                end
            end
        end
    end
    return eggTools
end

function StealAnEgg.PlaceAllEggs()
    local runtime = StealAnEgg.Runtime
    if runtime.IsPlacing then return 0 end
    runtime.IsPlacing = true

    local placeRemote = GetRemote("RF/EggWorld/AskPlaceEgg")
    local placedCount = 0
    if placeRemote then
        for _, tool in ipairs(GetInventoryEggTools()) do
            local eggUid = tool:GetAttribute("UID") or tool:GetAttribute("EggUid") or tool.Name
            local localCFrame = CFrame.new(math.random(-6, 6), 0, math.random(-6, 6))
            pcall(function()
                placeRemote:InvokeServer({Uid = eggUid, LocalCFrame = localCFrame})
            end)
            placedCount = placedCount + 1
            task.wait(0.05)
        end
    end

    runtime.IsPlacing = false
    return placedCount
end

function StealAnEgg.HatchAllEggs()
    local runtime = StealAnEgg.Runtime
    if runtime.IsHatching then return 0 end
    runtime.IsHatching = true

    local hatchRemote = GetRemote("RF/EggWorld/AskHatch")
    local finishRemote = GetRemote("RF/EggWorld/AskFinishHatch")
    local hatchedCount = 0

    if hatchRemote and finishRemote then
        for _, tool in ipairs(GetInventoryEggTools()) do
            local eggUid = tool:GetAttribute("UID") or tool.Name
            pcall(function() hatchRemote:InvokeServer(eggUid) end)
            task.wait(0.9)
            pcall(function() finishRemote:InvokeServer(eggUid) end)
            hatchedCount = hatchedCount + 1
            task.wait(0.05)
        end
    end

    runtime.IsHatching = false
    return hatchedCount
end

-- ============= MAIN STEAL LOOP =============
local stealLoopThread

local function StealLoop()
    local settings = StealAnEgg.Settings
    local runtime = StealAnEgg.Runtime

    while settings.AutoStealEnabled do
        local currentSessionId = runtime.CurrentSession

        task.wait(0.15)

        if not settings.AutoStealEnabled or currentSessionId ~= runtime.CurrentSession then
            break
        end

        if runtime.IsRagdolled then
            task.wait(0.3)
            continue
        end

        local eggList = StealAnEgg.EvaluateEggList()

        -- Pick target egg
        local targetEgg
        if settings.SelectedEggUid then
            for _, egg in ipairs(eggList) do
                if egg.Uid == settings.SelectedEggUid then
                    targetEgg = egg
                    break
                end
            end
        end

        if not targetEgg then
            for _, egg in ipairs(eggList) do
                if not runtime.UnreachableEggs[egg.Uid] then
                    targetEgg = egg
                    break
                end
            end
        end

        if not targetEgg then
            task.wait(2)
            continue
        end

        local rootPart = GetHumanoidRootPart()
        if not rootPart then
            task.wait(1)
            continue
        end

        -- Move to the egg
        TweenToPosition(
            targetEgg.Position + Vector3.new(0, 1.5, 0),
            settings.FlightSpeed,
            currentSessionId
        )

        if currentSessionId ~= runtime.CurrentSession then break end
        if not settings.AutoStealEnabled then break end

        -- Try to carry
        local carrySuccess = CarryEgg(targetEgg.Uid)
        if not carrySuccess then
            runtime.UnreachableEggs[targetEgg.Uid] = os.clock()
            task.wait(0.5)
            continue
        end

        -- Wait for the guard strike window
        task.wait(2.5)

        if currentSessionId ~= runtime.CurrentSession then break end
        if not settings.AutoStealEnabled then break end

        -- Retry carry (guard might have knocked it off)
        for _ = 1, 6 do
            if currentSessionId ~= runtime.CurrentSession then break end
            if IsCurrentlyCarryingEgg() then break end
            CarryEgg(targetEgg.Uid)
            task.wait(0.25)
        end

        -- Return home
        if settings.AutoReturn then
            TweenToPosition(GetBasePosition(), settings.FlightSpeed, currentSessionId)
            task.wait(0.6)
        end

        -- Auto place every 5 steals
        if settings.AutoPlaceEveryFive then
            runtime.BatchStealCount = runtime.BatchStealCount + 1
            if runtime.BatchStealCount >= 5 then
                runtime.BatchStealCount = 0
                task.spawn(function()
                    StealAnEgg.PlaceAllEggs()
                    if settings.AutoHatch then
                        StealAnEgg.HatchAllEggs()
                    end
                end)
            end
        end

        task.wait(0.4)
    end
end

function StealAnEgg.StartAutoSteal()
    if StealAnEgg.Settings.AutoStealEnabled then return end
    StealAnEgg.Settings.AutoStealEnabled = true
    _G.PhantomX_AutoStealRunning = true
    StealAnEgg.Runtime.CurrentSession = StealAnEgg.Runtime.CurrentSession + 1
    stealLoopThread = task.spawn(StealLoop)
end

function StealAnEgg.StopAutoSteal()
    StealAnEgg.Settings.AutoStealEnabled = false
    _G.PhantomX_AutoStealRunning = false
    StealAnEgg.Runtime.CurrentSession = StealAnEgg.Runtime.CurrentSession + 1
    if stealLoopThread then
        pcall(task.cancel, stealLoopThread)
        stealLoopThread = nil
    end
end

function EspSystem.Update(eggList)
    if type(eggList) ~= "table" then return end
    local visibleCount = math.min(#eggList, EspSystem.MaximumCards)

    for index = 1, visibleCount do
        local egg = eggList[index]
        local card = EspSystem.CardPool[index]
        if not card or not card.Anchor.Parent then
            card = CreateEspCard()
            EspSystem.CardPool[index] = card
        end

        card.Anchor.CFrame = CFrame.new(egg.Position)

        local dataKey = table.concat({
            egg.Uid or "",
            tostring(egg.Value or 0),
            tostring(egg.Rarity or ""),
            tostring(egg.Icon or ""),
        }, "|")

        if card.LastKey ~= dataKey then
            card.LastKey = dataKey
            card.NameLabel.Text = tostring(egg.Name or "Egg")
            card.PriceLabel.Text = "$" .. FormatNumber(egg.Value or 0) .. "/s"
            local rarityColor = egg.RarityColor or Color3.fromRGB(150, 150, 180)
            card.RarityStrip.BackgroundColor3 = rarityColor
            card.IconImage.Image = (egg.Icon and egg.Icon ~= "") and egg.Icon or ""
        end

        card.DistanceLabel.Text = math.floor(egg.Distance or 0) .. "m"

        if not card.Billboard.Enabled then
            card.Billboard.Enabled = true
        end
    end

    for index = visibleCount + 1, #EspSystem.CardPool do
        if EspSystem.CardPool[index] and EspSystem.CardPool[index].Billboard then
            EspSystem.CardPool[index].Billboard.Enabled = false
        end
    end
end

function EspSystem.ClearAll()
    for _, card in ipairs(EspSystem.CardPool) do
        pcall(function() card.Anchor:Destroy() end)
    end
    EspSystem.CardPool = {}
    EspSystem.LastHash = ""
end

task.spawn(function()
    while true do
        task.wait(1.5)
        if StealAnEgg.Settings.EspEnabled then
            local eggList = StealAnEgg.EvaluateEggList()

            -- Hash the list to detect changes
            local hash = #eggList .. ":"
            for index = 1, math.min(#eggList, 20) do
                hash = hash .. (eggList[index].Uid or "") .. ","
            end

            if hash ~= EspSystem.LastHash or #EspSystem.CardPool == 0 then
                EspSystem.LastHash = hash
                EspSystem.Update(eggList)
            end
        else
            if #EspSystem.CardPool > 0 then
                EspSystem.ClearAll()
            end
        end
    end
end)
