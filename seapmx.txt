local function getHum()
    local c = LP.Character
    return c and c:FindFirstChildOfClass("Humanoid")

local function isAlive()
    local h = getHum()
    return h and h.Health > 0
end
local function FNum(n)
    n = tonumber(n) or 0
    local u = {{1e12,"T"},{1e9,"B"},{1e6,"M"},{1e3,"K"}}
    for _, x in ipairs(u) do
        if n >= x[1] then
            local v = n/x[1]
            return (v < 10 and string.format("%.2f",v) or string.format("%.1f",v)):gsub("%.?0+$","")..x[2]
        end
    end
    return tostring(math.floor(n))
end

    if not setM then return end

    if not okU or type(ups) ~= "table" then return end
    for _, t in pairs(ups) do
        if typeof(t) == "table" then pcall(setM, t, {__newindex=function() end}) end
    end

pcall(function()
    local g = getgc or (debug and debug.getgc)
    local s = setrawmetatable or setmetatable
    local gm = getrawmetatable or getmetatable
    if not (g and s) then return end
    for _, o in ipairs(g(true)) do
        if typeof(o) == "table" and not (gm and gm(o)) then
            local main = false
            for _, v in pairs(o) do if v == o then main = true; break end end
            if main then
                for _, v in pairs(o) do
                    if typeof(v) == "number" and v >= 1 and v <= 3 and o[v] == nil then
                        pcall(s, o, {__newindex=function() end})
                        break
                    end
                end
            end
        end
    end
end)

    local g = getgc or (debug and debug.getgc)
    local gc = getconstants or (debug and debug.getconstants)
    local sc = setconstant or (debug and debug.setconstant)

    for _, fn in ipairs(g(true)) do

    end
end)

local function SAE_ReadField()
    if not EggStateModule or not EggStateModule.ReadFieldEggs then return nil end
    local ok, d = pcall(EggStateModule.ReadFieldEggs)
    if not ok or type(d) ~= "table" then return nil end
    return d.Records or d
end

local function SAE_GetValue(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if not d then return 0 end
    return math.max(1, math.floor((tonumber(d.EarningRate) or 0) * (tonumber(rec.AssetScale) or 1)))
end
local function SAE_GetName(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.DisplayName ~= "" then return d.DisplayName end
    return tostring(rec.AssetCategory or "Egg")
end
local function SAE_GetRarity(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.Rarity then return d.Rarity.DisplayName or d.Rarity._id or "?" end
    return "?"
end
local function SAE_GetRarityColor(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    if d and d.Rarity and typeof(d.Rarity.Color) == "Color3" then
        return d.Rarity.Color
    end
    return Color3.fromRGB(200, 200, 220)
end
local function SAE_GetIcon(rec)
    local d = AssetsDirectory and AssetsDirectory[rec.AssetCategory]
    return (d and tostring(d.Icon or "")) or ""
end

function SAE.Evaluate()
    local s = SAE.State
    local now = os.clock()
    if s.Cache and (now - s.CacheAt) < 3 then return s.Cache end
    local recs = SAE_ReadField()
    if not recs then return s.Cache or {} end
    local hrp = getHRP()
    local mp = hrp and hrp.Position or Vector3.new()
    local out = {}
    for _, r in ipairs(recs) do
        if type(r) == "table" and r.Uid and typeof(r.BoundsCFrame) == "CFrame"
            and (r.State == "Slot" or r.State == "Dropped") then
            local p = r.BoundsCFrame.Position
            table.insert(out, {
                Uid = r.Uid,
                Name = SAE_GetName(r),
                Value = SAE_GetValue(r),
                Rarity = SAE_GetRarity(r),
                RarityColor = SAE_GetRarityColor(r),
                Icon = SAE_GetIcon(r),
                Position = p,
                Distance = (p - mp).Magnitude,
                Area = r.AreaId,
                Record = r,
            })
        end
    end
    table.sort(out, function(a, b) return a.Value > b.Value end)
    s.Cache, s.CacheAt = out, now
    return out
end

function SAE.Carry(uid)
    if not EggStateModule then return false end
    for _ = 1, 15 do
        local ok, res = pcall(function() return EggStateModule.CarryFieldEgg(uid, nil) end)
        if ok and res then return true end
        task.wait(0.12)
    end
    local slots = WS:FindFirstChild("AreaEggSlotsClient")
    local m = slots and slots:FindFirstChild(uid)
    if m then
        for _, d in ipairs(m:GetDescendants()) do
            if d:IsA("ProximityPrompt") and d.Enabled then
                pcall(function()
                    d.HoldDuration = 0
                    if typeof(fireproximityprompt) == "function" then
                        fireproximityprompt(d, 0)
                        fireproximityprompt(d)
                    end
                end)
            end
        end
    end
    task.wait(0.25)
    local ok, res = pcall(function() return EggStateModule.CarryFieldEgg(uid, nil) end)
    return ok and res
end

function SAE.IsCarrying()
    if SAE.State.Carrying then return true end
    local c = LP.Character
    if c then
        for _, t in ipairs(c:GetChildren()) do
            if t:IsA("Tool") and (t:GetAttribute("ItemType") == "AssetEgg" or t:GetAttribute("UID")) then
                return true
            end
        end
    end
    return false
end

function SAE.GetBase()
    local p
    pcall(function()
        if PlotStateModule and PlotStateModule.ResolvePlot then
            local pl = PlotStateModule.ResolvePlot()
            if type(pl) == "table" and pl.RespawnPointCFrame then
                p = pl.RespawnPointCFrame.Position
            end
        end
    end)
    if not p then
        local s = WS:FindFirstChildOfClass("SpawnLocation")
        p = s and s.Position or Vector3.new(516, 70, -364)
    end
    return p
end

function SAE.TweenTo(target, speed, sid)
    local hrp = getHRP()
    if not hrp then return false end
    speed = math.clamp(speed or SAE.State.FlightSpeed, 100, 1000)
    local d = (target - hrp.Position).Magnitude
    if d < 3 then return true end
    local dir = (target - hrp.Position).Unit
    local goal = CFrame.lookAt(target, target + dir)
    local tw = TweenService:Create(hrp, TweenInfo.new(d/speed, Enum.EasingStyle.Linear), {CFrame = goal})
    local hb
    hb = RunService.Heartbeat:Connect(function()
        if sid and sid ~= SAE.State.Session then
            pcall(function() tw:Cancel() end)
            if hb then hb:Disconnect() end
            return
        end
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    tw:Play()
    tw.Completed:Wait()
    if hb then hb:Disconnect() end
    return true
end

function SAE.GetInventory()
    local out = {}
    local c = LP.Character
    local bp = LP:FindFirstChild("Backpack")
    for _, cont in ipairs({c, bp}) do
        if cont then
            for _, t in ipairs(cont:GetChildren()) do
                if t:IsA("Tool") and (t:GetAttribute("ItemType") == "AssetEgg"
                    or t:GetAttribute("UID") ~= nil or t.Name:lower():find("egg")) then
                    table.insert(out, t)
                end
            end
        end
    end
    return out
end

local stealTask
local function StealLoop()
    local s = SAE.State
    while s.Enabled do
        local sid = s.Session
        task.wait(0.15)
        if not s.Enabled or sid ~= s.Session then break end
        if s.Ragdolled then task.wait(0.3); continue end

        local eggs = SAE.Evaluate()
        local target
        if s.SelectedUid then
            for _, e in ipairs(eggs) do
                if e.Uid == s.SelectedUid then target = e; break end
            end
        end
        if not target then
            for _, e in ipairs(eggs) do
                if not s.Unreachable[e.Uid] then target = e; break end
            end
        end
        if not target then task.wait(2); continue end

        local hrp = getHRP()
        if not hrp then task.wait(1); continue end

        -- Off treadmill
        if s.Treadmilling then
            local d = GetRemote("RF/Treadmill/AskDoff")
            if d then pcall(function() d:InvokeServer() end) end
            s.Treadmilling = false
            task.wait(0.3)
        end

        -- Move to egg
        SAE.TweenTo(target.Position + Vector3.new(0, 1.5, 0), s.FlightSpeed, sid)
        if sid ~= s.Session or not s.Enabled then break end

        -- Try carry
        local ok = SAE.Carry(target.Uid)
        if not ok then
            s.Unreachable[target.Uid] = os.clock()
            task.wait(0.5)
            continue
        end

        -- Wait for guard strike window (like File 1 does)
        task.wait(2.5)
        if sid ~= s.Session or not s.Enabled then break end

        -- Retry carry after guard
        for _ = 1, 6 do
            if sid ~= s.Session then break end
            if SAE.IsCarrying() then break end
            SAE.Carry(target.Uid)
            task.wait(0.25)
        end

        -- Return home
        if s.AutoReturn then
            SAE.TweenTo(SAE.GetBase(), s.FlightSpeed, sid)
            task.wait(0.6)
        end

        -- Auto place every 5
        if s.AutoPlace5 then
            s.BatchCount = s.BatchCount + 1
            if s.BatchCount >= 5 then
                s.BatchCount = 0
                task.spawn(function()
                    SAE.PlaceAll()
                    if s.AutoHatch then SAE.HatchAll() end
                end)
            end
        end

        task.wait(0.4)
    end
end

function SAE.Start(mode)
    local s = SAE.State
    if s.Enabled then
        -- switch mode
        if s.Mode ~= mode then
            s.Mode = mode
            s.Session = s.Session + 1
            task.wait(0.1)
        else
            return
        end
    end
    s.Enabled = true
    s.Mode = mode
    _G.PhantomX_Stealing = true
    s.Session = s.Session + 1
    stealTask = task.spawn(StealLoop)
end

function SAE.Stop()
    local s = SAE.State
    s.Enabled = false
    s.Mode = "NONE"
    _G.PhantomX_Stealing = false
    s.Session = s.Session + 1
    if stealTask then pcall(task.cancel, stealTask) end
end

function ESP.Update(eggs)
    if type(eggs) ~= "table" then return end
    local n = math.min(#eggs, ESP.Max)
    for i = 1, n do
        local e = eggs[i]
        local c = ESP.Pool[i]
        if not c or not c.Anchor.Parent then
            c = MakeCard()
            ESP.Pool[i] = c
        end
        c.Anchor.CFrame = CFrame.new(e.Position)
        local key = table.concat({
            e.Uid or "",
            tostring(e.Value or 0),
            tostring(e.Rarity or ""),
            tostring(e.Icon or ""),
        }, "|")
        if c.LastKey ~= key then
            c.LastKey = key
            c.Name.Text = tostring(e.Name or "Egg")
            c.Price.Text = "$" .. FNum(e.Value or 0) .. "/s"
            local rc = e.RarityColor or Color3.fromRGB(150,150,180)
            c.Strip.BackgroundColor3 = rc
            c.Icon.Image = (e.Icon and e.Icon ~= "") and e.Icon or ""
        end
        c.Dist.Text = math.floor(e.Distance or 0) .. "m"
        if not c.Billboard.Enabled then c.Billboard.Enabled = true end
    end
    for i = n + 1, #ESP.Pool do
        if ESP.Pool[i] and ESP.Pool[i].Billboard then
            ESP.Pool[i].Billboard.Enabled = false
        end
    end
end

function ESP.Clear()
    for _, c in ipairs(ESP.Pool) do
        pcall(function() c.Anchor:Destroy() end)
    end
    ESP.Pool = {}
    ESP.LastHash = ""
end

task.spawn(function()

            local eggs = SAE.Evaluate()
            local hash = #eggs .. ":"
            for i = 1, math.min(#eggs, 20) do
                hash = hash .. (eggs[i].Uid or "") .. ","
            end
            if hash ~= ESP.LastHash or #ESP.Pool == 0 then
                ESP.LastHash = hash
                ESP.Update(eggs)
            end
        else
            if #ESP.Pool > 0 then ESP.Clear() end
        end
    end
end)

local Combat = {
    AuraOn = false,
    Radius = 20,
    Delay = 0.2,
    AntiTrap = true,
}

local auraTask
local function StartAura()
    if auraTask then return end
    auraTask = task.spawn(function()
        while Combat.AuraOn do
            task.wait(Combat.Delay)
            local hrp = getHRP()
            if not hrp then continue end
            local best, bd = nil, math.huge
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LP and p.Character then
                    local t = p.Character:FindFirstChild("HumanoidRootPart")
                    if t then
                        local d = (t.Position - hrp.Position).Magnitude
                        if d < bd and d <= Combat.Radius then bd = d; best = t end
                    end
                end
            end
            if best then
                pcall(function()
                    hrp.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(
                        best.Position.X, hrp.Position.Y, best.Position.Z))
                end)
                local r = GetRemote("RE/BatSwing/Trigger")
                if r then pcall(function() r:FireServer() end) end
            end
        end
        auraTask = nil
    end)
end

local function StopAura()
    Combat.AuraOn = false
    if auraTask then pcall(task.cancel, auraTask) auraTask = nil end
end

local function ApplyClone(ch)
    local h = ch:FindFirstChildOfClass("Humanoid")
    if not h then return end
    local cam = WS.CurrentCamera
    local an = ch:FindFirstChild("Animate")
    local jp, jh, hp, mhp = h.JumpPower, h.JumpHeight, h.Health, h.MaxHealth
    if an and an:IsA("LocalScript") then an.Disabled = true end
    local anim = h:FindFirstChildOfClass("Animator")
    if anim then for _, t in ipairs(anim:GetPlayingAnimationTracks()) do t:Stop(0) end end
    h.Archivable = true
    local nh = h:Clone()
    for _, c in ipairs(nh:GetChildren()) do
        if c:IsA("Animator") then c:Destroy() end
    end
    h.Name = "_Old"
    nh.Name = "Humanoid"
    nh.Parent = ch
    local na = Instance.new("Animator")
    na.Parent = nh
    nh.WalkSpeed = Char.Desired
    nh.JumpPower = jp
    nh.JumpHeight = jh
    nh.MaxHealth = mhp
    nh.Health = math.min(hp, mhp)
    if cam then cam.CameraSubject = nh end
    h:Destroy()
    if an and an:IsA("LocalScript") then
        task.wait()
        an.Disabled = false
        task.defer(function()
            if an.Parent then
                an.Disabled = true
                task.wait()
                an.Disabled = false
            end
        end)
    end
    task.defer(function()
        if nh.Parent then nh:ChangeState(Enum.HumanoidStateType.Running) end
    end)
    ForceWS(nh)
end

local function RestoreJoints()
    for n, d in pairs(joints) do

            local j = Instance.new("Motor6D")
            j.Name = n

            j.Part1 = d.P1
            j.C0 = d.C0
            j.C1 = d.C1
            j.Parent = d.P0
        end
    end
end

local function EnableAntiRejoin()
    if kickOrig then return end
    if typeof(hookmetamethod) ~= "function" then return end
    if typeof(getnamecallmethod) ~= "function" then return end
    local cc = typeof(checkcaller) == "function" and checkcaller or function() return false end
    local old
    old = hookmetamethod(game, "__namecall", function(self, ...)
        if Char.AntiRejoin and not cc() then
            local m = getnamecallmethod()
            if m == "Kick" and self == LP then return nil end
            if (m == "Teleport" or m == "TeleportAsync" or m == "TeleportToPlaceInstance") and self == TeleportService then return nil end
        end
        return old(self, ...)
    end)
    kickOrig = old
end

local function DoPlaceAll()
    local save = GetSave()
    if not save or type(save.EggInventory) ~= "table" then return 0 end
    if not EggStateModule or not EggStateModule.PlantEgg then return 0 end
    local n = 0
    for uid, egg in pairs(save.EggInventory) do
        if type(egg) == "table" and egg.Placement == nil then
            pcall(function()
                EggStateModule.PlantEgg(uid, CFrame.new(math.random(-5,5), 0, math.random(-5,5)))
            end)
            n = n + 1
            task.wait(0.15)
        end
    end
    return n
end

local function DoHatch()
    local save = GetSave()
    if not save or type(save.EggInventory) ~= "table" then return 0 end
    if not EggStateModule then return 0 end
    local n = 0
    for uid, egg in pairs(save.EggInventory) do
        if type(egg) == "table" then
            local ready = true
            if EggStateModule.IsReadyToHatch then
                local ok, r = pcall(EggStateModule.IsReadyToHatch, uid)
                ready = ok and r == true
            end
            if ready and EggStateModule.BeginHatch then
                pcall(function()
                    EggStateModule.BeginHatch(uid)
                    if EggStateModule.FinishHatch then EggStateModule.FinishHatch(uid) end
                end)
                n = n + 1
                task.wait(0.2)
            end
        end
    end
    return n
end

    if not Networking then return 0 end

        local r = Networking:FindFirstChild(name)
        if r then
            local ok, res = pcall(function() return r:InvokeServer() end)
            if ok and res == true then n = n + 1 end
        end
    end
    local c = Networking:FindFirstChild("RF/AwayEarnings/AskCollect")
    if c then pcall(function() c:InvokeServer() end) n = n + 1 end
    return n
end

    pill.BackgroundTransparency = 0.1
    pill.BorderSizePixel = 0
    pill.Parent = sg
    Instance.new("UICorner", pill).CornerRadius = UDim.new(1, 0)

    local sl = Instance.new("TextLabel")
    sl.BackgroundTransparency = 1
    sl.Size = UDim2.fromScale(1, 1)
    sl.Font = Enum.Font.GothamMedium
    sl.TextSize = 12
    sl.TextColor3 = Color3.fromRGB(220, 220, 235)
    sl.Text = T("nLoading")
    sl.Parent = pill

    local start0 = os.clock()
    local fc, lt = 0, tick()
    track(RunService.RenderStepped:Connect(function() fc = fc + 1 end))
    task.spawn(function()
        while true do
            task.wait(1)
            local now = tick()
            local fps = fc / math.max(now - lt, 0.001)
            fc, lt = 0, now
            local ping = 0
            pcall(function() ping = StatsService.Network.ServerStatsItem["Data Ping"]:GetValue() end)
            local secs = math.floor(os.clock() - start0)
            local h = math.floor(secs/3600)
            local m = math.floor(secs/60) % 60
            local s = secs % 60
            local ts = h > 0 and string.format("%d:%02d:%02d", h, m, s)
                or string.format("%02d:%02d", m, s)
            sl.Text = string.format("Time: %s  |  %.0f FPS  |  %.0f ms", ts, fps, ping)
        end
    end)
