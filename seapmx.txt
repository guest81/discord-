                return flat.Magnitude <= part.Size.Y * 0.5 + c

        local function orbitPoint(tpos, reach)
            local h = getHRP()

            if not h or not bh or not bh:IsA("BasePart") then return nil end
            local toH = Vector3.new(bh.Position.X - h.Position.X, 0, bh.Position.Z - h.Position.Z)
            if toH.Magnitude > K.ORBIT_TRIGGER then return nil end
            local rel = Vector3.new(h.Position.X - tpos.X, 0, h.Position.Z - tpos.Z)
            if rel.Magnitude < 1 then rel = Vector3.new(1,0,0) end
            local ang = math.atan2(rel.Z, rel.X)
            local r = math.max(reach, 6)
            local function at(a) return Vector3.new(tpos.X + math.cos(a)*r, h.Position.Y, tpos.Z + math.sin(a)*r) end
            local p1, p2 = at(ang + K.ORBIT_STEP), at(ang - K.ORBIT_STEP)
            local function fh(p) return Vector3.new(p.X - bh.Position.X, 0, p.Z - bh.Position.Z).Magnitude end
            local first, second = p1, p2
            if fh(p2) > fh(p1) then first, second = p2, p1 end
            if groundAt(first) then return first end
            if groundAt(second) then return second end
            return nil
        end

        local function arenaAntiFling()

            h.AssemblyLinearVelocity = Vector3.new(keep.X, math.min(v.Y, 0), keep.Z)

        local function stepToward(goal, dt)
            local h = getHRP()
            local hum = getHum()
            if not h or not hum or not goal then return end
            pcall(arenaAntiFling)
            if groundAt(h.Position) then S.lastSolid = h.Position
            elseif S.lastSolid then
                local back = Vector3.new(S.lastSolid.X - h.Position.X, 0, S.lastSolid.Z - h.Position.Z)
                if back.Magnitude > 1 then
                    local st = math.min(back.Magnitude, math.min(dt, 0.05) * K.MOVE_SPEED, K.MAX_STEP)
                    local nb = h.Position + back.Unit * st
                    local gy = groundAt(nb) or S.lastSolid.Y
                    pcall(function()
                        h.CFrame = CFrame.lookAt(Vector3.new(nb.X, gy, nb.Z), Vector3.new(nb.X, gy, nb.Z) + back.Unit)
                        h.AssemblyLinearVelocity = Vector3.zero
                    end)
                end
                return
            end
            local flat = Vector3.new(goal.pos.X - h.Position.X, 0, goal.pos.Z - h.Position.Z)
            local reach = goal.reach or K.REACH
            local left = flat.Magnitude - reach
            if left <= 1.5 then return end
            local now = os.clock()
            if not S.stuckSince then S.stuckSince = now; S.stuckBest = left end
            if left < (S.stuckBest or left) - 2 then S.stuckBest = left; S.stuckSince = now end
            local dirUse = flat.Unit
            if now - S.stuckSince > K.STUCK_TIME then
                S.stuckFlip = not S.stuckFlip
                local sgn = S.stuckFlip and 1 or -1
                dirUse = Vector3.new(-flat.Unit.Z * sgn, 0, flat.Unit.X * sgn)
                S.stuckSince = now; S.stuckBest = nil
            end
            local step = math.min(left, math.min(dt, 0.05) * K.MOVE_SPEED, K.MAX_STEP)
            local nxt = h.Position + dirUse * step
            if inAnyHazard(nxt, 0) then return end
            local gy = groundAt(Vector3.new(nxt.X, h.Position.Y, nxt.Z))
            if gy then S.arenaFloorY = gy end
            if not gy then
                for _, deg in ipairs({25,50,75,100,125,150}) do
                    for _, sgn in ipairs({1,-1}) do
                        local a = math.rad(deg * sgn)
                        local d = Vector3.new(dirUse.X*math.cos(a) - dirUse.Z*math.sin(a), 0, dirUse.X*math.sin(a) + dirUse.Z*math.cos(a))
                        local g2 = groundAt(Vector3.new(h.Position.X + d.X * step, h.Position.Y, h.Position.Z + d.Z * step))
                        if g2 then dirUse = d; gy = g2; nxt = h.Position + d * step; break end
                    end
                    if gy then break end
                end
                if not gy then return end
            end
            local dest = Vector3.new(nxt.X, gy, nxt.Z)
            pcall(function()
                hum.PlatformStand = false
                h.CFrame = CFrame.lookAt(dest, dest + flat.Unit)
                h.AssemblyLinearVelocity = Vector3.new(0, h.AssemblyLinearVelocity.Y, 0)
                h.AssemblyAngularVelocity = Vector3.zero
            end)
        end

        local dodgeScore
        local function dodge()
            if not S.hazardDodge then S.bossDodge = nil; return false end
            local h = getHRP()
            if not h then return false end
            local parts = hazardParts()
            if #parts == 0 then S.bossDodge = nil; return false end
            local hit
            for _, p in ipairs(parts) do if inHaz(p, h.Position) then hit = p; break end end
            if not hit then S.bossDodge = nil; return false end
            local here = h.Position
            local cands = {}
            if hit:IsA("Part") and hit.Shape == Enum.PartType.Cylinder then
                local want = hit.Size.Y * 0.5 + K.BLACK_HOLE_CLEAR + 4
                for i = 0, K.DODGE_RING_POINTS - 1 do
                    local a = (2 * math.pi / K.DODGE_RING_POINTS) * i
                    table.insert(cands, Vector3.new(hit.Position.X + math.cos(a)*want, here.Y, hit.Position.Z + math.sin(a)*want))
                end
            else
                local rel = hit.CFrame:PointToObjectSpace(here)
                local hf = hit.Size * 0.5
                local c = hazardClear(hit) + 4
                local ox = (rel.X >= 0 and 1 or -1) * (hf.X + c)
                local oz = (rel.Z >= 0 and 1 or -1) * (hf.Z + c)
                table.insert(cands, hit.CFrame:PointToWorldSpace(Vector3.new(rel.X, rel.Y, oz)))
                table.insert(cands, hit.CFrame:PointToWorldSpace(Vector3.new(ox, rel.Y, rel.Z)))
                table.insert(cands, hit.CFrame:PointToWorldSpace(Vector3.new(rel.X, rel.Y, -oz)))
                table.insert(cands, hit.CFrame:PointToWorldSpace(Vector3.new(-ox, rel.Y, rel.Z)))
            end
            local best, bestScore
            for _, p in ipairs(cands) do
                if groundAt(p) and not inAnyHazard(p, 0) then
                    local sc = S.bossAim and Vector3.new(p.X - S.bossAim.X, 0, p.Z - S.bossAim.Z).Magnitude
                        or Vector3.new(p.X - here.X, 0, p.Z - here.Z).Magnitude
                    if not bestScore or sc < bestScore then best, bestScore = p, sc end
                end
            end
            if not best then return true end
            S.bossDodge = {pos = best}
            S.dodges = S.dodges + 1
            return true
        end

        local function stopFight()

        local function startFight()
            if S.fightTask then return end
            S.fightTask = task.spawn(function()
                while S.autoFight do
                    task.wait(K.TICK)
                    if not inBossArena() then continue end
                    pcall(function()
                        local bat = equipBat()
                        if not bat then
                            if os.clock() - (S.fieldBatAt or 0) > 5 then
                                S.fieldBatAt = os.clock()
                                local rf = net and net:FindFirstChild("RF/Codex/AskWearFieldBat")
                                if rf then pcall(function() rf:InvokeServer() end) end
                            end
                            return
                        end
                        local hm = getHum()
                        if hm then
                            local st = hm:GetState()
                            if hm.PlatformStand or st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.PlatformStanding then
                                pcall(function()
                                    hm.PlatformStand = false
                                    hm:ChangeState(Enum.HumanoidStateType.GettingUp)
                                end)
                            end
                        end
                        dodge()
                        local snap
                        pcall(function()
                            local rf = net and net:FindFirstChild("RF/BossEvent/AskSnapshot")
                            if rf then snap = rf:InvokeServer() end
                        end)
                        if snap and tonumber(snap.BossHealth) and snap.BossHealth <= 0 then
                            stopFight()
                            return
                        end
                        local part, kind = bossTarget()
                        if not part then S.bossGoal, S.bossAim = nil, nil; return end
                        local tpos = (typeof(part) == "Vector3") and part or part.Position
                        local h = getHRP()
                        if not h then return end
                        local reach = K.REACH
                        local d = Vector3.new(tpos.X - h.Position.X, 0, tpos.Z - h.Position.Z).Magnitude
                        S.bossAim = tpos
                        if d > reach + 4 then
                            S.bossGoal = {pos = tpos, reach = reach}
                            return
                        end
                        local orb = orbitPoint(tpos, reach)
                        if orb then S.bossGoal = {pos = orb, reach = 0} else S.bossGoal = nil end
                        if inAnyHazard(h.Position, 0) then return end
                        local hh = getHRP()
                        if hh then
                            local flat = Vector3.new(tpos.X - hh.Position.X, 0, tpos.Z - hh.Position.Z)
                            if flat.Magnitude > 0.1 then
                                local wantDir = flat.Unit
                                local haveDir = hh.CFrame.LookVector * Vector3.new(1,0,1)
                                haveDir = haveDir.Magnitude > 0.001 and haveDir.Unit or wantDir
                                if haveDir:Dot(wantDir) < 0.906 then

            if type(eggs) ~= "table" then return end
            local n = math.min(#eggs, MAX)
            for i = 1, n do
                local e = eggs[i]
                local c = pool[i]
                if not c or not c.anc.Parent then c = makeCard(); pool[i] = c end
                c.anc.CFrame = CFrame.new(e.position)
                local key = table.concat({e.uid or "", tostring(e.value or 0), tostring(e.rarity or ""), tostring(e.icon or "")}, "|")
                if c.lastKey ~= key then
                    c.lastKey = key
                    c.title.Text = tostring(e.name or "Egg")
                    c.price.Text = "$" .. fmtNum(e.value or 0) .. "/s"

                if pool[i] and pool[i].bb then pool[i].bb.Enabled = false end
            end
            shown = n
        end

    local OXY = (function()
        local net = RS:FindFirstChild("Packages") and RS.Packages:FindFirstChild("Networking")
        local S = {batAura=false, auraRadius=20, auraDelay=0.2, antiTrap=true}
        local batTask

        local function neutralize()
            local db = WS:FindFirstChild("__DEBRIS")
            if not db then return end
            for _, d in ipairs(db:GetChildren()) do
                if d.Name == "PlayerTrap" and d:GetAttribute("Owner") ~= player.Name then
                    for _, c in ipairs(d:GetDescendants()) do
                        if c:IsA("BasePart") then
                            c.CanTouch = false
                            c.CanQuery = false
                            if c.Name == "Hitbox" then c.CFrame = CFrame.new(0,-999,0) end
                        end
                    end
                end
            end
        end

        local function swing()
            local rem = net and net:FindFirstChild("RE/BatSwing/Trigger")
            if rem then pcall(function() rem:FireServer() end) return true end
            return false
        end

        local function startAura()
            if batTask then return end
            batTask = task.spawn(function()
                while S.batAura do
                    task.wait(S.auraDelay)
                    local hrp = getHRP()
                    if not hrp then continue end
                    local nearest, dist = nil, math.huge
                    for _, pl in ipairs(Players:GetPlayers()) do
                        if pl ~= player and pl.Character then
                            local t = pl.Character:FindFirstChild("HumanoidRootPart")
                            if t then
                                local d = (t.Position - hrp.Position).Magnitude
                                if d < dist and d <= S.auraRadius then dist = d; nearest = t end
                            end
                        end
                    end
                    if nearest then
                        pcall(function()
                            hrp.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(nearest.Position.X, hrp.Position.Y, nearest.Position.Z))
                        end)
                        swing()
                    end
                end
            end)
        end

        local function stopAura()
            S.batAura = false
            if batTask then pcall(task.cancel, batTask) batTask = nil end
        end

        spawn(function()
            while not HUB.dead do
                task.wait(1.5)
                if S.antiTrap then pcall(neutralize) end
            end
        end)

        return {state=S, startAura=startAura, stopAura=stopAura, swing=swing}
    end)()
